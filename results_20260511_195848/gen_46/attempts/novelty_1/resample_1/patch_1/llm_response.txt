def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    # pct_change(lag) uses only past data, no shift needed
    for lag in [1, 3, 6, 12, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Momentum ---
    # Use momentum shifted by 1 to avoid lookahead
    feat['momentum_5'] = (close - close.shift(5)).shift(1)
    feat['momentum_10'] = (close - close.shift(10)).shift(1)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std().shift(1)

    # Add realized volatility ratio shifted by 1
    realized_vol = log_ret.rolling(20).std()
    vol_regime = realized_vol / (realized_vol.rolling(200).mean() + 1e-9)
    feat['vol_regime'] = vol_regime.shift(1)

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = (100 - 100 / (1 + rs)).shift(1)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = ((close - bb_mid) / (2 * bb_std + 1e-9)).shift(1)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    atr_14 = tr.rolling(14).mean()
    feat['atr_14'] = atr_14.shift(1)
    feat['atr_norm'] = (atr_14 / close).shift(1)

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    macd_signal = macd - macd.ewm(span=9, adjust=False).mean()
    feat['macd_signal'] = macd_signal.shift(1)

    # --- Volume ---
    feat['volume_ratio_5'] = (volume / (volume.rolling(5).mean() + 1e-9)).shift(1)
    feat['volume_ratio_20'] = (volume / (volume.rolling(20).mean() + 1e-9)).shift(1)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = ((close - vwap) / (close + 1e-9)).shift(1)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    # Scale trend_strength_60 by 10000 as in original code, shifted by 1
    feat['trend_strength_60'] = ((close / close.shift(60) - 1).abs() * 10000).shift(1)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1).shift(1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1).shift(1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr.shift(1)
        feat['funding_8h_ma'] = fr.rolling(32).mean().shift(1)   # 32 × 15min = 8h

    # --- Feature interactions to increase expressivity ---
    # Interaction: vol_20 * rsi_14
    feat['vol20_rsi14'] = feat['vol_20'] * feat['rsi_14']
    # Interaction: momentum_5 * vol_regime
    feat['mom5_volregime'] = feat['momentum_5'] * feat['vol_regime']

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
