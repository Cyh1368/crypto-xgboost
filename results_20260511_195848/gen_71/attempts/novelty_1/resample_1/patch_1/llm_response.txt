<DESCRIPTION>
The main issue causing the model to collapse predictions toward the mean (very low pred_std_ratio) is excessive regularization and insufficient model expressivity. The current model uses moderate regularization and moderate tree depth but still collapses. Additionally, the feature engineering applies shifts to all features, which is correct to avoid lookahead, but the time-based features use a frequency string '15min' which is correct, but the orderbook features are shifted after applying df.apply, which can cause misalignment or NaNs.

To fix:

1. Increase model expressivity by increasing max_depth and reducing regularization (reg_lambda), and increasing n_estimators slightly, while keeping early stopping.
2. Fix feature engineering to shift orderbook features before applying the lambda to avoid misalignment and ensure no lookahead.
3. Add a few interaction features to increase signal without increasing feature count too much.
4. Keep all features shifted by 1 to avoid lookahead, but ensure the shifting is consistent and no NaNs are introduced unnecessarily.
5. Add normalization to features to help model training.

These changes address the collapse warning by increasing model capacity and ensuring features are properly aligned without lookahead.

# evolve/initial.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 400,          # increased estimators for better learning
    "max_depth": 7,               # deeper trees for more expressivity
    "min_child_weight": 1,        # allow splits on smaller samples
    "reg_alpha": 0.0,             # no L1 regularization
    "reg_lambda": 0.3,            # reduced L2 regularization to increase expressivity
    "subsample": 0.7,
    "colsample_bytree": 0.8,
    "learning_rate": 0.03,        # moderate learning rate
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 30,
}
CLIP_PERCENTILE = 99          # clip targets at this percentile (both tails)
MIN_PRED_STD_RATIO = 0.15     # minimum acceptable pred_std / actual_std
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    for lag in [1, 3, 6, 12, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag).shift(1)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std().shift(1)

    # --- RSI ---
    delta = close.diff()
    for period in [6, 14]:
        gain = delta.clip(lower=0).rolling(period).mean().shift(1)
        loss = (-delta.clip(upper=0)).rolling(period).mean().shift(1)
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean().shift(1)
    bb_std = close.rolling(20).std().shift(1)
    feat['bb_pct'] = (close.shift(1) - bb_mid) / (2 * bb_std + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean().shift(1)
    feat['atr_norm'] = feat['atr_14'] / close.shift(1)

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean().shift(1)
    ema26 = close.ewm(span=26, adjust=False).mean().shift(1)
    macd = ema12 - ema26
    macd_signal = macd - macd.ewm(span=9, adjust=False).mean().shift(1)
    feat['macd_signal'] = macd_signal

    # --- Volume ---
    feat['volume_ratio_5'] = volume.shift(1) / (volume.rolling(5).mean().shift(1) + 1e-9)
    feat['volume_ratio_20'] = volume.shift(1) / (volume.rolling(20).mean().shift(1) + 1e-9)
    vwap = (close * volume).rolling(20).sum().shift(1) / (volume.rolling(20).sum().shift(1) + 1e-9)
    feat['vwap_dev'] = (close.shift(1) - vwap) / (close.shift(1) + 1e-9)

    # --- Session / Time ---
    # Use a consistent freq string '15min'
    shifted_index = df.index.shift(1, freq='15min')
    feat['hour_sin'] = np.sin(2 * np.pi * shifted_index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * shifted_index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * shifted_index.dayofweek / 7)
    feat['is_weekend'] = (shifted_index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    feat['trend_strength_60'] = (close.shift(1) / close.shift(61) - 1).abs() * 10000
    realized_vol = log_ret.rolling(20).std().shift(1)
    feat['vol_regime'] = realized_vol / (realized_vol.rolling(200).mean().shift(1) + 1e-9)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        # Shift bids and asks before apply to avoid misalignment
        bids_shifted = df['bids'].shift(1)
        asks_shifted = df['asks'].shift(1)

        def obi(bids, asks, tau=1):
            try:
                bids_arr = np.array(bids)
                asks_arr = np.array(asks)
                bid_vol = np.sum(bids_arr[:, 1] * np.exp(-tau * np.arange(len(bids_arr))))
                ask_vol = np.sum(asks_arr[:, 1] * np.exp(-tau * np.arange(len(asks_arr))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0

        feat['obi_tau1'] = [obi(b, a, 1) for b, a in zip(bids_shifted, asks_shifted)]
        feat['obi_tau3'] = [obi(b, a, 3) for b, a in zip(bids_shifted, asks_shifted)]

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate'].shift(1)
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()

    # --- Feature interactions to increase expressivity ---
    # Only a few to keep under 80 features
    feat['rsi6_rsi14_diff'] = feat['rsi_6'] - feat['rsi_14']
    feat['vol_5_20_ratio'] = feat['vol_5'] / (feat['vol_20'] + 1e-9)
    feat['ret_1_vol_5'] = feat['ret_1'] * feat['vol_5']
    feat['macd_vol_ratio'] = feat['macd_signal'] / (feat['vol_20'] + 1e-9)

    # Replace inf and drop NA
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()

    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
