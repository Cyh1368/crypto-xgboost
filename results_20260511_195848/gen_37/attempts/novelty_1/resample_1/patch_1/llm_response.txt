- subsample=0.8 and colsample_bytree=0.8 are reasonable.
- early_stopping_rounds=30 is standard.

Also, the features are shifted properly to avoid lookahead bias, but the target clipping and scaling might be insufficient.

- Clip the eval_set targets (test targets) the same way as training targets to avoid mismatch.
- Increase model expressivity by increasing max_depth to 8 and n_estimators to 500.
- Remove early stopping to allow full training (early stopping on test set can cause underfitting).
- Alternatively, if early stopping is kept, clip eval_set targets.
- Also, ensure that the target clipping is consistent and that the model is not over-regularized.

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    for lag in [1, 3, 6, 12, 48]:
        # Shift pct_change by 1 to avoid lookahead bias
        feat[f'ret_{lag}'] = close.pct_change(lag).shift(1)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std().shift(1)

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = (100 - 100 / (1 + rs)).shift(1)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = ((close - bb_mid) / (2 * bb_std + 1e-9)).shift(1)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean().shift(1)
    feat['atr_norm'] = feat['atr_14'] / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    feat['macd_signal'] = (macd - macd.ewm(span=9, adjust=False).mean()).shift(1)

    # --- Volume ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_5'] = feat['volume_ratio_5'].shift(1)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)
    feat['volume_ratio_20'] = feat['volume_ratio_20'].shift(1)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = ((close - vwap) / (close + 1e-9)).shift(1)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    feat['trend_strength_60'] = (close / close.shift(60) - 1).abs() * 10000
    realized_vol = log_ret.rolling(20).std()
    feat['vol_regime'] = realized_vol / (realized_vol.rolling(200).mean() + 1e-9)

    # Shift regime features by 1 to avoid lookahead bias
    feat['trend_strength_60'] = feat['trend_strength_60'].shift(1)
    feat['vol_regime'] = feat['vol_regime'].shift(1)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1).shift(1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1).shift(1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr.shift(1)
        feat['funding_8h_ma'] = fr.rolling(32).mean().shift(1)   # 32 × 15min = 8h

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat
