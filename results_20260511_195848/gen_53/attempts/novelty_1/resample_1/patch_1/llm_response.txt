<DESCRIPTION>
The main issue is that the model's predictions collapse toward the mean, indicated by a very low pred_std_ratio (~0.018), meaning the model is not capturing enough signal and is essentially predicting near the mean target. This suggests insufficient model expressivity or lack of informative features.

Analysis of prior attempts shows that adding momentum features shifted by 1 (to avoid lookahead) helped increase expressivity and reduce collapse warnings. The current program lacks these momentum features, which likely reduces signal strength.

Additionally, the current hyperparameters use a high learning rate (0.1) with no regularization and full subsample/colsample, which can cause unstable training or overfitting, but here the problem is under-expressivity (collapse), so we should reduce learning rate slightly and loosen regularization minimally.

Fix:
- Add momentum features (momentum_5 and momentum_10) shifted by 1 to build_features().
- Reduce learning_rate from 0.1 to 0.05 for more stable training.
- Loosen min_child_weight from 1 to 0.5 to allow more splits.
- Keep reg_alpha=0.0 to avoid L1 regularization but keep reg_lambda=1.0.
- Keep subsample and colsample_bytree at 1.0 to maximize data and features per tree.
- Keep n_estimators=200 to keep training time reasonable.

These changes increase model expressivity and add signal features, addressing the collapse warning and improving pred_std_ratio and directional accuracy.

All features remain shifted properly to avoid lookahead bias, and total feature count remains under 80.

import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 200,       # keep moderate number of trees
    "max_depth": 6,            # keep increased depth for expressivity
    "min_child_weight": 0.5,   # loosened from 1 to 0.5 for more splits
    "reg_alpha": 0.0,          # no L1 regularization
    "reg_lambda": 1.0,         # keep default L2 regularization
    "subsample": 1.0,          # use all data for each tree for expressivity
    "colsample_bytree": 1.0,   # use all features to boost signal
    "learning_rate": 0.05,     # reduced learning rate for more stable training
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 30,
}

CLIP_PERCENTILE = 99          # clip targets at this percentile (both tails)
MIN_PRED_STD_RATIO = 0.15     # minimum acceptable pred_std / actual_std
# EVOLVE-BLOCK-END

# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    # Use pct_change lag returns shifted by 1 to avoid lookahead
    for lag in [1, 3, 6, 12, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag).shift(1)

    # --- Momentum ---
    # Add simple momentum indicators shifted by 1
    feat['momentum_5'] = (close - close.shift(5)).shift(1)
    feat['momentum_10'] = (close - close.shift(10)).shift(1)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std().shift(1)

    # Realized volatility ratio
    realized_vol = log_ret.rolling(20).std()
    vol_regime = realized_vol / (realized_vol.rolling(200).mean() + 1e-9)
    feat['vol_regime'] = vol_regime.shift(1)

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = (100 - 100 / (1 + rs)).shift(1)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = ((close - bb_mid) / (2 * bb_std + 1e-9)).shift(1)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    atr_14 = tr.rolling(14).mean()
    feat['atr_14'] = atr_14.shift(1)
    feat['atr_norm'] = (atr_14 / close).shift(1)

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    macd_signal = macd - macd.ewm(span=9, adjust=False).mean()
    feat['macd_signal'] = macd_signal.shift(1)

    # --- Volume ---
    feat['volume_ratio_5'] = (volume / (volume.rolling(5).mean() + 1e-9)).shift(1)
    feat['volume_ratio_20'] = (volume / (volume.rolling(20).mean() + 1e-9)).shift(1)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = ((close - vwap) / (close + 1e-9)).shift(1)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    feat['trend_strength_60'] = ((close / close.shift(60) - 1).abs() * 10000).shift(1)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1).shift(1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1).shift(1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr.shift(1)
        feat['funding_8h_ma'] = fr.rolling(32).mean().shift(1)   # 32 × 15min = 8h

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
