def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features use only past data (explicitly shifted where needed).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)

    eps = 1e-9

    # --- Core past-only returns / momentum ---
    log_close = np.log(close.clip(lower=eps))
    log_ret1 = log_close.diff()

    for lag in [1, 2, 3, 6, 12, 24, 48]:
        feat[f"ret_{lag}"] = close.pct_change(lag)
        feat[f"logret_{lag}"] = log_close.diff(lag)

    # Multi-horizon momentum stack
    for w in [4, 8, 16, 32, 64]:
        feat[f"mom_{w}"] = close.pct_change(w)
        feat[f"mom_slope_{w}"] = close.pct_change(w).diff()
        feat[f"ret_mean_{w}"] = log_ret1.rolling(w).mean().shift(1)
        feat[f"ret_std_{w}"] = log_ret1.rolling(w).std().shift(1)

    # --- Volatility / compression regime ---
    tr = pd.concat(
        [
            (high - low),
            (high - close.shift(1)).abs(),
            (low - close.shift(1)).abs(),
        ],
        axis=1,
    ).max(axis=1)

    for w in [5, 10, 20, 60]:
        feat[f"atr_{w}"] = tr.rolling(w).mean().shift(1)
        feat[f"rv_{w}"] = log_ret1.rolling(w).std().shift(1)

    feat["atr_norm_20"] = feat["atr_20"] / (close.shift(1) + eps)
    feat["vol_regime"] = feat["rv_20"] / (feat["rv_60"].rolling(20).mean() + eps)
    feat["vol_compress"] = feat["rv_10"] / (feat["rv_60"] + eps)

    # --- Trend / mean reversion position ---
    for w in [12, 24, 48, 96]:
        ma = close.rolling(w).mean().shift(1)
        sd = close.rolling(w).std().shift(1)
        feat[f"z_close_{w}"] = (close - ma) / (sd + eps)
        feat[f"dist_ma_{w}"] = (close / (ma + eps)) - 1.0

    feat["ema_fast"] = close.ewm(span=8, adjust=False).mean().shift(1)
    feat["ema_slow"] = close.ewm(span=21, adjust=False).mean().shift(1)
    feat["ema_trend"] = (feat["ema_fast"] / (feat["ema_slow"] + eps)) - 1.0
    feat["ema_spread"] = (feat["ema_fast"] - feat["ema_slow"]) / (close.shift(1) + eps)

    # MACD-like and slope features
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    signal = macd.ewm(span=9, adjust=False).mean()
    feat["macd_hist"] = (macd - signal).shift(1)
    feat["macd_slope"] = macd.diff().shift(1)

    # --- Candle anatomy / range efficiency ---
    candle_range = (high - low).clip(lower=eps)
    body = (close - close.shift(1)).abs()
    upper_wick = (high - np.maximum(close, close.shift(1))).clip(lower=0)
    lower_wick = (np.minimum(close, close.shift(1)) - low).clip(lower=0)

    feat["range_pct"] = candle_range / (close.shift(1) + eps)
    feat["body_to_range"] = body / (candle_range + eps)
    feat["upper_wick_pct"] = upper_wick / (candle_range + eps)
    feat["lower_wick_pct"] = lower_wick / (candle_range + eps)
    feat["close_pos_range"] = (close - low) / (candle_range + eps)
    feat["efficiency_8"] = (close - close.shift(8)).abs() / (
        tr.rolling(8).sum().shift(1) + eps
    )

    # --- Volume / participation ---
    vol_z_windows = [12, 24, 48]
    for w in vol_z_windows:
        vma = volume.rolling(w).mean().shift(1)
        vsd = volume.rolling(w).std().shift(1)
        feat[f"vol_z_{w}"] = (volume - vma) / (vsd + eps)
        feat[f"vol_ratio_{w}"] = volume / (vma + eps)

    feat["vol_surge"] = feat["vol_ratio_24"] / (feat["vol_ratio_48"].rolling(12).mean() + eps)
    vwap20 = (close * volume).rolling(20).sum().shift(1) / (volume.rolling(20).sum().shift(1) + eps)
    feat["vwap_dev_20"] = (close - vwap20) / (close.shift(1) + eps)

    # --- Robust past-only normalized return features ---
    for w in [16, 32, 64]:
        mu = log_ret1.rolling(w).mean().shift(1)
        sd = log_ret1.rolling(w).std().shift(1)
        feat[f"ret_z_{w}"] = (log_ret1 - mu) / (sd + eps)

    # --- Regime interactions (novel part) ---
    feat["trend_x_vol"] = feat["ema_trend"] * feat["vol_regime"]
    feat["trend_x_volume"] = feat["ema_trend"] * feat["vol_z_24"]
    feat["meanrev_x_vol"] = feat["z_close_24"] * feat["vol_compress"]
    feat["momentum_x_range"] = feat["mom_16"] * feat["range_pct"]
    feat["momentum_x_wick"] = feat["mom_16"] * (feat["upper_wick_pct"] - feat["lower_wick_pct"])
    feat["breakout_pressure"] = feat["close_pos_range"] * feat["vol_surge"] * (feat["range_pct"] / (feat["atr_norm_20"] + eps))

    # --- Session / time ---
    hour = df.index.hour
    dow = df.index.dayofweek
    feat["hour_sin"] = np.sin(2 * np.pi * hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * dow / 7)
    feat["dow_cos"] = np.cos(2 * np.pi * dow / 7)
    feat["is_weekend"] = (dow >= 5).astype(float)

    # --- Higher-level regime feature ---
    trend_48 = close.pct_change(48)
    trend_96 = close.pct_change(96)
    feat["trend_accel"] = (trend_48 - trend_96).shift(1)
    feat["trend_strength"] = trend_48.abs().shift(1)
    feat["trend_to_vol"] = feat["trend_strength"] / (feat["rv_20"] + eps)

    # --- Orderbook imbalance (if available) ---
    if "bids" in df.columns and "asks" in df.columns:
        def book_imbalance(row, tau=1.0):
            try:
                bids = np.array(row["bids"], dtype=float)
                asks = np.array(row["asks"], dtype=float)
                if bids.ndim != 2 or asks.ndim != 2 or bids.shape[1] < 2 or asks.shape[1] < 2:
                    return 0.0
                bvol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                avol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bvol - avol) / (bvol + avol + eps)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: book_imbalance(r, 1.0), axis=1)
        feat["obi_tau3"] = df.apply(lambda r: book_imbalance(r, 3.0), axis=1)
        feat["obi_diff"] = feat["obi_tau1"] - feat["obi_tau3"]

    # --- Funding rate ---
    if "funding_rate" in df.columns:
        fr = pd.to_numeric(df["funding_rate"], errors="coerce")
        feat["funding_rate"] = fr.shift(1)
        feat["funding_ma_8h"] = fr.rolling(32).mean().shift(1)
        feat["funding_z_8h"] = (
            fr.shift(1) - fr.rolling(32).mean().shift(1)
        ) / (fr.rolling(32).std().shift(1) + eps)
        feat["funding_change"] = fr.diff().shift(1)

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()

    # Keep under 80 features
    if feat.shape[1] > 80:
        feat = feat.iloc[:, :80]

    return feat

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build strictly past-only feature matrix. Every feature at index t 
    uses data from index t-1 or earlier.
    """
    feat = pd.DataFrame(index=df.index)
    eps = 1e-9

    # Explicitly shift OHLCV data to ensure no lookahead 
    s_close = df["close"].astype(float).shift(1)
    s_high = df["high"].astype(float).shift(1)
    s_low = df["low"].astype(float).shift(1)
    s_vol = df["volume"].astype(float).shift(1)

    # --- Core Returns ---
    log_close = np.log(s_close.clip(lower=eps))
    log_ret1 = log_close.diff()

    for lag in [1, 2, 4, 8, 16, 32, 64]:
        feat[f"lret_{lag}"] = log_close.diff(lag)
        feat[f"ret_std_{lag}"] = log_ret1.rolling(lag).std()

    # --- Volatility Regimes ---
    tr = pd.concat([
        (s_high - s_low),
        (s_high - s_close.shift(1)).abs(),
        (s_low - s_close.shift(1)).abs(),
    ], axis=1).max(axis=1)

    for w in [15, 60]:
        feat[f"atr_{w}"] = tr.rolling(w).mean()
        feat[f"rv_{w}"] = log_ret1.rolling(w).std()

    feat["vol_regime"] = feat["rv_15"] / (feat["rv_60"] + eps)
    feat["rel_atr"] = feat["atr_15"] / (s_close + eps)

    # --- Normalized Price Positions ---
    for w in [24, 96]:
        ma = s_close.rolling(w).mean()
        sd = s_close.rolling(w).std()
        feat[f"z_close_{w}"] = (s_close - ma) / (sd + eps)
        feat[f"dist_ma_{w}"] = (s_close / (ma + eps)) - 1.0

    # Indicators
    ema12 = s_close.ewm(span=12, adjust=False).mean()
    ema26 = s_close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    signal = macd.ewm(span=9, adjust=False).mean()
    feat["macd_hist"] = macd - signal
    feat["macd_slope"] = macd.diff()

    # RSI
    delta = s_close.diff()
    gain = (delta.where(delta > 0, 0)).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    feat["rsi_14"] = 100 - (100 / (1 + (gain / (loss + eps))))

    # --- Candle Anatomy ---
    s_range = (s_high - s_low).clip(lower=eps)
    feat["body_pct"] = (s_close - s_close.shift(1)).abs() / (s_range + eps)
    feat["close_pos"] = (s_close - s_low) / (s_range + eps)
    feat["efficiency_10"] = (s_close - s_close.shift(10)).abs() / (tr.rolling(10).sum() + eps)
    feat["skew_20"] = log_ret1.rolling(20).skew()

    # --- Volume ---
    vma20 = s_vol.rolling(20).mean()
    feat["vol_ratio"] = s_vol / (vma20 + eps)
    vwap20 = (s_close * s_vol).rolling(20).sum() / (s_vol.rolling(20).sum() + eps)
    feat["vwap_dev"] = (s_close - vwap20) / (s_close + eps)

    # --- Interactions ---
    feat["trend_x_vol"] = (ema12 / (ema26 + eps) - 1.0) * feat["vol_regime"]
    feat["breakout_score"] = feat["close_pos"] * feat["vol_ratio"]

    # --- Time features (No shift needed as they come from timestamps) ---
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24)
    feat["is_weekend"] = (df.index.dayofweek >= 5).astype(float)

    # --- Orderbook Imbalance (SHIFTED) ---
    if "bids" in df.columns and "asks" in df.columns:
        def get_obi(row, tau=1.0):
            try:
                b, a = np.array(row["bids"], dtype=float), np.array(row["asks"], dtype=float)
                bv = np.sum(b[:, 1] * np.exp(-tau * np.arange(len(b))))
                av = np.sum(a[:, 1] * np.exp(-tau * np.arange(len(a))))
                return (bv - av) / (bv + av + eps)
            except: return 0.0
        feat["obi_t1"] = df.apply(lambda r: get_obi(r, 1.0), axis=1).shift(1)
        feat["obi_t3"] = df.apply(lambda r: get_obi(r, 3.0), axis=1).shift(1)

    # --- Funding ---
    if "funding_rate" in df.columns:
        fr = pd.to_numeric(df["funding_rate"], errors="coerce")
        feat["funding_val"] = fr.shift(1)
        feat["funding_z"] = (fr.shift(1) - fr.rolling(32).mean().shift(1)) / (fr.rolling(32).std().shift(1) + eps)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat
