# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 500,
    "max_depth": 7,
    "min_child_weight": 2,
    "reg_alpha": 0.01,
    "reg_lambda": 0.1,
    "subsample": 0.8,
    "colsample_bytree": 0.9,
    "learning_rate": 0.05,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 50,
}

CLIP_PERCENTILE = 99.5       # less aggressive clipping to preserve signal
MIN_PRED_STD_RATIO = 0.15    # minimum acceptable pred_std / actual_std
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Enhanced with regime-conditional, microstructure, and interaction features.
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    open_ = df['open']

    # --- Multi-timeframe Returns (momentum) ---
    for lag in [1, 2, 3, 5, 6, 12, 24, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Volatility (multiple windows) ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 10, 20, 40, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std()

    # --- Volatility regime indicator ---
    vol_20 = log_ret.rolling(20).std()
    vol_200 = log_ret.rolling(200).std()
    feat['vol_regime'] = vol_20 / (vol_200 + 1e-9)
    feat['vol_high'] = (vol_20 > vol_200.rolling(60).quantile(0.75)).astype(float)
    feat['vol_low'] = (vol_20 < vol_200.rolling(60).quantile(0.25)).astype(float)

    # --- RSI (multiple periods) ---
    for period in [6, 14, 28]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        rsi = 100 - 100 / (1 + rs)
        feat[f'rsi_{period}'] = rsi
        # Regime-conditional RSI
        feat[f'rsi_{period}_high_vol'] = rsi * feat['vol_high']
        feat[f'rsi_{period}_low_vol'] = rsi * feat['vol_low']

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)
    feat['bb_width'] = 4 * bb_std / bb_mid

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    atr_14 = tr.rolling(14).mean()
    feat['atr_14'] = atr_14
    feat['atr_norm'] = atr_14 / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    macd_signal = macd.ewm(span=9, adjust=False).mean()
    feat['macd_signal'] = macd - macd_signal
    feat['macd_hist'] = macd.diff()

    # --- Volume features ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)
    feat['volume_ma_ratio'] = volume.rolling(5).mean() / (volume.rolling(20).mean() + 1e-9)
    
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap) / (close + 1e-9)
    
    # Volume-weighted momentum
    feat['vol_weighted_ret'] = (close.pct_change(1) * volume) / (volume.rolling(5).mean() + 1e-9)

    # --- Price action microstructure ---
    feat['high_low_ratio'] = (high - close) / (close - low + 1e-9)
    feat['close_range'] = (close - low) / (high - low + 1e-9)
    feat['open_close_ratio'] = (close - open_) / (high - low + 1e-9)
    
    # Price acceleration
    feat['price_accel'] = close.diff(1) - close.diff(2)
    feat['price_accel_vol'] = feat['price_accel'] * feat['vol_20']

    # --- Session / Time features ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['dow_cos'] = np.cos(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Trend strength ---
    feat['trend_strength_60'] = (close / close.shift(60) - 1).abs() * 10000
    feat['trend_direction_20'] = np.sign(close - close.shift(20))
    feat['trend_direction_60'] = np.sign(close - close.shift(60))

    # --- Mean reversion signals ---
    sma_20 = close.rolling(20).mean()
    sma_60 = close.rolling(60).mean()
    feat['price_sma20_dev'] = (close - sma_20) / (sma_20 + 1e-9)
    feat['price_sma60_dev'] = (close - sma_60) / (sma_60 + 1e-9)
    feat['sma_crossover'] = np.sign(sma_20 - sma_60)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()   # 32 × 15min = 8h
        feat['funding_rate_trend'] = fr.diff(8)

    # --- Feature interactions (regime-conditional) ---
    feat['momentum_vol_interaction'] = feat['ret_6'] * feat['vol_20']
    feat['rsi_vol_interaction'] = feat['rsi_14'] * feat['vol_regime']
    feat['bb_vol_interaction'] = feat['bb_pct'] * feat['vol_20']

    # --- Adaptive features based on recent volatility ---
    recent_vol = log_ret.rolling(10).std()
    feat['adaptive_momentum'] = feat['ret_6'] / (recent_vol + 1e-9)
    feat['adaptive_rsi'] = feat['rsi_14'] / (recent_vol + 1e-9)

    # Clean up
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat
# EVOLVE-BLOCK-END

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets (less aggressive to preserve signal variance)
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale using RobustScaler to better preserve variance while handling outliers
    scaler = RobustScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 50)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
