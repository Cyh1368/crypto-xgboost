<DESCRIPTION>
This version switches from a heavily regularized, low-variance setup to a regime-aware, multi-scale feature stack with a more expressive XGBoost model.  
The key change is to build only lagged/past-safe features from shifted price, volatility, volume, and candle-shape data, then let a larger tree ensemble learn different behaviors across trending, mean-reverting, and volatile regimes.

Compared with the current implementation, this approach should improve directional accuracy by:
- removing collapse-to-mean pressure from overly strong regularization,
- adding richer regime and momentum/mean-reversion signals,
- using past-only normalized features so the model can generalize better,
- increasing model capacity enough to produce a wider prediction distribution without overfitting aggressively.
</DESCRIPTION>

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 900,
    "max_depth": 6,
    "min_child_weight": 2,
    "reg_alpha": 0.0,
    "reg_lambda": 0.5,
    "subsample": 0.85,
    "colsample_bytree": 0.85,
    "colsample_bylevel": 0.9,
    "learning_rate": 0.03,
    "gamma": 0.0,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 50,
}

CLIP_PERCENTILE = 99.8
MIN_PRED_STD_RATIO = 0.08
# EVOLVE-BLOCK-END

# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build a past-only, regime-aware feature matrix.
    All raw series are shifted by 1 bar before feature construction to avoid lookahead.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)

    # Past-only sources
    c1 = close.shift(1)
    h1 = high.shift(1)
    l1 = low.shift(1)
    v1 = volume.shift(1)

    logc = np.log(c1.replace(0, np.nan))
    log_ret = logc.diff()

    # --- Multi-horizon momentum / reversal ---
    for lag in [1, 2, 3, 6, 12, 24, 48]:
        feat[f"ret_{lag}"] = c1.pct_change(lag)

    for w in [4, 8, 16, 32, 64]:
        past_ret = c1.pct_change().shift(0)
        feat[f"mom_z_{w}"] = (past_ret.rolling(w).mean() /
                              (past_ret.rolling(w).std() + 1e-9))
        feat[f"ret_accel_{w}"] = c1.pct_change(1) - c1.pct_change(w)

    # --- Volatility structure ---
    for w in [5, 10, 20, 40, 80]:
        rv = log_ret.rolling(w).std()
        feat[f"rv_{w}"] = rv
        feat[f"rv_ratio_{w}"] = rv / (rv.rolling(max(2, w * 2)).mean() + 1e-9)

    # --- Candle anatomy ---
    rng = (h1 - l1).replace(0, np.nan)
    body = (c1 - close.shift(2))
    feat["candle_body"] = body / (rng + 1e-9)
    feat["upper_wick"] = (h1 - np.maximum(c1, close.shift(2))) / (rng + 1e-9)
    feat["lower_wick"] = (np.minimum(c1, close.shift(2)) - l1) / (rng + 1e-9)
    feat["range_pct"] = rng / (c1.abs() + 1e-9)

    # --- Trend / mean reversion distances ---
    for w in [10, 20, 50, 100]:
        ma = c1.rolling(w).mean()
        sd = c1.rolling(w).std()
        feat[f"z_ma_{w}"] = (c1 - ma) / (sd + 1e-9)
        feat[f"dist_ma_{w}"] = (c1 / (ma + 1e-9)) - 1.0

    # --- Bollinger-type compression / expansion ---
    for w in [20, 40]:
        mid = c1.rolling(w).mean()
        sd = c1.rolling(w).std()
        feat[f"bb_pos_{w}"] = (c1 - mid) / (2.0 * sd + 1e-9)
        feat[f"bb_width_{w}"] = (4.0 * sd) / (mid.abs() + 1e-9)

    # --- True range / ATR style ---
    tr = pd.concat([
        (h1 - l1),
        (h1 - close.shift(2)).abs(),
        (l1 - close.shift(2)).abs(),
    ], axis=1).max(axis=1)
    for w in [7, 14, 28]:
        atr = tr.rolling(w).mean()
        feat[f"atr_{w}"] = atr / (c1.abs() + 1e-9)

    # --- Volume regime ---
    vol_ret = v1.pct_change()
    for w in [5, 20, 60]:
        feat[f"vol_ratio_{w}"] = v1 / (v1.rolling(w).mean() + 1e-9)
        feat[f"vol_z_{w}"] = (v1 - v1.rolling(w).mean()) / (v1.rolling(w).std() + 1e-9)
        feat[f"vol_ret_mean_{w}"] = vol_ret.rolling(w).mean()

    # --- Price-volume interactions ---
    feat["vp_momentum"] = feat["ret_6"] * feat["vol_ratio_20"]
    feat["vp_reversal"] = feat["ret_1"] * feat["vol_z_20"]
    feat["range_volume"] = feat["range_pct"] * feat["vol_ratio_20"]

    # --- Rolling skew / asymmetry ---
    for w in [20, 60]:
        r = log_ret.rolling(w)
        mu = r.mean()
        sd = r.std()
        skew = ((r - mu) ** 3).mean() / (sd**3 + 1e-9)
        feat[f"skew_{w}"] = skew

    # --- Time / session ---
    idx = df.index
    feat["hour_sin"] = np.sin(2 * np.pi * idx.hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * idx.hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * idx.dayofweek / 7)
    feat["dow_cos"] = np.cos(2 * np.pi * idx.dayofweek / 7)
    feat["is_weekend"] = (idx.dayofweek >= 5).astype(float)

    # --- Regime indicators ---
    feat["trend_24"] = c1.pct_change(24)
    feat["trend_96"] = c1.pct_change(96)
    rv20 = log_ret.rolling(20).std()
    rv80 = log_ret.rolling(80).std()
    feat["vol_regime"] = rv20 / (rv80 + 1e-9)
    feat["trend_vs_vol"] = feat["trend_24"].abs() / (rv20 + 1e-9)
    feat["compression"] = feat["bb_width_20"] / (feat["bb_width_40"] + 1e-9)

    # --- Orderbook (if available) ---
    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1.0):
            try:
                bids = np.array(row["bids"], dtype=float)
                asks = np.array(row["asks"], dtype=float)
                if bids.size == 0 or asks.size == 0:
                    return 0.0
                w = np.exp(-tau * np.arange(min(len(bids), len(asks))))
                bid_vol = np.sum(bids[:len(w), 1] * w)
                ask_vol = np.sum(asks[:len(w), 1] * w)
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0

        feat["obi_1"] = df.apply(lambda r: obi(r, 1.0), axis=1)
        feat["obi_3"] = df.apply(lambda r: obi(r, 3.0), axis=1)

    # --- Funding rate (if available) ---
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding_rate"] = fr.shift(1)
        feat["funding_ma_8h"] = fr.shift(1).rolling(32).mean()
        feat["funding_z_8h"] = (fr.shift(1) - fr.shift(1).rolling(32).mean()) / (fr.shift(1).rolling(32).std() + 1e-9)

    # --- Clean up ---
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()

    # Keep feature count bounded
    if feat.shape[1] > 80:
        feat = feat.iloc[:, :80]

    return feat

# DO NOT MODIFY BELOW THIS LINE
# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
