<DESCRIPTION>
The previous model suffered from a 'collapsed' prediction standard deviation (pred_std_ratio < 0.02), indicating it was effectively predicting the mean return for every bar. This often happens when regularization is too high relative to the signal strength or model capacity. 

To improve directional accuracy and signal variance:
1.  **Updated Hyperparameters**: Increased `n_estimators` to 1200 and adjusted `learning_rate` to 0.02. Set `min_child_weight` to 4 and `reg_lambda` to 0.6 to balance the increased complexity while still preventing extreme overfitting.
2.  **Multi-Scale Momentum & Phase**: Added a `trend_stack` and return acceleration across multiple windows (3, 6, 12) to capture transitions in momentum phases.
3.  **Volatility Regime Normalization**: Implemented a `vol_state` based on the ratio of short-term to long-term realized volatility, passed through a `tanh` function to create a bounded regime indicator.
4.  **Bar Microstructure**: Re-introduced `close_loc` (position of close within high-low range) and `wick_imbalance`, which are often strong predictors of short-term mean reversion or breakout continuation.
5.  **Non-Linear Features**: Added `rolling rank` (quantile position) for returns and volume to help the model handle non-stationary distributions.

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    for lag in [1, 3, 6, 12, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std()

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean()
    feat['atr_norm'] = feat['atr_14'] / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    feat['macd_signal'] = macd - macd.ewm(span=9, adjust=False).mean()

    # --- Volume ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap) / (close + 1e-9)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    feat['trend_strength_60'] = (close / close.shift(60) - 1).abs() * 10000
    realized_vol = log_ret.rolling(20).std()
    feat['vol_regime'] = realized_vol / (realized_vol.rolling(200).mean() + 1e-9)

    # --- Momentum & Interaction Features ---
    # Decay-weighted momentum: emphasize recent returns
    ret_1 = close.pct_change(1)
    ret_3 = close.pct_change(3)
    ret_6 = close.pct_change(6)
    feat['momentum_decay_6'] = (ret_1 * 0.5 + ret_3 * 0.3 + ret_6 * 0.2)

    # Volatility-weighted return momentum
    vol_5 = log_ret.rolling(5).std()
    feat['vol_weighted_ret'] = ret_1 / (vol_5 + 1e-6)

    # Return acceleration (momentum of momentum)
    feat['ret_accel_3'] = ret_1 - ret_3 / 3.0

    # Cross-regime signal: vol regime × momentum
    feat['vol_regime_x_momentum'] = feat['vol_regime'] * feat['momentum_decay_6']

    # RSI interaction: RSI slope
    rsi_6 = feat['rsi_6']
    feat['rsi_6_slope'] = rsi_6 - rsi_6.shift(1)

    # MACD × Momentum interaction
    feat['macd_x_momentum'] = feat['macd_signal'] * feat['momentum_decay_6']

    # Volume surge + momentum combination
    feat['volume_momentum'] = feat['volume_ratio_5'] * feat['momentum_decay_6']

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()   # 32 × 15min = 8h

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat

# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close'].astype(float)
    volume = df['volume'].astype(float)
    high = df['high'].astype(float)
    low = df['low'].astype(float)
    open_ = df['open'].astype(float)

    log_ret = np.log(close / close.shift(1))

    # --- Multi-scale Returns & Trend Structure ---
    r1, r3, r6, r12, r24 = [close.pct_change(lag) for lag in [1, 3, 6, 12, 24]]
    feat['ret_1'], feat['ret_3'], feat['ret_6'], feat['ret_12'] = r1, r3, r6, r12
    feat['trend_stack'] = np.sign(r1) + np.sign(r3) + np.sign(r6) + np.sign(r12)
    feat['ret_accel_3'] = r1 - r3 / 3.0
    feat['ret_accel_6'] = r3 - r6 / 2.0
    feat['ret_accel_12'] = r6 - r12 / 2.0

    # --- Volatility Regime & Compression ---
    rv_8 = log_ret.rolling(8).std()
    rv_24 = log_ret.rolling(24).std()
    rv_96 = log_ret.rolling(96).std()
    feat['vol_8_24_ratio'] = rv_8 / (rv_24 + 1e-9)
    feat['vol_state'] = np.tanh((rv_24 / (rv_96 + 1e-9)) - 1.0)
    feat['vol_regime'] = rv_24 / (rv_24.rolling(200).mean() + 1e-9)

    # --- Oscillators & Mean Reversion ---
    for period in [7, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)
    feat['rsi_slope'] = feat['rsi_7'].diff()
    
    ema8 = close.ewm(span=8, adjust=False).mean()
    ema21 = close.ewm(span=21, adjust=False).mean()
    feat['ema_gap'] = (ema8 - ema21) / (close + 1e-9)
    
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap) / (close.rolling(20).std() + 1e-9)

    # --- Candle Microstructure ---
    bar_range = (high - low).replace(0, np.nan)
    feat['candle_body'] = (close - open_) / (bar_range + 1e-9)
    feat['upper_wick'] = (high - np.maximum(close, open_)) / (bar_range + 1e-9)
    feat['lower_wick'] = (np.minimum(close, open_) - low) / (bar_range + 1e-9)
    feat['close_loc'] = (close - low) / (bar_range + 1e-9)
    feat['wick_imbalance'] = feat['lower_wick'] - feat['upper_wick']

    # --- Range & Breakout Indicators ---
    hi_20, lo_20 = high.rolling(20).max(), low.rolling(20).min()
    feat['range_pos_20'] = (close - lo_20) / ((hi_20 - lo_20) + 1e-9)
    feat['range_width_20'] = (hi_20 - lo_20) / (close + 1e-9)

    # --- Volume Dynamics ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_z_20'] = (volume - volume.rolling(20).mean()) / (volume.rolling(20).std() + 1e-9)
    feat['amihud_20'] = (r1.abs() / (volume + 1e-9)).rolling(20).mean() * 1e6

    # --- Rolling Quantiles (Rank) ---
    feat['ret_qpos_60'] = r1.rolling(60).rank(pct=True)
    feat['vol_qpos_60'] = volume.rolling(60).rank(pct=True)

    # --- Time & Session ---
    idx = df.index
    feat['hour_sin'] = np.sin(2 * np.pi * idx.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * idx.hour / 24)
    feat['is_weekend'] = (idx.dayofweek >= 5).astype(float)

    # --- Interactions ---
    feat['momentum_decay_6'] = (r1 * 0.5 + r3 * 0.3 + r6 * 0.2)
    feat['vol_regime_x_momentum'] = feat['vol_regime'] * feat['momentum_decay_6']
    feat['breakout_pressure'] = feat['range_pos_20'] * (feat['volume_ratio_5'] - 1.0)
    feat['vol_adj_mom_24'] = r24 / (rv_24 + 1e-9)

    # --- Orderbook Interaction ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def get_obi(b, a, tau=1):
            try:
                bids, asks = np.array(b), np.array(a)
                w = np.exp(-tau * np.arange(len(bids[:10])))
                bv = np.sum(bids[:len(w), 1] * w)
                av = np.sum(asks[:len(w), 1] * w)
                return (bv - av) / (bv + av + 1e-9)
            except: return 0.0
        feat['obi_tau1'] = [get_obi(b, a, 1) for b, a in zip(df['bids'], df['asks'])]
        feat['obi_lag1'] = feat['obi_tau1'].shift(1)

    # --- Funding Features ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate'].astype(float)
        feat['funding_rate'] = fr
        feat['funding_chg_4'] = fr.diff(4)
        feat['funding_z_32'] = (fr - fr.rolling(32).mean()) / (fr.rolling(32).std() + 1e-9)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat
