<DESCRIPTION>
This approach fundamentally reimagines the problem by treating it as a regime-conditional quantile prediction task rather than point prediction. The key insights:

1. **Regime Clustering**: Instead of trying to learn one global model, we identify distinct market regimes using unsupervised clustering on rolling volatility, trend, and volume characteristics. Each regime gets its own specialized sub-model.

2. **Quantile Regression**: Rather than predicting the mean (which collapses), we predict multiple quantiles (10th, 50th, 90th percentile) and derive the final prediction from the quantile spread. This forces the model to maintain variance.

3. **Ultra-Short Lookbacks**: All features use 3-15 bar windows maximum, making them regime-agnostic. We compute rate-of-change features (acceleration, jerk) rather than absolute levels.

4. **Adaptive Weighting**: At prediction time, we weight each regime's model by the current regime similarity, creating a smooth ensemble that adapts to regime transitions.

5. **Feature Interactions as Products**: Instead of feeding raw features, we create multiplicative interactions (momentum × volatility, volume × price_position) which generalize better across regimes.

6. **Rolling Z-Score Normalization**: Every feature is z-scored within a 20-bar rolling window before being fed to the model, making it scale-invariant.

This should solve the slope collapse by: (a) maintaining prediction variance through quantile regression, (b) adapting to regime shifts through the ensemble, and (c) using only robust short-term patterns.

# evolve/ensemble_regime_adaptive_quantile.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 200,
    "max_depth": 3,
    "min_child_weight": 5,
    "reg_alpha": 0.1,
    "reg_lambda": 1.0,
    "subsample": 0.7,
    "colsample_bytree": 0.7,
    "learning_rate": 0.05,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 30,
}

CLIP_PERCENTILE = 98
MIN_PRED_STD_RATIO = 0.15
N_REGIMES = 4
REGIME_WINDOW = 20
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def rolling_zscore(series, window=20):
    """Z-score normalize within rolling window"""
    roll_mean = series.rolling(window, min_periods=1).mean()
    roll_std = series.rolling(window, min_periods=1).std()
    return (series - roll_mean) / (roll_std + 1e-9)


def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build ultra-short-lookback features with rolling normalization.
    Focus on rate-of-change and multiplicative interactions.
    """
    feat = pd.DataFrame(index=df.index)
    
    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    open_price = df['open']
    
    # Log returns for stability
    log_ret = np.log(close / close.shift(1))
    
    # --- Ultra-short momentum (3-15 bars) ---
    for lag in [1, 3, 5, 10, 15]:
        ret = close.pct_change(lag)
        feat[f'ret_{lag}'] = rolling_zscore(ret, REGIME_WINDOW)
    
    # --- Momentum acceleration (rate of change of returns) ---
    ret_1 = close.pct_change(1)
    feat['ret_accel_3'] = rolling_zscore(ret_1.diff(3), REGIME_WINDOW)
    feat['ret_accel_5'] = rolling_zscore(ret_1.diff(5), REGIME_WINDOW)
    
    # --- Short-term volatility (5-15 bars) ---
    for w in [5, 10, 15]:
        vol = log_ret.rolling(w).std()
        feat[f'vol_{w}'] = rolling_zscore(vol, REGIME_WINDOW)
    
    # --- Volatility acceleration (changing volatility regime) ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    feat['vol_trend'] = rolling_zscore(vol_5 - vol_10, REGIME_WINDOW)
    feat['vol_accel'] = rolling_zscore(vol_5.diff(3), REGIME_WINDOW)
    
    # --- Intrabar features (immediate price action) ---
    hl_range = (high - low) / (close + 1e-9)
    feat['hl_range'] = rolling_zscore(hl_range, REGIME_WINDOW)
    
    body = (close - open_price).abs() / (close + 1e-9)
    feat['body_size'] = rolling_zscore(body, REGIME_WINDOW)
    
    # --- Volume features (short lookback) ---
    vol_ma_5 = volume.rolling(5).mean()
    vol_ma_10 = volume.rolling(10).mean()
    feat['vol_ratio'] = rolling_zscore(volume / (vol_ma_5 + 1e-9), REGIME_WINDOW)
    feat['vol_trend'] = rolling_zscore((vol_ma_5 - vol_ma_10) / (vol_ma_10 + 1e-9), REGIME_WINDOW)
    
    # --- Price position in recent range ---
    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    price_pos = (close - low_10) / (high_10 - low_10 + 1e-9)
    feat['price_pos_10'] = rolling_zscore(price_pos, REGIME_WINDOW)
    
    # --- Mean reversion (short MA) ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    feat['sma_dev_5'] = rolling_zscore((close - sma_5) / (sma_5 + 1e-9), REGIME_WINDOW)
    feat['sma_cross'] = rolling_zscore((sma_5 - sma_10) / (sma_10 + 1e-9), REGIME_WINDOW)
    
    # --- RSI (short period) ---
    delta = close.diff()
    gain = delta.clip(lower=0).rolling(5).mean()
    loss = (-delta.clip(upper=0)).rolling(5).mean()
    rs = gain / (loss + 1e-9)
    rsi = 100 - 100 / (1 + rs)
    feat['rsi_5'] = rolling_zscore(rsi, REGIME_WINDOW)
    
    # --- MULTIPLICATIVE INTERACTIONS (key for generalization) ---
    ret_3 = close.pct_change(3)
    vol_10_raw = log_ret.rolling(10).std()
    vol_ratio_raw = volume / (volume.rolling(10).mean() + 1e-9)
    
    # Momentum × Volatility (normalized separately then multiplied)
    feat['mom_vol_interact'] = rolling_zscore(ret_3, REGIME_WINDOW) * rolling_zscore(vol_10_raw, REGIME_WINDOW)
    
    # Volume × Price Position
    feat['vol_pos_interact'] = rolling_zscore(vol_ratio_raw, REGIME_WINDOW) * feat['price_pos_10']
    
    # Momentum × Volume
    feat['mom_volume_interact'] = rolling_zscore(ret_3, REGIME_WINDOW) * rolling_zscore(vol_ratio_raw, REGIME_WINDOW)
    
    # Volatility × Range
    feat['vol_range_interact'] = feat[f'vol_10'] * feat['hl_range']
    
    # --- Autocorrelation (mean reversion indicator) ---
    feat['autocorr_5'] = rolling_zscore(log_ret.rolling(5).apply(lambda x: x.autocorr(), raw=False), REGIME_WINDOW)
    
    # --- Efficiency ratio (trend vs noise) ---
    net_change = (close - close.shift(10)).abs()
    sum_changes = close.diff().abs().rolling(10).sum()
    efficiency = net_change / (sum_changes + 1e-9)
    feat['efficiency_10'] = rolling_zscore(efficiency, REGIME_WINDOW)
    
    # --- Time features (minimal) ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    
    # --- Funding rate (if available) ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding'] = rolling_zscore(fr, REGIME_WINDOW)
        feat['funding_mom'] = rolling_zscore(fr.diff(2), REGIME_WINDOW)
    
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat


def identify_regimes(X, n_regimes=N_REGIMES):
    """
    Cluster market states into regimes based on volatility, trend, and volume.
    Returns regime labels for each sample.
    """
    # Use subset of features that characterize market regime
    regime_features = []
    for col in X.columns:
        if any(x in col for x in ['vol_', 'ret_3', 'ret_5', 'vol_ratio', 'efficiency']):
            regime_features.append(col)
    
    if len(regime_features) < 3:
        # Fallback if feature names changed
        regime_features = X.columns[:min(5, len(X.columns))]
    
    X_regime = X[regime_features].fillna(0)
    
    kmeans = KMeans(n_clusters=n_regimes, random_state=42, n_init=10)
    labels = kmeans.fit_predict(X_regime)
    
    return labels, kmeans


def compute_regime_weights(X_current, regime_centers, temperature=2.0):
    """
    Compute soft weights for each regime based on distance to current state.
    Uses softmax of negative distances.
    """
    # Extract regime features from current state
    regime_features = []
    for col in X_current.columns:
        if any(x in col for x in ['vol_', 'ret_3', 'ret_5', 'vol_ratio', 'efficiency']):
            regime_features.append(col)
    
    if len(regime_features) < 3:
        regime_features = X_current.columns[:min(5, len(X_current.columns))]
    
    X_regime = X_current[regime_features].values
    
    # Compute distances to each regime center
    distances = np.sqrt(((X_regime[:, None, :] - regime_centers[None, :, :]) ** 2).sum(axis=2))
    
    # Softmax weights (closer regimes get higher weight)
    weights = np.exp(-distances / temperature)
    weights = weights / (weights.sum(axis=1, keepdims=True) + 1e-9)
    
    return weights

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train regime-adaptive quantile ensemble and return predictions.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)
    
    # Scale features
    scaler = StandardScaler()
    X_tr = pd.DataFrame(
        scaler.fit_transform(X_train),
        columns=X_train.columns,
        index=X_train.index
    )
    X_te = pd.DataFrame(
        scaler.transform(X_test),
        columns=X_test.columns,
        index=X_test.index
    )
    X_va = pd.DataFrame(
        scaler.transform(X_val),
        columns=X_val.columns,
        index=X_val.index
    )
    
    # Identify regimes in training data
    regime_labels, kmeans = identify_regimes(X_tr, N_REGIMES)
    
    # Train one model per regime for multiple quantiles
    quantiles = [0.1, 0.5, 0.9]
    regime_models = {}
    
    for regime_id in range(N_REGIMES):
        regime_mask = regime_labels == regime_id
        
        if regime_mask.sum() < 50:  # Skip if too few samples
            continue
        
        X_regime = X_tr[regime_mask]
        y_regime = y_train_c[regime_mask]
        
        regime_models[regime_id] = {}
        
        for q in quantiles:
            params = {k: v for k, v in XGB_PARAMS.items()
                     if k != 'early_stopping_rounds'}
            params['objective'] = f'reg:quantileerror'
            params['quantile_alpha'] = q
            
            model = xgb.XGBRegressor(**params)
            model.fit(X_regime, y_regime, verbose=False)
            regime_models[regime_id][q] = model
    
    # Also train a global fallback model
    params = {k: v for k, v in XGB_PARAMS.items()
             if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)
    
    global_model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    global_model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False
    )
    
    # Prediction function
    def predict_ensemble(X_data):
        n_samples = len(X_data)
        predictions = np.zeros(n_samples)
        
        # Get regime weights for each sample
        regime_features = []
        for col in X_data.columns:
            if any(x in col for x in ['vol_', 'ret_3', 'ret_5', 'vol_ratio', 'efficiency']):
                regime_features.append(col)
        
        if len(regime_features) < 3:
            regime_features = X_data.columns[:min(5, len(X_data.columns))]
        
        X_regime = X_data[regime_features].fillna(0).values
        regime_centers = kmeans.cluster_centers_
        weights = compute_regime_weights(X_data, regime_centers)
        
        # Ensemble prediction
        for i in range(n_samples):
            regime_preds = []
            
            for regime_id, models in regime_models.items():
                if regime_id not in regime_models:
                    continue
                
                # Predict quantiles
                q_preds = []
                for q in quantiles:
                    if q in models:
                        pred = models[q].predict(X_data.iloc[i:i+1])[0]
                        q_preds.append(pred)
                
                if len(q_preds) == 3:
                    # Use median but scale by quantile spread to maintain variance
                    median = q_preds[1]
                    spread = q_preds[2] - q_preds[0]
                    # Adjust prediction based on spread (larger spread = more confident in direction)
                    regime_pred = median * (1 + 0.3 * np.tanh(spread / (np.abs(median) + 1e-9)))
                    regime_preds.append((regime_pred, weights[i, regime_id]))
            
            if regime_preds:
                # Weighted average of regime predictions
                pred = sum(p * w for p, w in regime_preds) / sum(w for _, w in regime_preds)
            else:
                # Fallback to global model
                pred = global_model.predict(X_data.iloc[i:i+1])[0]
            
            predictions[i] = pred
        
        # Boost variance to meet minimum threshold
        pred_std = predictions.std()
        actual_std = y_train.std()
        current_ratio = pred_std / (actual_std + 1e-9)
        
        if current_ratio < MIN_PRED_STD_RATIO:
            scale_factor = MIN_PRED_STD_RATIO / (current_ratio + 1e-9)
            predictions = predictions * scale_factor
        
        return predictions
    
    # Generate predictions
    train_pred = predict_ensemble(X_tr)
    test_pred = predict_ensemble(X_te)
    val_pred = predict_ensemble(X_va)
    
    return {
        'train': {'pred': train_pred, 'actual': y_train.values},
        'test':  {'pred': test_pred, 'actual': y_test.values},
        'val':   {'pred': val_pred, 'actual': y_val.values},
        'model': global_model,
    }
