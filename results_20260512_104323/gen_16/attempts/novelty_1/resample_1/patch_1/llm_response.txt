<DESCRIPTION>
This iteration focuses on solving the 'slope collapse' and 'prediction collapse' issues by implementing strictly regime-invariant features and adjusting model expressivity. 

Key improvements:
1.  **Rolling Z-Score Normalization**: Most features (returns, RSI, volume, volatility) are now normalized using rolling z-scores (window of 30-60 bars). This ensures that a "1-sigma" move in a low-volatility regime is treated similarly to a "1-sigma" move in a high-volatility regime, preventing the model from over-weighting high-volatility training periods.
2.  **Market Character Features**: Added Kaufman Efficiency Ratio and Volatility Ratios (short-term vs long-term) to capture the "state" of the market (trending vs. mean-reverting) independently of price scale.
3.  **Increased Model Expressivity**: To counter the "collapse toward mean" warning, I reduced L2 regularization (`reg_lambda`) and increased the `learning_rate`. I also set `min_child_weight` to a moderate level (12) to balance between robustness and the ability to capture distinct market patterns.
4.  **Feature Interactions**: Retained and refined the `momentum * volatility_regime` interaction, which helps the model scale its directional conviction based on current market activity.
5.  **Target Clipping**: Adjusted `CLIP_PERCENTILE` to 96 to focus the model on the most reliable 96% of the distribution, reducing the influence of extreme tail events that often degrade generalization.

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 220,
    "max_depth": 4,
    "min_child_weight": 12,
    "reg_alpha": 0.1,
    "reg_lambda": 1.5,
    "subsample": 0.7,
    "colsample_bytree": 0.7,
    "learning_rate": 0.09,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 40,
}

CLIP_PERCENTILE = 96          # focus on the core distribution
MIN_PRED_STD_RATIO = 0.18     # target higher prediction variance
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix using rolling z-scores and regime-invariant indicators.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    log_ret = np.log(close / close.shift(1))

    def rolling_z(ser, window, z_window=40):
        """Helper to compute rolling z-score of a series."""
        rolling_mean = ser.rolling(z_window).mean()
        rolling_std = ser.rolling(z_window).std()
        return (ser - rolling_mean) / (rolling_std + 1e-9)

    # --- Returns (Z-scored for regime invariance) ---
    for lag in [1, 3, 5, 10]:
        r = close.pct_change(lag)
        feat[f'ret_z_{lag}'] = rolling_z(r, lag, z_window=40)

    # --- Volatility Dynamics ---
    vol_short = log_ret.rolling(5).std()
    vol_med = log_ret.rolling(15).std()
    vol_long = log_ret.rolling(50).std()
    
    feat['vol_z_5'] = rolling_z(vol_short, 5, z_window=60)
    feat['vol_z_15'] = rolling_z(vol_med, 15, z_window=60)
    feat['vol_regime'] = vol_short / (vol_long + 1e-9)
    feat['vol_trend'] = vol_short / (vol_short.shift(5) + 1e-9) - 1

    # --- RSI (Z-scored) ---
    for period in [7, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        rsi = 100 - 100 / (1 + rs)
        feat[f'rsi_{period}_z'] = rolling_z(rsi, period, z_window=40)

    # --- Efficiency Ratio (Kaufman) ---
    def efficiency_ratio(c, w):
        direction = (c - c.shift(w)).abs()
        volatility = c.diff().abs().rolling(w).sum()
        return direction / (volatility + 1e-9)

    feat['efficiency_10'] = efficiency_ratio(close, 10)
    feat['efficiency_30'] = efficiency_ratio(close, 30)

    # --- Price Location ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)
    
    feat['price_z_20'] = rolling_z(close, 20, z_window=60)

    # --- Volume Dynamics ---
    feat['volume_z_20'] = rolling_z(volume, 20, z_window=60)
    feat['volume_ratio_5_20'] = volume.rolling(5).mean() / (volume.rolling(20).mean() + 1e-9)

    # --- Interactions (Momentum x Regime) ---
    feat['mom_vol_interact'] = feat['ret_z_3'] * feat['vol_regime']
    feat['mom_eff_interact'] = feat['ret_z_5'] * feat['efficiency_10']
    feat['vol_price_pressure'] = feat['volume_z_20'] * feat['price_z_20']

    # --- Candle Structure ---
    feat['hl_range_norm'] = (high - low) / (close + 1e-9)
    feat['hl_range_z'] = rolling_z(feat['hl_range_norm'], 10, z_window=40)
    feat['candle_body_ratio'] = (close - df['open']).abs() / ((high - low) + 1e-9)

    # --- Time Features ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)

    # --- Orderbook & Funding (Conditional) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except: return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)

    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_z'] = rolling_z(fr, 20, z_window=60)
        feat['funding_trend'] = fr.diff(4)

    # --- Trend Consistency ---
    feat['trend_consistency_12'] = np.sign(close.diff()).rolling(12).mean()

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
