<DESCRIPTION>
The strategy focuses on solving the 'slope collapse' and 'prediction collapse' by ensuring features are regime-invariant and the model is forced to learn robust, high-sample patterns.

Key improvements:
1. **Rolling Z-Score Normalization**: Most features (returns, volatility, volume, RSI) are transformed into rolling z-scores (40-bar window). This ensures that a "big move" is relative to current local volatility, making the features generalize better across different market regimes (e.g., from high-vol to low-vol periods).
2. **Short Lookbacks**: Lookback windows are restricted to 5-30 bars to prevent the model from latching onto long-term trends that don't persist into the validation period.
3. **Market Character Features**: Added Kaufman's Efficiency Ratio and Rolling Autocorrelation to measure "trendiness" vs "mean-reversion" character, which helps the model adapt to regime shifts.
4. **Interaction Terms**: Included a Momentum × Volatility Regime interaction. This allows the model to scale its directional conviction based on whether the market is currently in an expanding or contracting volatility state.
5. **Regularization Tuning**: Increased `min_child_weight` and decreased `max_depth` to prevent overfitting to training-specific noise, while keeping `learning_rate` moderate to ensure the model captures enough signal to maintain a healthy predictive slope.

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 300,
    "max_depth": 4,
    "min_child_weight": 12,       # Force more samples per leaf for robustness
    "reg_alpha": 0.5,
    "reg_lambda": 1.5,
    "subsample": 0.75,
    "colsample_bytree": 0.75,
    "learning_rate": 0.07,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 35,
}

CLIP_PERCENTILE = 98          # Maintain target distribution
MIN_PRED_STD_RATIO = 0.22     # Target higher pred variation
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build regime-invariant features using rolling z-scores and short lookbacks.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    
    # Helper for rolling z-score
    def z_score(series, window=40):
        return (series - series.rolling(window).mean()) / (series.rolling(window).std() + 1e-9)

    # --- Returns (Z-scored) ---
    # Short horizons are more likely to generalize across regime shifts
    for lag in [1, 3, 5, 8, 13]:
        ret = close.pct_change(lag)
        feat[f'ret_z_{lag}'] = z_score(ret, 40)

    # --- Volatility Dynamics ---
    log_ret = np.log(close / close.shift(1))
    vol_5 = log_ret.rolling(5).std()
    vol_20 = log_ret.rolling(20).std()
    
    feat['vol_z_5'] = z_score(vol_5, 40)
    feat['vol_regime'] = vol_5 / (vol_20 + 1e-9)
    feat['vol_regime_z'] = z_score(feat['vol_regime'], 40)
    
    # --- Momentum x Volatility Interaction ---
    # Scaled momentum by the current volatility regime
    feat['mom_vol_interact'] = z_score(close.pct_change(5), 40) * feat['vol_regime']

    # --- RSI (Z-scored) ---
    for period in [7, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        rsi = 100 - 100 / (1 + rs)
        feat[f'rsi_z_{period}'] = (rsi - 50) / 25.0 # Simple normalization

    # --- Market Efficiency (Kaufman's ER) ---
    # Measures how "noisy" the price action is
    er_10 = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    feat['er_z_10'] = z_score(er_10, 40)

    # --- Trend Persistence (Autocorrelation) ---
    feat['autocorr_10'] = log_ret.rolling(10).apply(lambda x: x.autocorr(lag=1) if len(x) > 2 else 0, raw=False)

    # --- Volume Dynamics ---
    vol_ratio = volume / (volume.rolling(20).mean() + 1e-9)
    feat['volume_z'] = z_score(vol_ratio, 40)
    
    # --- Price Position ---
    # Bollinger-like position but shorter window
    sma_15 = close.rolling(15).mean()
    std_15 = close.rolling(15).std()
    feat['bb_pos_z'] = (close - sma_15) / (std_15 + 1e-9)
    
    # VWAP deviation
    vwap_15 = (close * volume).rolling(15).sum() / (volume.rolling(15).sum() + 1e-9)
    feat['vwap_dev_z'] = z_score((close - vwap_15) / (close + 1e-9), 40)

    # --- Candle Structure (Regime robust) ---
    hl_range = (high - low) / (close + 1e-9)
    feat['hl_range_z'] = z_score(hl_range, 40)
    feat['body_ratio'] = (close - df['open']).abs() / (high - low + 1e-9)
    feat['wick_ratio_up'] = (high - df[['open', 'close']].max(axis=1)) / (high - low + 1e-9)

    # --- Time Features ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Orderbook & Funding (if available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row):
            try:
                bids, asks = np.array(row['bids']), np.array(row['asks'])
                b_v, a_v = np.sum(bids[:5, 1]), np.sum(asks[:5, 1])
                return (b_v - a_v) / (b_v + a_v + 1e-9)
            except: return 0.0
        feat['obi_z'] = z_score(df.apply(obi, axis=1), 40)

    if 'funding_rate' in df.columns:
        feat['funding_z'] = z_score(df['funding_rate'], 40)
        feat['funding_mom'] = df['funding_rate'].diff(4)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
