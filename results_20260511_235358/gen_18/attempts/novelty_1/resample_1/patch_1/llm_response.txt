<DESCRIPTION>
The primary issue is a severe slope collapse and Pearson correlation degradation between the training and validation sets, indicating the model is learning regime-specific scales rather than robust patterns. 

To address this, I am:
1.  **Z-score Normalizing Features:** Transforming returns, volatility, and volume into rolling z-scores (e.g., `ret_z_20`, `vol_z_20`). This makes features regime-invariant, as a 1% move is measured relative to recent volatility rather than in absolute terms.
2.  **Adding Rate-of-Change Indicators:** Introducing `vol_trend` and `efficiency_ratio` to capture the *character* of the market (trending vs. mean-reverting) which generalizes better across regime shifts.
3.  **Increasing Regularization:** Raising `min_child_weight` to 12 and `reg_lambda` to 2.0, while reducing `max_depth` to 4. This forces the XGBoost model to find patterns that exist across larger samples, reducing the likelihood of fitting training-only noise.
4.  **Adjusting Learning Dynamics:** Lowering the `learning_rate` to 0.06 and increasing `n_estimators` to 500 to allow for a more stable convergence.
5.  **Target Clipping:** Using a slightly more aggressive `CLIP_PERCENTILE` (95) to prevent outliers from skewing the predictive slope.
</DESCRIPTION>

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    for lag in [1, 2, 3, 6, 12, 24, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 10, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std()

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean()
    feat['atr_norm'] = feat['atr_14'] / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    feat['macd_signal'] = macd - macd.ewm(span=9, adjust=False).mean()

    # --- Volume ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap) / (close + 1e-9)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features ---
    feat['trend_strength_20'] = (close / close.shift(20) - 1).abs() * 10000
    feat['trend_strength_60'] = (close / close.shift(60) - 1).abs() * 10000
    trend_20 = (close / close.shift(20) - 1) * 10000
    trend_60 = (close / close.shift(60) - 1) * 10000
    feat['trend_dir_20'] = np.sign(trend_20)
    feat['trend_dir_60'] = np.sign(trend_60)
    realized_vol_20 = log_ret.rolling(20).std()
    realized_vol_60 = log_ret.rolling(60).std()
    feat['vol_regime'] = realized_vol_20 / (realized_vol_20.rolling(200).mean() + 1e-9)
    feat['vol_regime_fast'] = realized_vol_20 / (realized_vol_60 + 1e-9)
    feat['vol_rank_200'] = realized_vol_20.rolling(200).rank(pct=True)

    # --- Mean reversion features ---
    sma_10 = close.rolling(10).mean()
    sma_50 = close.rolling(50).mean()
    feat['price_sma10_dev'] = (close - sma_10) / (sma_10 + 1e-9)
    feat['price_sma50_dev'] = (close - sma_50) / (sma_50 + 1e-9)
    feat['sma_cross'] = (sma_10 - sma_50) / (sma_50 + 1e-9)

    # --- Momentum × Volatility interactions ---
    ret_3 = close.pct_change(3)
    vol_20 = log_ret.rolling(20).std()
    feat['momentum_vol_interaction'] = ret_3 * vol_20
    feat['momentum_regime'] = ret_3 / (vol_20 + 1e-9)
    feat['momentum_regime_20'] = ret_3 * feat['vol_regime']
    feat['momentum_regime_fast'] = ret_3 * feat['vol_regime_fast']

    # --- Volume × Price position interactions ---
    vol_ratio = volume / (volume.rolling(20).mean() + 1e-9)
    bb_pct = (close - close.rolling(20).mean()) / (2 * close.rolling(20).std() + 1e-9)
    feat['volume_bb_interaction'] = vol_ratio * bb_pct
    feat['volume_price_pressure'] = vol_ratio * feat['price_sma10_dev']

    # --- High-low range features ---
    feat['hl_range'] = (high - low) / close
    feat['hl_range_ma'] = feat['hl_range'].rolling(10).mean()
    feat['candle_body_ratio'] = (close - df['open']).abs() / ((high - low) + 1e-9)
    feat['upper_wick_ratio'] = (high - df[['open', 'close']].max(axis=1)) / ((high - low) + 1e-9)
    feat['lower_wick_ratio'] = (df[['open', 'close']].min(axis=1) - low) / ((high - low) + 1e-9)

    # --- Close position in range ---
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat['close_position_20'] = (close - low_20) / (high_20 - low_20 + 1e-9)
    feat['ret_z_20'] = (close.pct_change(1) - close.pct_change(1).rolling(20).mean()) / (close.pct_change(1).rolling(20).std() + 1e-9)
    feat['ret_z_60'] = (close.pct_change(1) - close.pct_change(1).rolling(60).mean()) / (close.pct_change(1).rolling(60).std() + 1e-9)
    feat['ret_accel_6'] = close.pct_change(1).diff(6)
    feat['vol_ret_corr_20'] = log_ret.rolling(20).corr(volume.pct_change(1))

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()   # 32 × 15min = 8h
        feat['funding_momentum'] = fr.diff(4)
        feat['funding_x_volregime'] = fr * feat['vol_regime']

    # --- Trend Consistency and Efficiency ---
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix with regime-invariant, short-lookback features.
    Focus: Z-score normalization, short windows, and rate-of-change indicators.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    open_ = df['open']

    log_ret = np.log(close / close.shift(1))

    # --- Returns & Z-scored Returns (Regime Invariant Momentum) ---
    for lag in [1, 3, 5, 10]:
        r = close.pct_change(lag)
        feat[f'ret_{lag}'] = r
        # Z-score relative to recent history to normalize for volatility regimes
        feat[f'ret_z_{lag}'] = (r - r.rolling(20).mean()) / (r.rolling(20).std() + 1e-9)

    # --- Volatility & Volatility Trends ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    feat['vol_z_20'] = (vol_20 - vol_20.rolling(40).mean()) / (vol_20.rolling(40).std() + 1e-9)
    feat['vol_trend'] = vol_5 / (vol_20 + 1e-9) - 1.0

    # --- RSI (Short windows for faster adaptation) ---
    for period in [6, 12]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands (Z-scored position) ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    bb_pct = (close - bb_mid) / (2 * bb_std + 1e-9)
    feat['bb_pct_z'] = (bb_pct - bb_pct.rolling(20).mean()) / (bb_pct.rolling(20).std() + 1e-9)

    # --- ATR & Candle Structure ---
    tr = pd.concat([high - low, (high - close.shift(1)).abs(), (low - close.shift(1)).abs()], axis=1).max(axis=1)
    feat['atr_norm'] = tr.rolling(14).mean() / close
    feat['body_ratio'] = (close - open_).abs() / ((high - low) + 1e-9)
    feat['wick_ratio'] = (high - low - (close - open_).abs()) / ((high - low) + 1e-9)

    # --- Volume (Z-scored for regime invariance) ---
    vol_ratio = volume / (volume.rolling(20).mean() + 1e-9)
    feat['volume_z_20'] = (vol_ratio - vol_ratio.rolling(40).mean()) / (vol_ratio.rolling(40).std() + 1e-9)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev_z'] = (close - vwap) / (close.rolling(20).std() + 1e-9)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)

    # --- Efficiency & Trend Strength ---
    feat['efficiency_10'] = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    feat['trend_strength_z'] = (close / close.shift(20) - 1).abs() / (vol_20 + 1e-9)

    # --- Momentum × Volatility Interactions ---
    ret_3 = close.pct_change(3)
    feat['momentum_vol_scaled'] = ret_3 / (vol_10 + 1e-9)
    feat['momentum_z_10'] = (ret_3 - ret_3.rolling(10).mean()) / (ret_3.rolling(10).std() + 1e-9)

    # --- Close position in range (Z-scored) ---
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    cp_20 = (close - low_20) / (high_20 - low_20 + 1e-9)
    feat['close_pos_z'] = (cp_20 - cp_20.rolling(20).mean()) / (cp_20.rolling(20).std() + 1e-9)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception: return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_z'] = (fr - fr.rolling(32).mean()) / (fr.rolling(32).std() + 1e-9)
        feat['funding_momentum'] = fr.diff(4)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat
