<DESCRIPTION>
The structural redesign focuses on transforming raw market indicators into "Regime-Invariant" features using rolling Z-score normalization. This addresses the slope collapse problem by ensuring that a feature value (e.g., a 1% return or a volume spike) is interpreted relative to the recent market context rather than absolute levels, which change across regimes. 

Key changes:
1.  **Rolling Z-Score Normalization:** Most features (returns, volume, volatility, RSI) are transformed into rolling Z-scores (window 20-60). This forces the model to learn patterns based on relative extremes.
2.  **Robustness Regularization:** Increased `min_child_weight` and reduced `max_depth` to prevent the model from fitting to regime-specific noise.
3.  **Efficiency and Acceleration:** Added Kaufman Efficiency Ratio and momentum acceleration features to capture the "quality" of price moves.
4.  **Interaction Terms:** Explicitly calculated momentum-volatility interactions to help the model distinguish between high-volatility breakouts and low-volatility drift.
5.  **Lookback Compression:** Shifted focus to shorter lookback windows (5-30 bars) to ensure features adapt quickly to the regime shifts observed between the test and validation sets.

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 500,
    "max_depth": 4,             # Shallower trees for better generalization
    "min_child_weight": 20,     # High weight to prevent fitting small-sample noise
    "reg_alpha": 0.5,           # L1 regularization
    "reg_lambda": 2.0,          # L2 regularization
    "subsample": 0.7,
    "colsample_bytree": 0.7,
    "learning_rate": 0.08,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 40,
}

CLIP_PERCENTILE = 98          # clip targets at this percentile
MIN_PRED_STD_RATIO = 0.18     # target ratio for prediction spread
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build regime-invariant features using rolling Z-scores and efficiency ratios.
    """
    feat = pd.DataFrame(index=df.index)
    
    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    
    def rolling_z(series, window):
        roll = series.rolling(window)
        return (series - roll.mean()) / (roll.std() + 1e-9)

    # --- 1. Regime-Invariant Returns (Z-scored) ---
    log_ret = np.log(close / close.shift(1))
    for lag in [1, 3, 5, 10]:
        ret_val = close.pct_change(lag)
        feat[f'z_ret_{lag}'] = rolling_z(ret_val, 40)

    # --- 2. Volatility Dynamics ---
    vol_short = log_ret.rolling(10).std()
    vol_long = log_ret.rolling(50).std()
    feat['vol_ratio'] = vol_short / (vol_long + 1e-9)
    feat['z_vol_10'] = rolling_z(vol_short, 60)
    
    # --- 3. Efficiency and Trend Quality ---
    # Kaufman Efficiency Ratio: net change / sum of absolute changes
    for w in [10, 30]:
        net_chg = (close - close.shift(w)).abs()
        sum_chg = close.diff().abs().rolling(w).sum()
        feat[f'efficiency_{w}'] = net_chg / (sum_chg + 1e-9)

    # --- 4. Momentum Acceleration ---
    feat['mom_accel'] = close.pct_change(3).diff(3)
    feat['z_mom_accel'] = rolling_z(feat['mom_accel'], 40)

    # --- 5. Volume Profile ---
    v_ma_short = volume.rolling(10).mean()
    v_ma_long = volume.rolling(50).mean()
    feat['rel_volume'] = v_ma_short / (v_ma_long + 1e-9)
    feat['z_volume'] = rolling_z(volume, 40)

    # --- 6. Price Location (Oscillators) ---
    # RSI Z-scored
    for p in [14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(p).mean()
        loss = (-delta.clip(upper=0)).rolling(p).mean()
        rs = gain / (loss + 1e-9)
        rsi = 100 - 100 / (1 + rs)
        feat[f'z_rsi_{p}'] = rolling_z(rsi, 40)

    # Distance from MA Z-scored
    for w in [20, 50]:
        ma = close.rolling(w).mean()
        dist = (close - ma) / (ma + 1e-9)
        feat[f'z_dist_ma_{w}'] = rolling_z(dist, 60)

    # --- 7. Range and Intraday Structure ---
    hl_range = (high - low) / (close + 1e-9)
    feat['z_hl_range'] = rolling_z(hl_range, 40)
    
    # Candle body relative to range
    body = (close - df['open']).abs()
    feat['body_ratio'] = body / (high - low + 1e-9)
    feat['z_body_ratio'] = rolling_z(feat['body_ratio'], 40)

    # --- 8. Feature Interactions ---
    # Momentum scaled by volatility regime
    feat['mom_vol_inter'] = feat['z_ret_3'] * feat['vol_ratio']
    feat['vol_volume_inter'] = feat['z_vol_10'] * feat['rel_volume']

    # --- 9. Time Features (Cyclical) ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)

    # --- 10. External Data (Funding) ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['z_funding'] = rolling_z(fr, 64) # ~16 hours
        feat['funding_mom'] = fr.diff(4)
        feat['funding_ret_inter'] = feat['z_funding'] * feat['z_ret_5']

    # --- 11. Orderbook (Vectorized) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        # Simplified vectorized OBI
        try:
            # Assuming bids/asks are lists of [price, size]
            # We take the top level if possible
            bid_size = df['bids'].apply(lambda x: x[0][1] if len(x)>0 else 0)
            ask_size = df['asks'].apply(lambda x: x[0][1] if len(x)>0 else 0)
            feat['obi_top'] = (bid_size - ask_size) / (bid_size + ask_size + 1e-9)
        except:
            pass

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.fillna(method='ffill').dropna()
    
    # Final check to keep feature count under 80
    if len(feat.columns) > 80:
        feat = feat.iloc[:, :80]
        
    return feat
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# DO NOT MODIFY BELOW THIS LINE

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
