<DESCRIPTION>
The primary challenge in the previous iterations was a "slope collapse" where the model's predictive power evaporated during regime shifts (test to validation). This implementation addresses this by:
1. **Regime-Invariant Features**: Shifting from absolute returns and volatility to rolling Z-scores and ratios. This ensures that a "large move" is defined relative to the current local volatility rather than an absolute percentage, which generalizes better across high and low volatility regimes.
2. **Robustness over Complexity**: Reducing `max_depth` to 3 and significantly increasing `min_child_weight` to 40. This forces XGBoost to find patterns that are supported by a large number of samples, preventing it from fitting noise in the training set that doesn't persist.
3. **Efficiency and Momentum Interactions**: Using Kaufman's Efficiency Ratio to distinguish between trending and range-bound markets, and interacting momentum with volatility ratios to capture how price action changes when volatility accelerates.
4. **Target Noise Reduction**: Tightening the `CLIP_PERCENTILE` to 95 to focus the model on the bulk of the distribution rather than being skewed by extreme outliers that are often regime-specific.

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 1000,         # Rely on early stopping
    "max_depth": 3,              # Shallow trees generalize better in finance
    "min_child_weight": 40,      # Force robust patterns with many samples
    "reg_alpha": 0.5,            # L1 regularization for feature selection
    "reg_lambda": 2.0,           # L2 regularization for stability
    "subsample": 0.7,            # Prevent overfitting to specific time periods
    "colsample_bytree": 0.7,     # Prevent overfitting to specific features
    "learning_rate": 0.08,       # Slower learning for better generalization
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 50,
}

CLIP_PERCENTILE = 95          # Focus on the 95% most common moves
MIN_PRED_STD_RATIO = 0.18     # Target higher prediction variance
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build a robust, regime-invariant feature set using rolling Z-scores,
    efficiency ratios, and momentum-volatility interactions.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    
    # Helper for rolling Z-score
    def zscore(series, window):
        return (series - series.rolling(window).mean()) / (series.rolling(window).std() + 1e-9)

    # --- Returns & Z-scored Returns ---
    ret_1 = close.pct_change(1)
    feat['ret_1'] = ret_1
    feat['ret_4'] = close.pct_change(4)
    feat['ret_12'] = close.pct_change(12)
    
    feat['zret_1_30'] = zscore(ret_1, 30)
    feat['zret_4_30'] = zscore(feat['ret_4'], 30)
    feat['zret_1_100'] = zscore(ret_1, 100)

    # --- Volatility & Regime ---
    log_ret = np.log(close / close.shift(1))
    vol_5 = log_ret.rolling(5).std()
    vol_20 = log_ret.rolling(20).std()
    vol_60 = log_ret.rolling(60).std()
    
    feat['vol_ratio_5_20'] = vol_5 / (vol_20 + 1e-9)
    feat['vol_ratio_20_60'] = vol_20 / (vol_60 + 1e-9)
    feat['vol_z_30'] = zscore(vol_20, 100)

    # --- Efficiency Ratio (Kaufman) ---
    # Measures trend strength vs noise
    def get_er(window):
        net_chg = (close - close.shift(window)).abs()
        sum_abs_chg = ret_1.abs().rolling(window).sum()
        return net_chg / (sum_abs_chg + 1e-9)

    feat['er_10'] = get_er(10)
    feat['er_30'] = get_er(30)

    # --- RSI (Standard) ---
    def get_rsi(window):
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(window).mean()
        loss = (-delta.clip(upper=0)).rolling(window).mean()
        rs = gain / (loss + 1e-9)
        return 100 - 100 / (1 + rs)

    feat['rsi_14'] = get_rsi(14)
    feat['rsi_30'] = get_rsi(30)

    # --- Relative Price Position ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)
    
    # Range Z-score
    hl_range = (high - low) / (close + 1e-9)
    feat['range_z_30'] = zscore(hl_range, 30)

    # --- Volume Dynamics ---
    log_vol = np.log(volume + 1)
    feat['volume_z_30'] = zscore(log_vol, 30)
    feat['volume_z_100'] = zscore(log_vol, 100)

    # --- Time Features ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['dow_cos'] = np.cos(2 * np.pi * df.index.dayofweek / 7)

    # --- Interactions (The "Secret Sauce") ---
    # Momentum scaled by volatility regime
    feat['mom_vol_adj'] = feat['ret_4'] * feat['vol_ratio_5_20']
    # RSI scaled by trend efficiency
    feat['rsi_er_adj'] = (feat['rsi_14'] - 50) * feat['er_10']
    # Volume pressure
    feat['vol_price_z'] = feat['volume_z_30'] * feat['zret_1_30']

    # --- Funding Rate (if available) ---
    if 'funding_rate' in df.columns:
        feat['funding_z_32'] = zscore(df['funding_rate'], 32)
        feat['funding_ret_corr'] = df['funding_rate'].rolling(16).corr(ret_1)

    # --- Orderbook (if available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row):
            try:
                b_vol = np.sum([b[1] for b in row['bids'][:5]])
                a_vol = np.sum([a[1] for a in row['asks'][:5]])
                return (b_vol - a_vol) / (b_vol + a_vol + 1e-9)
            except: return 0.0
        feat['obi_simple'] = df.apply(obi, axis=1)

    # --- Clean up ---
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.fillna(method='ffill').fillna(0)
    
    # Limit features to 80 (currently ~30)
    return feat

# DO NOT MODIFY BELOW THIS LINE
def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 50)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
