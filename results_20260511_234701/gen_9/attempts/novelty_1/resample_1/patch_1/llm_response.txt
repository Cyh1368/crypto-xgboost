<DESCRIPTION>
This edit targets the core failure mode: slope collapse on the validation regime. It removes several long-horizon, regime-specific features and replaces them with shorter-window, rolling z-scored features that are more invariant to volatility regime shifts. It also adds rate-of-change regime features and a stronger momentum × volatility interaction, which should help the model retain predictive slope in the most recent period without inflating feature count.

On the model side, I reduce depth and increase child-weight / regularization slightly while using more estimators with a lower learning rate. This typically improves out-of-sample stability for XGBoost on noisy crypto perp data and discourages the model from memorizing regime-specific splits. The target is to increase val slope and prediction dispersion while keeping DA near or above the current floor.

    volume = df['volume']
    high = df['high']
    low = df['low']

        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 10, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std()

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean()
    feat['atr_norm'] = feat['atr_14'] / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    feat['macd_signal'] = macd - macd.ewm(span=9, adjust=False).mean()

    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)

    feat['vwap_dev'] = (close - vwap) / (close + 1e-9)

    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)

    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    open_ = df['open']

    log_ret = np.log(close / close.shift(1))
    ret_1 = close.pct_change(1)

    def rolling_z(x, w):
        mu = x.rolling(w).mean()
        sd = x.rolling(w).std()
        return (x - mu) / (sd + 1e-9)

    # --- Short-horizon returns / normalized momentum ---
    for lag in [1, 2, 3, 6, 12, 24]:
        r = close.pct_change(lag)
        feat[f'ret_{lag}'] = r
        feat[f'ret_z_{lag}'] = rolling_z(r, 20)

    feat['ret_accel_3'] = ret_1.diff(3)
    feat['ret_accel_6'] = ret_1.diff(6)

    # --- Short-horizon volatility with regime normalization ---
    for w in [5, 10, 20]:
        vol = log_ret.rolling(w).std()
        feat[f'vol_{w}'] = vol
        feat[f'vol_z_{w}'] = rolling_z(vol, 60)

    realized_vol_10 = log_ret.rolling(10).std()
    realized_vol_20 = log_ret.rolling(20).std()
    vol_ref = realized_vol_20.rolling(100).mean()
    feat['vol_regime'] = realized_vol_20 / (vol_ref + 1e-9)
    feat['vol_regime_chg'] = feat['vol_regime'].pct_change(4)
    feat['vol_trend_20'] = rolling_z(realized_vol_20, 60) - rolling_z(realized_vol_20.shift(4), 60)

    # --- Trend / efficiency at short windows ---
    for w in [8, 16, 24]:
        trend = close / close.shift(w) - 1
        feat[f'trend_dir_{w}'] = np.sign(trend)
        feat[f'trend_strength_{w}'] = trend.abs() * 10000
        feat[f'trend_z_{w}'] = rolling_z(trend, 40)

    feat['efficiency_10'] = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    feat['efficiency_trend'] = feat['efficiency_10'] - feat['efficiency_20']

    # --- Oscillators, normalized to recent regime ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        rsi = 100 - 100 / (1 + rs)
        feat[f'rsi_{period}'] = rsi
        feat[f'rsi_z_{period}'] = rolling_z(rsi, 60)

    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    bb_pct = (close - bb_mid) / (2 * bb_std + 1e-9)
    feat['bb_pct'] = bb_pct
    feat['bb_pct_z'] = rolling_z(bb_pct, 60)

    ema8 = close.ewm(span=8, adjust=False).mean()
    ema21 = close.ewm(span=21, adjust=False).mean()
    macd = ema8 - ema21
    macd_sig = macd - macd.ewm(span=5, adjust=False).mean()
    feat['macd_signal'] = macd_sig
    feat['macd_z'] = rolling_z(macd_sig, 60)

    # --- Volume / price pressure ---
    vol_ratio_5 = volume / (volume.rolling(5).mean() + 1e-9)
    vol_ratio_10 = volume / (volume.rolling(10).mean() + 1e-9)
    feat['volume_ratio_5'] = vol_ratio_5
    feat['volume_ratio_10'] = vol_ratio_10
    feat['volume_ratio_z'] = rolling_z(vol_ratio_10, 60)
    vwap_10 = (close * volume).rolling(10).sum() / (volume.rolling(10).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap_10) / (close + 1e-9)

    # --- Range / candle structure ---
    tr = pd.concat([high - low, (high - close.shift(1)).abs(), (low - close.shift(1)).abs()], axis=1).max(axis=1)
    feat['atr_10'] = tr.rolling(10).mean()
    feat['atr_norm'] = feat['atr_10'] / close
    feat['hl_range'] = (high - low) / close
    feat['hl_range_z'] = rolling_z(feat['hl_range'], 60)
    feat['candle_body_ratio'] = (close - open_).abs() / ((high - low) + 1e-9)
    feat['upper_wick_ratio'] = (high - df[['open', 'close']].max(axis=1)) / ((high - low) + 1e-9)
    feat['lower_wick_ratio'] = (df[['open', 'close']].min(axis=1) - low) / ((high - low) + 1e-9)
    feat['close_position_20'] = (close - low.rolling(20).min()) / (high.rolling(20).max() - low.rolling(20).min() + 1e-9)

    # --- Regime interactions ---
    feat['momentum_vol_interaction'] = feat['ret_3'] * feat['vol_regime']
    feat['momentum_regime'] = feat['ret_3'] / (realized_vol_10 + 1e-9)
    feat['momentum_regime_chg'] = feat['momentum_regime'].diff(4)
    feat['volume_pressure'] = vol_ratio_10 * feat['bb_pct']
    feat['volume_pressure_z'] = rolling_z(feat['volume_pressure'], 60)

    # --- Time features ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()
        feat['funding_momentum'] = fr.diff(4)
        feat['funding_x_volregime'] = fr * feat['vol_regime']

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
