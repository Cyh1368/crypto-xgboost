<DESCRIPTION>
The core issue identified in the feedback is "slope collapse," where the model's predictive power evaporates during regime shifts (Test to Val). To combat this, I am implementing a "Regime-Agnostic" feature set. 

Key changes include:
1. **Dynamic Scaling**: All return-based and distance-based features are normalized by both rolling volatility (z-scoring) and Kaufman's Efficiency Ratio. This transforms absolute moves into "standardized effort" units, which are more consistent across market regimes.
2. **Acceleration Features**: Instead of just using momentum, I include the 'rate of change' for volatility, price position (Bollinger Band momentum), and market efficiency. This helps the model detect the *onset* of a regime shift rather than just reacting to it.
3. **Volatility-of-Volatility**: Added a feature to measure the stability of the current volatility regime, allowing the model to discount signals during "chaotic" periods.
4. **Regularization for Robustness**: Increased `min_child_weight` and `reg_lambda` to force the XGBoost model to find broader, more universal patterns that survive the transition from the training distribution to the validation distribution.
5. **Shortened Lookbacks**: Heavily focused on 5-20 bar windows for features to ensure the model reacts quickly to the most relevant recent information while using a longer (100-bar) window for z-score normalization to define the local "normal."

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 420,
    "max_depth": 3,               # Keep shallow for better generalization
    "min_child_weight": 25,       # High robustness requirement
    "reg_alpha": 0.8,
    "reg_lambda": 12.0,           # Strong regularization to prevent slope collapse
    "subsample": 0.65,
    "colsample_bytree": 0.65,
    "learning_rate": 0.04,
    "gamma": 0.15,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 45,
}

CLIP_PERCENTILE = 98
MIN_PRED_STD_RATIO = 0.20
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build regime-robust features emphasizing standardized market dynamics.
    """
    feat = pd.DataFrame(index=df.index)
    close = df["close"]
    high = df["high"]
    low = df["low"]
    volume = df["volume"]
    eps = 1e-9

    log_ret = np.log(close / close.shift(1))

    def rz(x, w):
        """Rolling Z-score"""
        return (x - x.rolling(w).mean()) / (x.rolling(w).std() + eps)

    # --- 1. Normalized Momentum & Adaptive Returns ---
    # Using z-scored returns helps normalize for vol-regime shifts
    for lag in [1, 3, 5, 10]:
        r = close.pct_change(lag)
        feat[f"ret_z_{lag}"] = rz(r, 100)
    
    # Intraday momentum relative to recent volatility
    vol_fast = log_ret.rolling(20).std()
    feat["mom_vol_adj"] = close.pct_change(3) / (vol_fast + eps)

    # --- 2. Market Character (Efficiency & Volatility) ---
    # Efficiency ratio measures how 'clean' the trend is
    for w in [10, 30]:
        direction = (close - close.shift(w)).abs()
        volatility = close.diff().abs().rolling(w).sum()
        feat[f"eff_{w}"] = direction / (volatility + eps)

    vol_10 = log_ret.rolling(10).std()
    vol_60 = log_ret.rolling(60).std()
    feat["vol_z_10"] = rz(vol_10, 100)
    feat["vol_regime"] = vol_10 / (vol_60 + eps)
    
    # Vol-of-vol: Measures regime instability
    feat["vov_10"] = vol_10.rolling(10).std() / (vol_10.rolling(10).mean() + eps)

    # --- 3. Normalized Price Structure ---
    # Bollinger percentage z-scored to reveal mean-reversion strength
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    bb_pct = (close - bb_mid) / (2 * bb_std + eps)
    feat["bb_pct_z"] = rz(bb_pct, 40)
    
    # Relative position in recent range
    low_20, high_20 = low.rolling(20).min(), high.rolling(20).max()
    feat["range_pos"] = (close - low_20) / (high_20 - low_20 + eps)

    # Distances from moving averages normalized by ATR
    tr = pd.concat([high-low, (high-close.shift(1)).abs(), (low-close.shift(1)).abs()], axis=1).max(axis=1)
    atr = tr.rolling(20).mean()
    feat["dist_sma_atr"] = (close - bb_mid) / (atr + eps)

    # --- 4. Acceleration & Market Shifts ---
    # Rate of change of momentum (2nd derivative)
    feat["mom_accel"] = feat["ret_z_1"].diff(3)
    # Rate of change of efficiency (market breakdown detection)
    feat["eff_change"] = feat["eff_10"].diff(5)

    # --- 5. Candle & Volume Dynamics ---
    feat["hl_z"] = rz((high - low) / (close + eps), 100)
    feat["body_ratio"] = (close - df["open"]).abs() / (high - low + eps)
    feat["wick_ratio"] = (high - df[["open", "close"]].max(axis=1)) / (high - low + eps)
    
    # Volume pressure
    vol_ma = volume.rolling(20).mean()
    feat["vol_surge"] = volume / (vol_ma + eps)
    feat["vol_z"] = rz(np.log(volume + 1), 100)

    # --- 6. Interactions ---
    # High momentum is only meaningful if efficiency is high
    feat["mom_x_eff"] = feat["ret_z_3"] * feat["eff_10"]
    # Mean reversion is stronger in high vol
    feat["rev_x_vol"] = bb_pct * feat["vol_regime"]

    # --- 7. Session Context ---
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * df.index.dayofweek / 7)

    # --- 8. External Data (if present) ---
    if "funding_rate" in df.columns:
        fr = df["funding_rate"]
        feat["funding_z"] = rz(fr, 100)
        feat["funding_mom"] = fr.diff(4)
        feat["funding_x_vol"] = fr * feat["vol_regime"]

    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau):
            try:
                b, a = np.array(row["bids"]), np.array(row["asks"])
                bv = np.sum(b[:,1] * np.exp(-tau * np.arange(len(b))))
                av = np.sum(a[:,1] * np.exp(-tau * np.arange(len(a))))
                return (bv - av) / (bv + av + eps)
            except: return 0.0
        feat["obi_1"] = df.apply(lambda r: obi(r, 1), axis=1)
        feat["obi_3"] = df.apply(lambda r: obi(r, 3), axis=1)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    """
    # Clip targets to remove extreme outliers that skew the slope
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Robust scaling
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items() if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
