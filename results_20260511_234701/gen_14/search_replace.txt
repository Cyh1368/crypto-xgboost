    high = df['high']
    low = df['low']
    log_ret = np.log(close / close.shift(1))

    # Helper: Z-score within rolling window (regime-invariant)
    def z_score_rolling(s, window=20):
        mean = s.rolling(window).mean()
        std = s.rolling(window).std()
        return (s - mean) / (std + 1e-9)

    # --- Short-lookback Returns (5-12 bars, Z-scored) ---
    for lag in [1, 2, 3, 6, 12]:
        ret = close.pct_change(lag)
        feat[f'ret_{lag}'] = ret
        feat[f'ret_z_{lag}'] = z_score_rolling(ret, 15)

    # --- Short-lookback Volatility (5-15 bars) ---
    for w in [5, 10, 15]:
        vol = log_ret.rolling(w).std()
        feat[f'vol_{w}'] = vol
        feat[f'vol_z_{w}'] = z_score_rolling(vol, 20)

    # --- Volatility Trend (rate of change, regime-change signal) ---
    vol_5 = log_ret.rolling(5).std()
    vol_15 = log_ret.rolling(15).std()
    feat['vol_trend_5_15'] = (vol_5 - vol_15) / (vol_15 + 1e-9)
    feat['vol_accel'] = vol_5.diff(3)

    for period in [6, 9]:
