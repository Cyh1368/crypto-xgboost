<DESCRIPTION>
The model suffers from severe slope collapse (0.0013 on val vs 0.018 on train) and prediction collapse (std ratio 0.06). This indicates the model is overfitting to noise and failing to generalize across market regimes.

I am proposing three key changes:
1. **Regime-Invariant Normalization**: Increasing the Z-score rolling window from 15-20 to 100-200. A 15-bar window (3.75 hours) is too short and removes the actual signal, while a 100-200 bar window (25-50 hours) captures the "state" of the market relative to recent history without losing the trend.
2. **Feature Interaction**: Adding a `momentum_vol_interaction` (Return * Volatility Regime). This allows the model to learn that momentum in a high-volatility regime has different predictive power than in a low-volatility regime.
3. **Hyperparameter Tuning**: Reducing `max_depth` to 4 and increasing `min_child_weight` to 5 to force the model to learn more robust, higher-sample patterns. Lowering `reg_lambda` and `reg_alpha` to combat the "prediction collapse" and encourage the model to output a wider range of values (improving slope and std ratio).

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build regime-invariant features using short lookbacks and Z-score normalization.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    log_ret = np.log(close / close.shift(1))

    # Helper: Z-score within rolling window (regime-invariant)
    def z_score_rolling(s, window=20):
        mean = s.rolling(window).mean()
        std = s.rolling(window).std()
        return (s - mean) / (std + 1e-9)

    # --- Short-lookback Returns (5-12 bars, Z-scored) ---
    for lag in [1, 2, 3, 6, 12]:
        ret = close.pct_change(lag)
        feat[f'ret_{lag}'] = ret
        feat[f'ret_z_{lag}'] = z_score_rolling(ret, 15)

    # --- Short-lookback Volatility (5-15 bars) ---
    for w in [5, 10, 15]:
        vol = log_ret.rolling(w).std()
        feat[f'vol_{w}'] = vol
        feat[f'vol_z_{w}'] = z_score_rolling(vol, 20)

    # --- Volatility Trend (rate of change, regime-change signal) ---
    vol_5 = log_ret.rolling(5).std()
    vol_15 = log_ret.rolling(15).std()
    feat['vol_trend_5_15'] = (vol_5 - vol_15) / (vol_15 + 1e-9)
    feat['vol_accel'] = vol_5.diff(3)

    # --- RSI (short periods for faster adaptation) ---
    for period in [6, 9]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands (short window) ---
    bb_mid = close.rolling(10).mean()
    bb_std = close.rolling(10).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)
    feat['bb_pct_z'] = z_score_rolling(feat['bb_pct'], 15)

    # --- ATR (short window) ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build regime-invariant features using Z-score normalization and interactions.
    """
    feat = pd.DataFrame(index=df.index)
    close, volume, high, low = df['close'], df['volume'], df['high'], df['low']
    log_ret = np.log(close / close.shift(1))

    def z_score(s, window=100):
        return (s - s.rolling(window).mean()) / (s.rolling(window).std() + 1e-9)

    # --- Returns & Momentum (Z-scored over longer window for stability) ---
    for lag in [1, 4, 12]:
        ret = close.pct_change(lag)
        feat[f'ret_z_{lag}'] = z_score(ret, 200)

    # --- Volatility Regime ---
    vol_20 = log_ret.rolling(20).std()
    vol_100 = log_ret.rolling(100).std()
    feat['vol_z_20'] = z_score(vol_20, 100)
    feat['vol_regime'] = vol_20 / (vol_100 + 1e-9)

    # --- Feature Interactions ---
    feat['mom_vol_interaction'] = close.pct_change(4) * feat['vol_regime']
    feat['mom_vol_adj'] = close.pct_change(4) / (vol_20 + 1e-9)

    # --- Technical Indicators ---
    # RSI
    delta = close.diff()
    gain = delta.clip(lower=0).rolling(14).mean()
    loss = (-delta.clip(upper=0)).rolling(14).mean()
    feat['rsi_14'] = 100 - 100 / (1 + gain / (loss + 1e-9))

    # Bollinger Bands
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    # ATR
    tr = pd.concat([high - low, (high - close.shift(1)).abs(), (low - close.shift(1)).abs()], axis=1).max(axis=1)
    feat['atr_norm'] = tr.rolling(14).mean() / (close + 1e-9)

    # --- Volume & Efficiency ---
    feat['rel_vol_z'] = z_score(volume / (volume.rolling(20).mean() + 1e-9), 100)
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    feat['close_pos_20'] = (close - low.rolling(20).min()) / (high.rolling(20).max() - low.rolling(20).min() + 1e-9)

    # --- Time & Structure ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)
    feat['candle_body_ratio'] = (close - df['open']).abs() / ((high - low) + 1e-9)

    # --- Conditional Data ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row):
            try:
                b, a = np.array(row['bids']), np.array(row['asks'])
                bv, av = np.sum(b[:, 1]), np.sum(a[:, 1])
                return (bv - av) / (bv + av + 1e-9)
            except: return 0.0
        feat['obi_simple'] = df.apply(obi, axis=1)

    if 'funding_rate' in df.columns:
        feat['funding_rate'] = df['funding_rate']
        feat['funding_z'] = z_score(df['funding_rate'], 100)

    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()
    return feat
