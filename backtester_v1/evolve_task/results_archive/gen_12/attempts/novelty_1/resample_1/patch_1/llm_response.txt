    Translates XGBoost 15-min return prediction into a trading decision.

    """

    close = float(bar_context.get('close', 0))
    atr = float(bar_context.get('atr', 0))
    adx = float(bar_context.get('adx', 0))
    rsi = float(bar_context.get('rsi', 50))

    signal = 0
    if predicted_return > long_thresh:
        if adx > 20 and rsi < 75:

        if adx > 20 and rsi > 25:
            signal = -1

    # Target 10% base size, scaled inversely by volatility (ATR as % of price)

        position_size = max(0.05, min(position_size, 0.20))
    else:

    # Take profit at 3.0x ATR and Stop Loss at 1.5x ATR
    if close > 0 and atr > 0:
