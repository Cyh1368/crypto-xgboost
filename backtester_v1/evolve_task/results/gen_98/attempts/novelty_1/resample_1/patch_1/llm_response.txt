<DESCRIPTION>
This algorithm takes a fundamentally different approach by implementing a **hierarchical regime-detection system** rather than static thresholds. Instead of applying fixed entry filters uniformly, it:

1. **Detects market microstructure regimes** using a composite score of OBI, pressure, spread, and Kyle's Lambda to classify the market as "toxic," "neutral," or "favorable"
2. **Applies regime-specific entry logic** where favorable regimes allow more aggressive entries while toxic regimes enforce stricter filters
3. **Uses dynamic position sizing based on regime confidence** rather than fixed sizes, scaling up in high-confidence favorable regimes
4. **Implements asymmetric risk management** where TP/SL ratios adapt to regime volatility and autocorrelation patterns
5. **Incorporates volatility-regime clustering** using ATR and recent returns to avoid over-leverage in unstable conditions
6. **Applies session-aware micro-filters** that relax in high-alpha sessions (US) but tighten in choppy sessions (Asia)

This differs fundamentally from the current linear threshold approach by recognizing that market conditions are non-stationary. The same predicted return signal should be treated differently depending on whether the microstructure is supporting or fighting the trade. This should improve Sharpe ratio by reducing drawdowns in toxic regimes while maintaining win rate in favorable ones.

def generate_signal(predicted_return: float, bar_context: dict) -> dict:
    """
    Regime-adaptive hierarchical filter: Detects microstructure regimes and applies
    asymmetric entry/sizing/risk logic based on regime favorability.
    """
    
    # ============================================================================
    # 1. CONTEXT EXTRACTION
    # ============================================================================
    is_us = bar_context.get('is_us_session', False)
    is_asia = bar_context.get('is_asia_session', False)
    is_weekend = bar_context.get('is_weekend', False)
    
    # Microstructure signals
    obi_tau1 = bar_context.get('obi_tau1', 0.0)
    obi_tau5 = bar_context.get('obi_tau5', 0.0)
    obi_tau10 = bar_context.get('obi_tau10', 0.0)
    pressure = bar_context.get('book_pressure_3', 0.0)
    spread_bps = bar_context.get('spread_bps', 1.0)
    depth_ratio_5 = bar_context.get('depth_ratio_5', 1.0)
    kyle_lambda = bar_context.get('kyle_lambda_est', 0.0)
    
    # Price action
    close = bar_context.get('close', 1.0)
    atr = bar_context.get('atr_14', 0.002)
    vol_5 = bar_context.get('vol_5', 0.01)
    vol_20 = bar_context.get('vol_20', 0.01)
    ret_1 = bar_context.get('ret_1', 0.0)
    ret_3 = bar_context.get('ret_3', 0.0)
    ret_6 = bar_context.get('ret_6', 0.0)
    autocorr = bar_context.get('autocorr_5', 0.0)
    trend = bar_context.get('trend_strength', 0.0)
    vwap_dev = bar_context.get('vwap_dev', 0.0)
    skew = bar_context.get('skew', 0.0)
    
    # Macro
    funding = bar_context.get('funding_rate', 0.0)
    funding_ma8h = bar_context.get('funding_rate_ma8h', 0.0)
    
    # ============================================================================
    # 2. REGIME DETECTION (Composite Microstructure Score)
    # ============================================================================
    
    # OBI consensus: average across timeframes
    obi_avg = (obi_tau1 + obi_tau5 + obi_tau10) / 3.0
    
    # Normalize pressure to [-1, 1] range (book_pressure_3 is typically -100 to 100)
    pressure_norm = np.clip(pressure / 100.0, -1.0, 1.0)
    
    # Spread toxicity: higher spread = more toxic
    spread_toxicity = min(1.0, spread_bps / 15.0)
    
    # Liquidity toxicity: higher Kyle's Lambda = more toxic
    liquidity_toxicity = min(1.0, kyle_lambda * 50.0)
    
    # Depth toxicity: lower depth ratio = more toxic
    depth_toxicity = max(0.0, 1.0 - depth_ratio_5) if depth_ratio_5 > 0 else 0.5
    
    # Composite toxicity score [0, 1]
    toxicity = (spread_toxicity + liquidity_toxicity + depth_toxicity) / 3.0
    
    # Directional alignment score: how well OBI and pressure agree
    obi_pressure_alignment = (obi_avg + pressure_norm) / 2.0
    
    # Volatility regime: elevated vol increases toxicity
    vol_ratio = vol_5 / (vol_20 + 1e-8)
    vol_spike = max(0.0, vol_ratio - 1.2)  # Penalize vol spikes
    
    # Regime classification
    regime_score = obi_pressure_alignment * (1.0 - toxicity) * (1.0 - vol_spike)
    
    # Classify regime
    if regime_score > 0.3:
        regime = "favorable"
    elif regime_score < -0.3:
        regime = "adverse"
    else:
        regime = "neutral"
    
    regime_confidence = abs(regime_score)
    
    # ============================================================================
    # 3. DYNAMIC ENTRY THRESHOLDS (Regime-Aware)
    # ============================================================================
    
    # Base threshold adjusted by session
    if is_us:
        base_thresh = 0.00168
    elif is_asia:
        base_thresh = 0.00195
    else:
        base_thresh = 0.00180
    
    # Regime-based threshold adjustment
    if regime == "favorable":
        # Relax threshold in favorable regimes
        threshold_adj = -0.00008
    elif regime == "adverse":
        # Tighten threshold in adverse regimes
        threshold_adj = 0.00012
    else:
        threshold_adj = 0.0
    
    effective_thresh = base_thresh + threshold_adj
    
    # ============================================================================
    # 4. SIGNAL GENERATION (Hierarchical Filters)
    # ============================================================================
    
    signal = 0
    
    # Long signal
    if predicted_return > effective_thresh:
        # Primary filter: OBI alignment (relaxed in favorable regime)
        obi_threshold = -0.65 if regime == "favorable" else -0.55
        obi_pass = obi_tau5 > obi_threshold
        
        # Secondary filter: VWAP deviation (stricter in adverse)
        vwap_threshold = 0.027 if regime == "adverse" else 0.030
        vwap_pass = vwap_dev < vwap_threshold
        
        # Tertiary filter: Spread (stricter in adverse)
        spread_threshold = 9.5 if regime == "adverse" else 11.0
        spread_pass = spread_bps < spread_threshold
        
        # Quaternary filter: Funding alignment
        funding_pass = funding < 0.0016
        
        # Quinary filter: Volatility safety (stricter in spikes)
        atr_pct = (atr / close) if close > 0 else 0.002
        vol_safety = atr_pct < 0.015
        
        # Senary filter: Trend alignment (bonus in favorable)
        if regime == "favorable" and trend > 0.2:
            trend_bonus = True
        else:
            trend_bonus = False
        
        # Aggregate: all primary filters must pass
        if obi_pass and vwap_pass and spread_pass and funding_pass and vol_safety:
            signal = 1
    
    # Short signal
    elif predicted_return < -effective_thresh:
        # Primary filter: OBI alignment (relaxed in favorable regime)
        obi_threshold = 0.65 if regime == "favorable" else 0.55
        obi_pass = obi_tau5 < obi_threshold
        
        # Secondary filter: VWAP deviation (stricter in adverse)
        vwap_threshold = -0.027 if regime == "adverse" else -0.030
        vwap_pass = vwap_dev > vwap_threshold
        
        # Tertiary filter: Spread (stricter in adverse)
        spread_threshold = 9.5 if regime == "adverse" else 11.0
        spread_pass = spread_bps < spread_threshold
        
        # Quaternary filter: Funding alignment
        funding_pass = funding > -0.0016
        
        # Quinary filter: Volatility safety (stricter in spikes)
        atr_pct = (atr / close) if close > 0 else 0.002
        vol_safety = atr_pct < 0.015
        
        # Senary filter: Trend alignment (bonus in favorable)
        if regime == "favorable" and trend < -0.2:
            trend_bonus = True
        else:
            trend_bonus = False
        
        # Aggregate: all primary filters must pass
        if obi_pass and vwap_pass and spread_pass and funding_pass and vol_safety:
            signal = -1
    
    # ============================================================================
    # 5. ADAPTIVE POSITION SIZING (Regime & Confidence Weighted)
    # ============================================================================
    
    if signal != 0:
        # Base size
        base_size = 0.14
        
        # Confidence multiplier: how strong is the predicted return
        confidence = abs(predicted_return) / effective_thresh
        confidence_mult = min(1.35, max(1.0, confidence))
        
        # Regime confidence multiplier: scale up in high-confidence favorable regimes
        if regime == "favorable":
            regime_mult = 1.0 + (regime_confidence * 0.25)
        elif regime == "adverse":
            regime_mult = max(0.75, 1.0 - (regime_confidence * 0.30))
        else:
            regime_mult = 1.0
        
        # Volatility dampening: reduce size in high volatility
        atr_pct = (atr / close) if close > 0 else 0.002
        if atr_pct > 0.010:
            vol_mult = max(0.80, 1.0 - (atr_pct - 0.010) * 50.0)
        else:
            vol_mult = 1.0
        
        # Autocorrelation dampening: reduce size if market is choppy
        if abs(autocorr) < 0.05:
            autocorr_mult = 0.90
        else:
            autocorr_mult = 1.0
        
        # Combine multipliers
        position_size = base_size * confidence_mult * regime_mult * vol_mult * autocorr_mult
        
        # Bounds
        position_size = max(0.09, min(0.22, position_size))
    else:
        position_size = 0.0
    
    # ============================================================================
    # 6. ASYMMETRIC RISK MANAGEMENT (TP/SL Regime-Aware)
    # ============================================================================
    
    if signal != 0:
        # Base targets
        take_profit = 0.0042
        stop_loss = 0.0032
        
        # Regime adjustment
        if regime == "favorable":
            # Extend targets in favorable regimes
            take_profit += 0.0004
            stop_loss -= 0.0002
        elif regime == "adverse":
            # Tighten targets in adverse regimes
            take_profit -= 0.0002
            stop_loss += 0.0001
        
        # Funding alignment bonus
        if (signal == 1 and funding < 0) or (signal == -1 and funding > 0):
            take_profit += 0.0003
        
        # Trend strength bonus
        if abs(trend) > 0.40:
            take_profit += 0.0003
            stop_loss -= 0.0001
        
        # Volatility adjustment to risk
        atr_pct = (atr / close) if close > 0 else 0.002
        if atr_pct > 0.012:
            stop_loss += 0.0002
        
        # Skew adjustment: if negative skew, tighten stop
        if skew < -0.3:
            stop_loss += 0.0001
        
        # Ensure SL < TP
        if stop_loss >= take_profit:
            stop_loss = take_profit * 0.75
    else:
        take_profit = 0.0
        stop_loss = 0.0
    
    # ============================================================================
    # 7. DYNAMIC HOLD TIME (Regime & Session Based)
    # ============================================================================
    
    if signal != 0:
        # Base hold time
        max_bars = 4
        
        # Session extension
        if is_us and abs(trend) > 0.35:
            max_bars = 6
        elif is_asia and abs(autocorr) > 0.20:
            max_bars = 3  # Exit faster in choppy Asia
        elif abs(trend) > 0.45:
            max_bars = 5
        
        # Regime adjustment
        if regime == "adverse":
            max_bars = max(3, max_bars - 1)  # Exit faster in adverse regimes
        elif regime == "favorable" and regime_confidence > 0.5:
            max_bars = min(6, max_bars + 1)  # Hold longer in high-confidence favorable
    else:
        max_bars = 0
    
    # ============================================================================
    # 8. RETURN SIGNAL DICT
    # ============================================================================
    
    return {
        "signal": int(signal),
        "position_size": float(round(position_size, 4)),
        "take_profit": float(round(take_profit, 5)),
        "stop_loss": float(round(stop_loss, 5)),
        "max_bars": int(max_bars),
    }
