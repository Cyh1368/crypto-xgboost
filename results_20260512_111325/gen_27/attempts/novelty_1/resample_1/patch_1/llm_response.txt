<DESCRIPTION>
This crossover keeps the strongest ideas from both prior scripts while targeting the main failure mode: regime-specific features that collapse on validation. The feature set is shifted toward short lookbacks, rolling z-scores, and rate-of-change regime descriptors so the model relies more on fast-adapting patterns. I also keep a few stable interaction terms, but reduce long-horizon, regime-fragile signals. On the model side, the XGBoost configuration is made slightly more expressive than the current baseline while still using enough regularization and sampling to avoid overfitting. The goal is to recover validation slope and prediction dispersion without regressing train/test DA too much.

# evolve/initial.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 320,
    "max_depth": 4,
    "min_child_weight": 3,
    "reg_alpha": 0.15,
    "reg_lambda": 1.5,
    "gamma": 0.05,
    "subsample": 0.82,
    "colsample_bytree": 0.78,
    "learning_rate": 0.06,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 45,
}

CLIP_PERCENTILE = 98
MIN_PRED_STD_RATIO = 0.18
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)

    log_ret = np.log(close / close.shift(1))
    ret1 = close.pct_change(1)
    ret3 = close.pct_change(3)
    ret5 = close.pct_change(5)

    # --- Short-horizon returns ---
    for lag in [1, 2, 3, 5, 8, 13, 20]:
        feat[f"ret_{lag}"] = close.pct_change(lag)

    # regime-invariant return normalization
    for w in [5, 10, 20]:
        m = ret1.rolling(w).mean()
        s = ret1.rolling(w).std()
        feat[f"ret_z_{w}"] = (ret1 - m) / (s + 1e-9)

    # --- Volatility / volatility change ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    vol_40 = log_ret.rolling(40).std()

    feat["vol_5"] = vol_5
    feat["vol_10"] = vol_10
    feat["vol_20"] = vol_20
    feat["vol_trend_5_10"] = (vol_5 - vol_10) / (vol_10 + 1e-9)
    feat["vol_trend_10_20"] = (vol_10 - vol_20) / (vol_20 + 1e-9)
    feat["vol_trend_20_40"] = (vol_20 - vol_40) / (vol_40 + 1e-9)

    # rolling z of realized vol to make it more regime invariant
    feat["vol_z_20"] = (vol_20 - vol_20.rolling(40).mean()) / (vol_20.rolling(40).std() + 1e-9)

    # --- RSI / momentum oscillators (short) ---
    delta = close.diff()
    gain = delta.clip(lower=0)
    loss = (-delta.clip(upper=0))
    for period in [5, 9, 14]:
        rs = gain.rolling(period).mean() / (loss.rolling(period).mean() + 1e-9)
        feat[f"rsi_{period}"] = 100 - 100 / (1 + rs)

    feat["rsi_z_20"] = (feat["rsi_9"] - feat["rsi_9"].rolling(20).mean()) / (feat["rsi_9"].rolling(20).std() + 1e-9)

    # --- Price location / mean reversion ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    sma_40 = close.rolling(40).mean()

    feat["price_sma5_dev"] = (close - sma_5) / (sma_5 + 1e-9)
    feat["price_sma10_dev"] = (close - sma_10) / (sma_10 + 1e-9)
    feat["price_sma20_dev"] = (close - sma_20) / (sma_20 + 1e-9)
    feat["sma_cross_10_20"] = (sma_10 - sma_20) / (sma_20 + 1e-9)
    feat["sma_cross_20_40"] = (sma_20 - sma_40) / (sma_40 + 1e-9)

    # z-score of deviation from local mean
    dev_20 = feat["price_sma20_dev"]
    feat["price_sma20_dev_z"] = (dev_20 - dev_20.rolling(40).mean()) / (dev_20.rolling(40).std() + 1e-9)

    # --- Trend / efficiency / autocorr change ---
    feat["trend_strength_5"] = (close / close.shift(5) - 1).abs()
    feat["trend_strength_10"] = (close / close.shift(10) - 1).abs()
    feat["trend_strength_20"] = (close / close.shift(20) - 1).abs()

    feat["trend_dir_5"] = np.sign(close / close.shift(5) - 1)
    feat["trend_dir_10"] = np.sign(close / close.shift(10) - 1)
    feat["trend_dir_20"] = np.sign(close / close.shift(20) - 1)

    eff_10 = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    eff_20 = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    feat["efficiency_10"] = eff_10
    feat["efficiency_20"] = eff_20
    feat["efficiency_trend"] = eff_10 - eff_20

    feat["ret_autocorr_5"] = ret1.rolling(5).corr(ret1.shift(1))
    feat["ret_autocorr_10"] = ret1.rolling(10).corr(ret1.shift(1))
    feat["ret_autocorr_20"] = ret1.rolling(20).corr(ret1.shift(1))
    feat["autocorr_trend"] = feat["ret_autocorr_5"] - feat["ret_autocorr_20"]

    # --- Volatility regime and change ---
    realized_vol_20 = vol_20
    realized_vol_40 = vol_40
    feat["vol_regime"] = realized_vol_20 / (realized_vol_20.rolling(80).mean() + 1e-9)
    feat["vol_regime_fast"] = realized_vol_20 / (realized_vol_40 + 1e-9)
    feat["vol_regime_change"] = feat["vol_regime"] - feat["vol_regime"].shift(5)
    feat["vol_rank_80"] = realized_vol_20.rolling(80).rank(pct=True)

    # --- Volume / liquidity ---
    vol_ma_5 = volume.rolling(5).mean()
    vol_ma_10 = volume.rolling(10).mean()
    vol_ma_20 = volume.rolling(20).mean()
    vol_std_20 = volume.rolling(20).std()

    feat["volume_ratio_5"] = volume / (vol_ma_5 + 1e-9)
    feat["volume_ratio_10"] = volume / (vol_ma_10 + 1e-9)
    feat["volume_ratio_20"] = volume / (vol_ma_20 + 1e-9)
    feat["volume_z_20"] = (volume - vol_ma_20) / (vol_std_20 + 1e-9)

    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat["vwap_dev"] = (close - vwap_20) / (close + 1e-9)

    # --- Range / candle structure ---
    tr = pd.concat(
        [
            high - low,
            (high - close.shift(1)).abs(),
            (low - close.shift(1)).abs(),
        ],
        axis=1,
    ).max(axis=1)

    feat["atr_10"] = tr.rolling(10).mean()
    feat["atr_14"] = tr.rolling(14).mean()
    feat["atr_norm"] = feat["atr_10"] / (close + 1e-9)
    feat["hl_range"] = (high - low) / (close + 1e-9)
    feat["hl_range_ma"] = feat["hl_range"].rolling(10).mean()

    denom = (high - low) + 1e-9
    feat["body_ratio"] = (close - open_).abs() / denom
    feat["upper_wick_ratio"] = (high - pd.concat([open_, close], axis=1).max(axis=1)) / denom
    feat["lower_wick_ratio"] = (pd.concat([open_, close], axis=1).min(axis=1) - low) / denom

    # --- Close position in range ---
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat["close_position_20"] = (close - low_20) / (high_20 - low_20 + 1e-9)

    # --- Momentum x regime interactions ---
    feat["momentum_vol_interaction"] = ret3 * vol_20
    feat["momentum_regime"] = ret3 / (vol_20 + 1e-9)
    feat["momentum_regime_20"] = ret3 * feat["vol_regime"]
    feat["momentum_regime_fast"] = ret3 * feat["vol_regime_fast"]
    feat["mom_vol_short"] = ret5 * feat["vol_regime_change"]

    # stronger cross-feature terms that may generalize better
    feat["efficiency_vol_interaction"] = feat["efficiency_trend"] * feat["vol_regime"]
    feat["trend_vol_interaction"] = feat["trend_strength_10"] * feat["vol_regime"]
    feat["momentum_x_voltrend"] = ret3 * feat["vol_regime_change"]

    # --- Volume x price pressure ---
    feat["volume_bb_interaction"] = feat["volume_ratio_20"] * feat["price_sma20_dev"]
    feat["volume_price_pressure"] = feat["volume_ratio_10"] * feat["price_sma10_dev"]

    # --- Time / session ---
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat["is_weekend"] = (df.index.dayofweek >= 5).astype(float)

    # --- Orderbook ---
    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row["bids"])
                asks = np.array(row["asks"])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: obi(r, 1), axis=1)
        feat["obi_tau3"] = df.apply(lambda r: obi(r, 3), axis=1)

    # --- Funding ---
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding_rate"] = fr
        feat["funding_8h_ma"] = fr.rolling(32).mean()
        feat["funding_momentum"] = fr.diff(4)
        feat["funding_x_volregime"] = fr * feat["vol_regime"]

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()

    # cap feature count under limit with a stable selection if needed
    if feat.shape[1] > 80:
        keep_cols = [
            "ret_1", "ret_2", "ret_3", "ret_5", "ret_8", "ret_13", "ret_20",
            "ret_z_5", "ret_z_10", "ret_z_20",
            "vol_5", "vol_10", "vol_20", "vol_trend_5_10", "vol_trend_10_20", "vol_trend_20_40",
            "vol_z_20",
            "rsi_5", "rsi_9", "rsi_14", "rsi_z_20",
            "price_sma5_dev", "price_sma10_dev", "price_sma20_dev", "sma_cross_10_20", "sma_cross_20_40", "price_sma20_dev_z",
            "trend_strength_5", "trend_strength_10", "trend_strength_20",
            "trend_dir_5", "trend_dir_10", "trend_dir_20",
            "efficiency_10", "efficiency_20", "efficiency_trend",
            "ret_autocorr_5", "ret_autocorr_10", "ret_autocorr_20", "autocorr_trend",
            "vol_regime", "vol_regime_fast", "vol_regime_change", "vol_rank_80",
            "volume_ratio_5", "volume_ratio_10", "volume_ratio_20", "volume_z_20", "vwap_dev",
            "atr_10", "atr_14", "atr_norm", "hl_range", "hl_range_ma",
            "body_ratio", "upper_wick_ratio", "lower_wick_ratio", "close_position_20",
            "momentum_vol_interaction", "momentum_regime", "momentum_regime_20",
            "momentum_regime_fast", "mom_vol_short", "efficiency_vol_interaction",
            "trend_vol_interaction", "momentum_x_voltrend", "volume_bb_interaction",
            "volume_price_pressure", "hour_sin", "hour_cos", "dow_sin", "is_weekend",
            "obi_tau1", "obi_tau3", "funding_rate", "funding_8h_ma", "funding_momentum", "funding_x_volregime"
        ]
        keep_cols = [c for c in keep_cols if c in feat.columns]
        feat = feat[keep_cols[:80]]

    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items() if k != "early_stopping_rounds"}
    early = XGB_PARAMS.get("early_stopping_rounds", 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr,
        y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        "train": {"pred": model.predict(X_tr), "actual": y_train.values},
        "test": {"pred": model.predict(X_te), "actual": y_test.values},
        "val": {"pred": model.predict(X_va), "actual": y_val.values},
        "model": model,
    }
