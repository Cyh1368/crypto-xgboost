import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os

# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 520,
    "max_depth": 4,
    "min_child_weight": 4,
    "reg_alpha": 0.08,
    "reg_lambda": 1.2,
    "subsample": 0.88,
    "colsample_bytree": 0.78,
    "learning_rate": 0.045,
    "gamma": 0.02,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 50,
}

CLIP_PERCENTILE = 98
MIN_PRED_STD_RATIO = 0.18
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Regime-invariant short-horizon feature set.
    Uses only past information.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)
    eps = 1e-9

    ret1 = close.pct_change(1)
    log_ret = np.log(close / close.shift(1))

    # Short returns
    for lag in [1, 2, 3, 5, 8, 12, 16, 20]:
        feat[f"ret_{lag}"] = close.pct_change(lag)

    # Rolling z-scores of returns
    for w in [5, 10, 20]:
        mu = ret1.rolling(w).mean()
        sd = ret1.rolling(w).std()
        feat[f"ret_z_{w}"] = (ret1 - mu) / (sd + eps)

    # Volatility + volatility regime changes
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    vol_40 = log_ret.rolling(40).std()

    feat["vol_5"] = vol_5
    feat["vol_10"] = vol_10
    feat["vol_20"] = vol_20
    feat["vol_40"] = vol_40
    feat["vol_trend_5_10"] = (vol_5 - vol_10) / (vol_10 + eps)
    feat["vol_trend_10_20"] = (vol_10 - vol_20) / (vol_20 + eps)
    feat["vol_trend_20_40"] = (vol_20 - vol_40) / (vol_40 + eps)
    feat["vol_rank_40"] = vol_20.rolling(40).rank(pct=True)
    feat["vol_rank_60"] = vol_20.rolling(60).rank(pct=True)

    # Trend efficiency and change in trend character
    diff_abs = close.diff().abs()
    eff_5 = (close - close.shift(5)).abs() / (diff_abs.rolling(5).sum() + eps)
    eff_10 = (close - close.shift(10)).abs() / (diff_abs.rolling(10).sum() + eps)
    eff_20 = (close - close.shift(20)).abs() / (diff_abs.rolling(20).sum() + eps)

    feat["efficiency_5"] = eff_5
    feat["efficiency_10"] = eff_10
    feat["efficiency_20"] = eff_20
    feat["eff_trend_5_10"] = eff_5 - eff_10
    feat["eff_trend_10_20"] = eff_10 - eff_20
    feat["eff_rank_40"] = eff_10.rolling(40).rank(pct=True)

    # Autocorrelation regime changes
    feat["autocorr_5"] = ret1.rolling(5).corr(ret1.shift(1))
    feat["autocorr_8"] = ret1.rolling(8).corr(ret1.shift(1))
    feat["autocorr_10"] = ret1.rolling(10).corr(ret1.shift(1))
    feat["autocorr_trend_5_10"] = feat["autocorr_5"] - feat["autocorr_10"]
    feat["autocorr_trend_10_20"] = feat["autocorr_10"] - ret1.rolling(20).corr(ret1.shift(1))
    feat["autocorr_rank_40"] = feat["autocorr_10"].rolling(40).rank(pct=True)

    # RSI-like oscillators
    delta = close.diff()
    gain = delta.clip(lower=0)
    loss = (-delta.clip(upper=0))

    for period in [5, 8, 10, 14]:
        rs = gain.rolling(period).mean() / (loss.rolling(period).mean() + eps)
        feat[f"rsi_{period}"] = 100 - 100 / (1 + rs)
        feat[f"rsi_{period}_rank_40"] = feat[f"rsi_{period}"].rolling(40).rank(pct=True)

    # Price location / moving average structure
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    sma_40 = close.rolling(40).mean()

    feat["price_sma5_dev"] = (close - sma_5) / (sma_5 + eps)
    feat["price_sma10_dev"] = (close - sma_10) / (sma_10 + eps)
    feat["price_sma20_dev"] = (close - sma_20) / (sma_20 + eps)

    feat["price_sma5_z"] = feat["price_sma5_dev"] / (feat["price_sma5_dev"].rolling(20).std() + eps)
    feat["price_sma10_z"] = feat["price_sma10_dev"] / (feat["price_sma10_dev"].rolling(20).std() + eps)

    feat["sma_5_10_cross"] = (sma_5 - sma_10) / (sma_10 + eps)
    feat["sma_10_20_cross"] = (sma_10 - sma_20) / (sma_20 + eps)
    feat["sma_10_40_cross"] = (sma_10 - sma_40) / (sma_40 + eps)

    bb_std_20 = close.rolling(20).std()
    feat["bb_pct_20"] = (close - sma_20) / (2.0 * bb_std_20 + eps)
    feat["bb_pct_20_rank_40"] = feat["bb_pct_20"].rolling(40).rank(pct=True)

    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat["close_pos_10"] = (close - low_10) / (high_10 - low_10 + eps)
    feat["close_pos_20"] = (close - low_20) / (high_20 - low_20 + eps)

    # ATR / candle structure
    tr = pd.concat(
        [
            high - low,
            (high - close.shift(1)).abs(),
            (low - close.shift(1)).abs(),
        ],
        axis=1,
    ).max(axis=1)

    atr_5 = tr.rolling(5).mean()
    atr_10 = tr.rolling(10).mean()
    atr_20 = tr.rolling(20).mean()

    feat["atr_5"] = atr_5
    feat["atr_10"] = atr_10
    feat["atr_20"] = atr_20
    feat["atr_norm_10"] = atr_10 / (close + eps)
    feat["atr_norm_20"] = atr_20 / (close + eps)

    feat["hl_range"] = (high - low) / (close + eps)
    feat["hl_range_ma_10"] = feat["hl_range"].rolling(10).mean()
    feat["hl_range_rank_40"] = feat["hl_range"].rolling(40).rank(pct=True)

    feat["candle_body_ratio"] = (close - open_).abs() / ((high - low) + eps)
    feat["upper_wick_ratio"] = (high - pd.concat([open_, close], axis=1).max(axis=1)) / ((high - low) + eps)
    feat["lower_wick_ratio"] = (pd.concat([open_, close], axis=1).min(axis=1) - low) / ((high - low) + eps)

    # Responsive MACD-style features
    ema5 = close.ewm(span=5, adjust=False).mean()
    ema8 = close.ewm(span=8, adjust=False).mean()
    ema13 = close.ewm(span=13, adjust=False).mean()
    macd_fast = ema5 - ema8
    macd_mid = ema8 - ema13
    feat["macd_fast_signal"] = macd_fast - macd_fast.ewm(span=4, adjust=False).mean()
    feat["macd_mid_signal"] = macd_mid - macd_mid.ewm(span=4, adjust=False).mean()

    # Volume features
    vol_ma_5 = volume.rolling(5).mean()
    vol_ma_10 = volume.rolling(10).mean()
    vol_ma_20 = volume.rolling(20).mean()

    feat["volume_ratio_5"] = volume / (vol_ma_5 + eps)
    feat["volume_ratio_10"] = volume / (vol_ma_10 + eps)
    feat["volume_ratio_20"] = volume / (vol_ma_20 + eps)
    feat["volume_z_20"] = (volume - vol_ma_20) / (volume.rolling(20).std() + eps)
    feat["volume_rank_40"] = volume.rolling(40).rank(pct=True)

    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + eps)
    feat["vwap_dev_20"] = (close - vwap_20) / (close + eps)

    # Explicit regime interaction terms
    ret_3 = close.pct_change(3)
    feat["momentum_vol_interaction"] = ret_3 * vol_10
    feat["momentum_regime"] = ret_3 / (vol_10 + eps)
    feat["momentum_regime_fast"] = ret_3 * (vol_5 / (vol_10 + eps))
    feat["momentum_regime_slow"] = ret_3 * (vol_10 / (vol_20 + eps))
    feat["momentum_z_20"] = (ret_3 - ret_3.rolling(20).mean()) / (ret_3.rolling(20).std() + eps)
    feat["momentum_z_10"] = (close.pct_change(5) - close.pct_change(5).rolling(10).mean()) / (close.pct_change(5).rolling(10).std() + eps)

    feat["trend_vol_interaction"] = feat["sma_5_10_cross"] * feat["vol_trend_10_20"]
    feat["volume_price_pressure"] = feat["volume_ratio_20"] * feat["price_sma10_dev"]
    feat["volume_bb_interaction"] = feat["volume_ratio_20"] * feat["bb_pct_20"]
    feat["eff_vol_interaction"] = eff_10 * feat["vol_rank_40"]

    # Time features
    hour = df.index.hour
    dow = df.index.dayofweek
    feat["hour_sin"] = np.sin(2 * np.pi * hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * dow / 7)
    feat["dow_cos"] = np.cos(2 * np.pi * dow / 7)
    feat["is_weekend"] = (dow >= 5).astype(float)

    # Compact regime summary
    trend_10 = close / close.shift(10) - 1.0
    trend_20 = close / close.shift(20) - 1.0
    trend_40 = close / close.shift(40) - 1.0
    feat["trend_strength_10"] = trend_10.abs() * 10000
    feat["trend_strength_20"] = trend_20.abs() * 10000
    feat["trend_strength_40"] = trend_40.abs() * 10000
    feat["trend_dir_10"] = np.sign(trend_10)
    feat["trend_dir_20"] = np.sign(trend_20)
    feat["vol_regime_10_20"] = vol_10 / (vol_20 + eps)
    feat["vol_regime_20_40"] = vol_20 / (vol_40 + eps)
    feat["vol_regime_rank"] = vol_20.rolling(80).rank(pct=True)

    # Optional orderbook
    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row["bids"])
                asks = np.array(row["asks"])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + eps)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: obi(r, 1), axis=1)
        feat["obi_tau3"] = df.apply(lambda r: obi(r, 3), axis=1)

    # Optional funding
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding_rate"] = fr
        feat["funding_8h_ma"] = fr.rolling(32).mean()
        feat["funding_momentum"] = fr.diff(4)
        feat["funding_z_32"] = (fr - fr.rolling(32).mean()) / (fr.rolling(32).std() + eps)
        feat["funding_x_volregime"] = fr * feat["vol_regime_10_20"]

    # Final cleanup
    feat = feat.replace([np.inf, -np.inf], np.nan).dropna()

    # Feature cap
    if feat.shape[1] > 80:
        feat = feat.iloc[:, :80]

    return feat
# EVOLVE-BLOCK-END

# DO NOT MODIFY BELOW THIS LINE
# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != "early_stopping_rounds"}
    early = XGB_PARAMS.get("early_stopping_rounds", 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr,
        y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        "train": {"pred": model.predict(X_tr), "actual": y_train.values},
        "test": {"pred": model.predict(X_te), "actual": y_test.values},
        "val": {"pred": model.predict(X_va), "actual": y_val.values},
        "model": model,
    }
