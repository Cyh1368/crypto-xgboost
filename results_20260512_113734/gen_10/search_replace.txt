<DESCRIPTION>
This edit aims to reduce regime-specific overfitting by shifting the feature set away from long-horizon absolute levels and toward short-horizon, rolling standardized signals. The main idea is:

- Replace the broad set of long lookbacks with shorter, more responsive windows.
- Add rolling z-score normalization for returns, volatility, and volume so the model sees regime-invariant deviations instead of raw magnitudes.
- Add “market character change” features such as volatility trend, efficiency trend, and autocorrelation trend.
- Keep the important momentum × volatility interaction, but make it more robust via short-window inputs.
- Reduce model capacity slightly and add regularization to force the booster to focus on stable patterns and improve validation slope.

This should help the model preserve slope on the validation regime while also increasing prediction dispersion away from the current near-mean collapse.

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns ---
    for lag in [1, 2, 3, 6, 12, 24, 48]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Volatility ---
    log_ret = np.log(close / close.shift(1))
    for w in [5, 10, 20, 60]:
        feat[f'vol_{w}'] = log_ret.rolling(w).std()

    # --- RSI ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands ---
    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean()
    feat['atr_norm'] = feat['atr_14'] / close

    # --- MACD ---
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    macd = ema12 - ema26
    feat['macd_signal'] = macd - macd.ewm(span=9, adjust=False).mean()

    # --- Volume ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + 1e-9)
    vwap = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap) / (close + 1e-9)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)

    feat['trend_strength_60'] = (close / close.shift(60) - 1).abs() * 10000

                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1)

    if 'funding_rate' in df.columns:

        feat['funding_x_volregime'] = fr * feat['vol_regime']

    # --- Trend Consistency and Efficiency ---
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    open_ = df['open']

    log_ret = np.log(close / close.shift(1))
    ret_1 = close.pct_change(1)
    ret_3 = close.pct_change(3)

    def zscore(s, w):
        m = s.rolling(w).mean()
        sd = s.rolling(w).std()
        return (s - m) / (sd + 1e-9)

    # --- Short-horizon returns / momentum ---
    for lag in [1, 2, 3, 5, 8, 13, 20]:
        feat[f'ret_{lag}'] = close.pct_change(lag)
    feat['ret_z_10'] = zscore(ret_1, 10)
    feat['ret_z_20'] = zscore(ret_1, 20)
    feat['mom_5'] = close.pct_change(5)
    feat['mom_10'] = close.pct_change(10)
    feat['mom_accel_3'] = ret_1.diff(3)

    # --- Rolling regime-normalized volatility ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    feat['vol_5'] = vol_5
    feat['vol_10'] = vol_10
    feat['vol_20'] = vol_20
    feat['vol_z_20'] = zscore(vol_20, 20)
    feat['vol_z_60'] = zscore(vol_20, 60)
    feat['vol_trend_20'] = vol_20.pct_change(5)
    feat['vol_trend_60'] = vol_20.pct_change(10)
    feat['vol_regime'] = vol_20 / (vol_20.rolling(60).mean() + 1e-9)

    # --- RSI / efficiency / autocorrelation trends ---
    for period in [5, 10, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + 1e-9)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)
    feat['efficiency_10'] = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    feat['eff_trend'] = feat['efficiency_10'] - feat['efficiency_20']
    feat['autocorr_10'] = ret_1.rolling(10).corr(ret_1.shift(1))
    feat['autocorr_20'] = ret_1.rolling(20).corr(ret_1.shift(1))
    feat['autocorr_trend'] = feat['autocorr_10'] - feat['autocorr_20']

    # --- Compact trend / range structure ---
    sma_5 = close.rolling(5).mean()
    sma_20 = close.rolling(20).mean()
    feat['price_sma5_dev'] = (close - sma_5) / (sma_5 + 1e-9)
    feat['price_sma20_dev'] = (close - sma_20) / (sma_20 + 1e-9)
    feat['sma_cross'] = (sma_5 - sma_20) / (sma_20 + 1e-9)

    bb_mid = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    feat['bb_pct'] = (close - bb_mid) / (2 * bb_std + 1e-9)

    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_10'] = tr.rolling(10).mean()
    feat['atr_norm'] = feat['atr_10'] / close
    feat['hl_range'] = (high - low) / close
    feat['candle_body_ratio'] = (close - open_).abs() / ((high - low) + 1e-9)

    # --- Volume regime normalization ---
    vol_mean_10 = volume.rolling(10).mean()
    vol_mean_20 = volume.rolling(20).mean()
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + 1e-9)
    feat['volume_ratio_20'] = volume / (vol_mean_20 + 1e-9)
    feat['volume_z_20'] = zscore(volume, 20)
    feat['volume_z_60'] = zscore(volume, 60)
    feat['vwap_dev'] = (close - (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)) / (close + 1e-9)

    # --- Momentum × volatility interaction ---
    feat['mom_x_volregime'] = ret_3 * feat['vol_regime']
    feat['mom_div_vol'] = ret_3 / (vol_20 + 1e-9)

    # --- Close position / wick structure ---
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat['close_position_20'] = (close - low_20) / (high_20 - low_20 + 1e-9)
    feat['upper_wick_ratio'] = (high - df[['open', 'close']].max(axis=1)) / ((high - low) + 1e-9)
    feat['lower_wick_ratio'] = (df[['open', 'close']].min(axis=1) - low) / ((high - low) + 1e-9)

    # --- Time features ---
    feat['hour_sin
