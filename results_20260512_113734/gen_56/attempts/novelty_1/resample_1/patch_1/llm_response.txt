def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)
    eps = 1e-9

    ret1 = close.pct_change(1)
    log_ret = np.log(close / close.shift(1))

    # -----------------------
    # Short-horizon returns
    # -----------------------
    for lag in [1, 2, 3, 5, 8, 12, 20]:
        feat[f"ret_{lag}"] = close.pct_change(lag)

    # Rolling z-scores of returns (regime invariant)
    for w in [5, 10, 20]:
        mu = ret1.rolling(w).mean()
        sd = ret1.rolling(w).std()
        feat[f"ret_z_{w}"] = (ret1 - mu) / (sd + eps)

    # -----------------------
    # Volatility + volatility changes
    # -----------------------
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    vol_40 = log_ret.rolling(40).std()

    feat["vol_5"] = vol_5
    feat["vol_10"] = vol_10
    feat["vol_20"] = vol_20
    feat["vol_40"] = vol_40
    feat["vol_trend_5_10"] = (vol_5 - vol_10) / (vol_10 + eps)
    feat["vol_trend_10_20"] = (vol_10 - vol_20) / (vol_20 + eps)
    feat["vol_trend_20_40"] = (vol_20 - vol_40) / (vol_40 + eps)
    feat["vol_rank_40"] = vol_20.rolling(40).rank(pct=True)

    # -----------------------
    # Efficiency / autocorrelation regime changes
    # -----------------------
    diff_abs = close.diff().abs()
    feat["efficiency_5"] = (close - close.shift(5)).abs() / (diff_abs.rolling(5).sum() + eps)
    feat["efficiency_10"] = (close - close.shift(10)).abs() / (diff_abs.rolling(10).sum() + eps)
    feat["efficiency_20"] = (close - close.shift(20)).abs() / (diff_abs.rolling(20).sum() + eps)
    feat["eff_trend_5_10"] = feat["efficiency_5"] - feat["efficiency_10"]
    feat["eff_trend_10_20"] = feat["efficiency_10"] - feat["efficiency_20"]

    feat["autocorr_5"] = ret1.rolling(5).corr(ret1.shift(1))
    feat["autocorr_10"] = ret1.rolling(10).corr(ret1.shift(1))
    feat["autocorr_trend_5_10"] = feat["autocorr_5"] - feat["autocorr_10"]

    # -----------------------
    # RSI-like oscillators
    # -----------------------
    delta = close.diff()
    gain = delta.clip(lower=0)
    loss = (-delta.clip(upper=0))
    for period in [5, 10, 14]:
        rs = gain.rolling(period).mean() / (loss.rolling(period).mean() + eps)
        feat[f"rsi_{period}"] = 100 - 100 / (1 + rs)

    # -----------------------
    # Price location / moving averages
    # -----------------------
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    sma_40 = close.rolling(40).mean()

    feat["price_sma5_dev"] = (close - sma_5) / (sma_5 + eps)
    feat["price_sma10_dev"] = (close - sma_10) / (sma_10 + eps)
    feat["price_sma20_dev"] = (close - sma_20) / (sma_20 + eps)
    feat["price_sma5_z"] = feat["price_sma5_dev"] / (feat["price_sma5_dev"].rolling(20).std() + eps)
    feat["price_sma10_z"] = feat["price_sma10_dev"] / (feat["price_sma10_dev"].rolling(20).std() + eps)
    feat["sma_5_20_cross"] = (sma_5 - sma_20) / (sma_20 + eps)
    feat["sma_10_40_cross"] = (sma_10 - sma_40) / (sma_40 + eps)

    # Bollinger / range position
    bb_std_20 = close.rolling(20).std()
    feat["bb_pct_20"] = (close - sma_20) / (2.0 * bb_std_20 + eps)

    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat["close_pos_10"] = (close - low_10) / (high_10 - low_10 + eps)
    feat["close_pos_20"] = (close - low_20) / (high_20 - low_20 + eps)

    # -----------------------
    # ATR / candle structure
    # -----------------------
    tr = pd.concat(
        [
            high - low,
            (high - close.shift(1)).abs(),
            (low - close.shift(1)).abs(),
        ],
        axis=1,
    ).max(axis=1)

    atr_5 = tr.rolling(5).mean()
    atr_10 = tr.rolling(10).mean()
    atr_20 = tr.rolling(20).mean()

    feat["atr_5"] = atr_5
    feat["atr_10"] = atr_10
    feat["atr_20"] = atr_20
    feat["atr_norm_10"] = atr_10 / (close + eps)
    feat["atr_norm_20"] = atr_20 / (close + eps)

    feat["hl_range"] = (high - low) / (close + eps)
    feat["hl_range_ma_10"] = feat["hl_range"].rolling(10).mean()
    feat["candle_body_ratio"] = (close - open_).abs() / ((high - low) + eps)
    feat["upper_wick_ratio"] = (high - pd.concat([open_, close], axis=1).max(axis=1)) / ((high - low) + eps)
    feat["lower_wick_ratio"] = (pd.concat([open_, close], axis=1).min(axis=1) - low) / ((high - low) + eps)

    # -----------------------
    # MACD-style momentum
    # -----------------------
    ema8 = close.ewm(span=8, adjust=False).mean()
    ema13 = close.ewm(span=13, adjust=False).mean()
    ema21 = close.ewm(span=21, adjust=False).mean()
    macd_fast = ema8 - ema13
    macd_slow = ema13 - ema21
    feat["macd_fast_signal"] = macd_fast - macd_fast.ewm(span=5, adjust=False).mean()
    feat["macd_slow_signal"] = macd_slow - macd_slow.ewm(span=5, adjust=False).mean()

    # -----------------------
    # Volume features
    # -----------------------
    vol_ma_5 = volume.rolling(5).mean()
    vol_ma_20 = volume.rolling(20).mean()
    vol_ma_40 = volume.rolling(40).mean()
    feat["volume_ratio_5"] = volume / (vol_ma_5 + eps)
    feat["volume_ratio_20"] = volume / (vol_ma_20 + eps)
    feat["volume_ratio_40"] = volume / (vol_ma_40 + eps)

    vol_mu_20 = volume.rolling(20).mean()
    vol_sd_20 = volume.rolling(20).std()
    feat["volume_z_20"] = (volume - vol_mu_20) / (vol_sd_20 + eps)

    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + eps)
    feat["vwap_dev_20"] = (close - vwap_20) / (close + eps)

    # -----------------------
    # Market character change via interactions
    # -----------------------
    ret_3 = close.pct_change(3)
    ret_5 = close.pct_change(5)
    feat["momentum_vol_interaction"] = ret_3 * vol_20
    feat["momentum_regime"] = ret_3 / (vol_20 + eps)
    feat["momentum_regime_fast"] = ret_3 * (vol_10 / (vol_20 + eps))
    feat["momentum_regime_slow"] = ret_3 * (vol_20 / (vol_40 + eps))
    feat["momentum_z_20"] = (ret_3 - ret_3.rolling(20).mean()) / (ret_3.rolling(20).std() + eps)
    feat["momentum_z_10"] = (ret_5 - ret_5.rolling(10).mean()) / (ret_5.rolling(10).std() + eps)
    feat["trend_vol_interaction"] = feat["sma_5_20_cross"] * feat["vol_trend_10_20"]

    # Volume × price location interactions
    feat["volume_price_pressure"] = feat["volume_ratio_20"] * feat["price_sma10_dev"]
    feat["volume_bb_interaction"] = feat["volume_ratio_20"] * feat["bb_pct_20"]

    # -----------------------
    # Session / time features
    # -----------------------
    hour = df.index.hour
    dow = df.index.dayofweek
    feat["hour_sin"] = np.sin(2 * np.pi * hour / 24)
    feat["hour_cos"] = np.cos(2 * np.pi * hour / 24)
    feat["dow_sin"] = np.sin(2 * np.pi * dow / 7)
    feat["dow_cos"] = np.cos(2 * np.pi * dow / 7)
    feat["is_weekend"] = (dow >= 5).astype(float)

    # -----------------------
    # Regime features
    # -----------------------
    trend_10 = (close / close.shift(10) - 1.0)
    trend_20 = (close / close.shift(20) - 1.0)
    trend_40 = (close / close.shift(40) - 1.0)
    feat["trend_strength_10"] = trend_10.abs() * 10000
    feat["trend_strength_20"] = trend_20.abs() * 10000
    feat["trend_strength_40"] = trend_40.abs() * 10000
    feat["trend_dir_10"] = np.sign(trend_10)
    feat["trend_dir_20"] = np.sign(trend_20)
    feat["vol_regime_20_40"] = vol_20 / (vol_40 + eps)
    feat["vol_regime_10_20"] = vol_10 / (vol_20 + eps)
    feat["vol_regime_rank"] = vol_20.rolling(120).rank(pct=True)

    # -----------------------
    # Optional orderbook features
    # -----------------------
    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row["bids"])
                asks = np.array(row["asks"])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + eps)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: obi(r, 1), axis=1)
        feat["obi_tau3"] = df.apply(lambda r: obi(r, 3), axis=1)

    # -----------------------
    # Optional funding features
    # -----------------------
    if "funding_rate" in df.columns
