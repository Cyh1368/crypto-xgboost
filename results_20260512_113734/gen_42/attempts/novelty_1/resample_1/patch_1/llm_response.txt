<DESCRIPTION>
This crossover keeps the strongest ideas from the prior scripts while targeting the main failure mode: train-to-validation slope collapse under a second regime shift. I combined the most transferable short-horizon and regime-change features from the better-performing variants, while removing longer, more regime-specific lookbacks that likely overfit the earlier market state.

Key changes:
- Shorter, more regime-invariant windows dominate the feature set.
- Added rolling z-score normalization for returns, volatility, volume, and price position.
- Added change-of-character features: volatility trend, autocorrelation trend, efficiency trend, dispersion trend.
- Added multiplicative interactions that often generalize better than standalone components.
- Kept feature count safely under 80 and used only past data.
- Tuned XGBoost to be a bit more expressive than the heavily regularized runs, while still controlling overfit.

The goal is to improve validation slope and prediction dispersion without regressing DA too much.

# evolve/initial.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 320,
    "max_depth": 4,
    "min_child_weight": 2,
    "reg_alpha": 0.03,
    "reg_lambda": 0.9,
    "gamma": 0.0,
    "subsample": 0.88,
    "colsample_bytree": 0.88,
    "learning_rate": 0.045,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 60,
}

CLIP_PERCENTILE = 97
MIN_PRED_STD_RATIO = 0.18
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)
    eps = 1e-9

    log_close = np.log(close.replace(0, np.nan))
    log_ret = log_close.diff()
    ret1 = close.pct_change(1)
    ret2 = close.pct_change(2)
    ret3 = close.pct_change(3)

    def zscore(s, w):
        m = s.rolling(w).mean()
        sd = s.rolling(w).std()
        return (s - m) / (sd + eps)

    def rank_pct(s, w):
        return s.rolling(w).rank(pct=True)

    # --- Short-horizon returns / normalized momentum ---
    for lag in [1, 2, 3, 5, 8, 13, 20]:
        feat[f"ret_{lag}"] = close.pct_change(lag)

    feat["ret1_z_10"] = zscore(ret1, 10)
    feat["ret1_z_20"] = zscore(ret1, 20)
    feat["ret3_z_20"] = zscore(ret3, 20)
    feat["ret3_z_40"] = zscore(ret3, 40)
    feat["mom_3_8"] = close.pct_change(3) - close.pct_change(8)
    feat["mom_5_13"] = close.pct_change(5) - close.pct_change(13)
    feat["mom_8_20"] = close.pct_change(8) - close.pct_change(20)
    feat["mom_accel_3"] = ret1.diff(3)
    feat["mom_accel_6"] = ret1.diff(6)

    # --- Volatility regime and change in volatility ---
    for w in [5, 10, 20]:
        feat[f"vol_{w}"] = log_ret.rolling(w).std()

    feat["vol_5_z_40"] = zscore(feat["vol_5"], 40)
    feat["vol_10_z_60"] = zscore(feat["vol_10"], 60)
    feat["vol_20_z_80"] = zscore(feat["vol_20"], 80)
    feat["vol_trend_5"] = feat["vol_10"].diff(3)
    feat["vol_trend_10"] = feat["vol_20"].diff(5)
    feat["vol_trend_20"] = feat["vol_20"].diff(10)
    feat["vol_rank_80"] = rank_pct(feat["vol_20"], 80)

    realized_vol_20 = feat["vol_20"]
    realized_vol_40 = log_ret.rolling(40).std()
    feat["vol_regime"] = realized_vol_20 / (realized_vol_20.rolling(80).mean() + eps)
    feat["vol_regime_fast"] = realized_vol_20 / (realized_vol_40 + eps)

    # --- Directional efficiency / trend quality ---
    net_5 = (close - close.shift(5)).abs()
    path_5 = close.diff().abs().rolling(5).sum()
    net_10 = (close - close.shift(10)).abs()
    path_10 = close.diff().abs().rolling(10).sum()
    net_20 = (close - close.shift(20)).abs()
    path_20 = close.diff().abs().rolling(20).sum()

    feat["eff_5"] = net_5 / (path_5 + eps)
    feat["eff_10"] = net_10 / (path_10 + eps)
    feat["eff_20"] = net_20 / (path_20 + eps)
    feat["eff_trend"] = feat["eff_5"] - feat["eff_20"]
    feat["trend_dir_5"] = np.sign(close.pct_change(5))
    feat["trend_dir_10"] = np.sign(close.pct_change(10))
    feat["trend_dir_20"] = np.sign(close.pct_change(20))

    # --- Rolling autocorrelation / market memory ---
    feat["autocorr_10"] = ret1.rolling(10).corr(ret1.shift(1))
    feat["autocorr_20"] = ret1.rolling(20).corr(ret1.shift(1))
    feat["autocorr_40"] = ret1.rolling(40).corr(ret1.shift(1))
    feat["autocorr_trend"] = feat["autocorr_10"] - feat["autocorr_40"]

    # --- Short standardized price position ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    sma_40 = close.rolling(40).mean()
    std_10 = close.rolling(10).std()
    std_20 = close.rolling(20).std()
    std_40 = close.rolling(40).std()

    feat["price_z_10"] = (close - sma_10) / (std_10 + eps)
    feat["price_z_20"] = (close - sma_20) / (std_20 + eps)
    feat["price_z_40"] = (close - sma_40) / (std_40 + eps)
    feat["sma5_sma10"] = (sma_5 - sma_10) / (sma_10 + eps)
    feat["sma10_sma20"] = (sma_10 - sma_20) / (sma_20 + eps)
    feat["sma20_sma40"] = (sma_20 - sma_40) / (sma_40 + eps)

    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat["close_pos_10"] = (close - low_10) / (high_10 - low_10 + eps)
    feat["close_pos_20"] = (close - low_20) / (high_20 - low_20 + eps)
    feat["mr_pressure_10"] = 1.0 - feat["close_pos_10"]
    feat["mr_pressure_20"] = 1.0 - feat["close_pos_20"]

    # --- Range / candle structure ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)

    feat["atr_5"] = tr.rolling(5).mean()
    feat["atr_10"] = tr.rolling(10).mean()
    feat["atr_20"] = tr.rolling(20).mean()
    feat["atr_norm_10"] = feat["atr_10"] / (close + eps)
    feat["atr_norm_20"] = feat["atr_20"] / (close + eps)
    feat["range_z_20"] = zscore((high - low) / (close + eps), 20)

    candle_range = (high - low).replace(0, np.nan)
    feat["body_ratio"] = (close - open_).abs() / (candle_range + eps)
    feat["upper_wick"] = (high - np.maximum(open_, close)) / (candle_range + eps)
    feat["lower_wick"] = (np.minimum(open_, close) - low) / (candle_range + eps)

    # --- Volume regime and volume-price interactions ---
    vol_mean_5 = volume.rolling(5).mean()
    vol_mean_10 = volume.rolling(10).mean()
    vol_mean_20 = volume.rolling(20).mean()
    vol_std_10 = volume.rolling(10).std()
    vol_std_20 = volume.rolling(20).std()

    feat["vol_ratio_5"] = volume / (vol_mean_5 + eps)
    feat["vol_ratio_10"] = volume / (vol_mean_10 + eps)
    feat["vol_ratio_20"] = volume / (vol_mean_20 + eps)
    feat["vol_z_10"] = (volume - vol_mean_10) / (vol_std_10 + eps)
    feat["vol_z_20"] = (volume - vol_mean_20) / (vol_std_20 + eps)
    feat["vol_z_60"] = zscore(volume, 60)
    feat["vol_trend"] = feat["vol_ratio_5"] - feat["vol_ratio_20"]

    feat["mom_x_volreg"] = ret3 * feat["vol_z_60"]
    feat["mom_x_atrreg"] = ret3 * feat["atr_norm_20"]
    feat["eff_x_volreg"] = feat["eff_10"] * feat["vol_rank_80"]
    feat["range_x_volume"] = feat["range_z_20"] * feat["vol_z_20"]
    feat["vol_x_pricez"] = feat["vol_z_20"] * feat["price_z_20"]
    feat["mom_x_regime"] = ret3 * feat["vol_regime"]
    feat["mom_x_eff"] = ret3 * feat["eff_10"]

    # --- Breakout pressure / VWAP-style anchors ---
    feat["breakout_10"] = ((close > high_10.shift(1)).astype(float) - (close < low_10.shift(1)).astype(float))
    feat["breakout_20"] = ((close > high_20.shift(1)).astype(float) - (close < low_20.shift(1)).astype(float))

    vwap_10 = (close * volume).rolling(10).sum() / (volume.rolling(10).sum() + eps)
    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + eps)
    feat["vwap_dev_10"] = (close - vwap_10) / (close + eps)
    feat["vwap_dev_20"] = (close - vwap_20) / (close + eps)
    feat["vwap_dev_x_vol"] = feat["vwap_dev_20"] * feat["vol_z_20"]

    # --- RSI family ---
    delta = close.diff()
    for period in [5, 9, 14]:
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + eps)
        rsi = 100 - 100 / (1 + rs)
        feat[f"rsi_{period}"] = rsi
        feat[f"rsi_{period}_z_40"] = zscore(rsi, 40)

    # --- Short-term return shape / dispersion ---
    feat["ret_z_20"] = zscore(ret1, 20)
    feat["ret_z_40"] = zscore(ret1, 40)
    feat["return_dispersion_20"] = ret1.rolling(20).std() / (ret1.rolling(20).mean().abs() + eps)
    feat["return_dispersion_40"] = ret1.rolling(40).std() / (ret1.rolling(40).mean().abs() + eps)
    feat["dispersion_trend"] = feat["return_dispersion_20"] - feat["return_dispersion_40"]

    # --- Session effects ---
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24.0)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24.0)
    feat["dow_sin"] = np.sin(2 * np.pi * df.index.dayofweek / 7.0)
    feat["dow_cos"] = np.cos(2 * np.pi * df.index.dayofweek / 7.0)
    feat["is_weekend"] = (df.index.dayofweek >= 5).astype(float)

    # --- Funding / orderbook if present ---
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding_rate"] = fr
        feat["funding_z_32"] = zscore(fr, 32)
        feat["funding_z_96"] = zscore(fr, 96)
        feat["funding_mom_4"] = fr.diff(4)
        feat["funding_mom_12"] = fr.diff(12)
        feat["funding_x_mom"] = feat["funding_z_96"] * ret3
        feat["funding_x_volregime"] = fr * feat["vol_regime"]

    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row["bids"], dtype=float)
                asks = np.array(row["asks"], dtype=float)
                if bids.ndim != 2 or asks.ndim != 2 or bids.shape[1] < 2 or asks.shape[1] < 2:
                    return 0.0
                b = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                a = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (b - a) / (b + a + eps)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: obi(r, 1), axis=1)
        feat["obi_tau3"] = df.apply(lambda r: obi(r, 3), axis=1)
        feat["obi_diff"] = feat["obi_tau1"] - feat["obi_tau3"]

    # --- Stability / regime shift indicators ---
    feat["volatility_ratio_10_20"] = feat["vol_10"] / (feat["vol_20"] + eps)
    feat["volatility_ratio_5_20"] = feat["vol_5"] / (feat["vol_20"] + eps)
    feat["range_to_vol_20"] = feat["range_z_20"] / (feat["vol_z_20"].abs() + 1.0)
    feat["momentum_regime"] = ret3 / (feat["vol_20"] + eps)
    feat["momentum_regime_x_eff"] = feat["momentum_regime"] * feat["eff_10"]
    feat["efficiency_20"] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + eps)
    feat["efficiency_5"] = (close - close.shift(5
