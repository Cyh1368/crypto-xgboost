def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)
    eps = 1e-9

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)

    log_ret = np.log(close / close.shift(1))
    ret_1 = close.pct_change(1)
    ret_3 = close.pct_change(3)

    # --- Short-horizon returns (focus on 1-20 bars) ---
    for lag in [1, 2, 3, 5, 8, 13]:
        feat[f"ret_{lag}"] = close.pct_change(lag)

    # Z-scored returns within rolling windows (regime-invariant)
    for w in [8, 13, 20]:
        mu = ret_1.rolling(w).mean()
        sd = ret_1.rolling(w).std()
        feat[f"ret_z_{w}"] = (ret_1 - mu) / (sd + eps)

    # Rank-based features for robustness
    feat["ret_rank_20"] = ret_1.rolling(20).rank(pct=True)
    feat["ret3_rank_30"] = ret_3.rolling(30).rank(pct=True)

    # --- Volatility with emphasis on CHANGE (rate of change) ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()

    feat["vol_5"] = vol_5
    feat["vol_10"] = vol_10
    feat["vol_20"] = vol_20
    feat["vol_ratio_5_20"] = vol_5 / (vol_20 + eps)
    feat["vol_ratio_10_20"] = vol_10 / (vol_20 + eps)

    # Volatility trend (rate of change - key for regime detection)
    feat["vol_trend_5"] = vol_5.pct_change(5)
    feat["vol_trend_10"] = vol_10.pct_change(8)
    feat["vol_chg_5_20"] = (vol_5 - vol_20) / (vol_20 + eps)

    # Z-scored volatility
    feat["vol_z_20"] = (vol_20 - vol_20.rolling(20).mean()) / (vol_20.rolling(20).std() + eps)
    feat["vol_rank_30"] = vol_20.rolling(30).rank(pct=True)

    # --- RSI (short periods only) ---
    delta = close.diff()
    gain = delta.clip(lower=0)
    loss = (-delta.clip(upper=0))

    for period in [5, 8, 14]:
        avg_gain = gain.rolling(period).mean()
        avg_loss = loss.rolling(period).mean()
        rs = avg_gain / (avg_loss + eps)
        feat[f"rsi_{period}"] = 100 - 100 / (1 + rs)

    # --- Bollinger / price location (short windows) ---
    bb_mid_10 = close.rolling(10).mean()
    bb_std_10 = close.rolling(10).std()
    bb_mid_15 = close.rolling(15).mean()
    bb_std_15 = close.rolling(15).std()
    
    feat["bb_pct_10"] = (close - bb_mid_10) / (2 * bb_std_10 + eps)
    feat["bb_pct_15"] = (close - bb_mid_15) / (2 * bb_std_15 + eps)

    # --- ATR / candle structure ---
    tr = pd.concat(
        [
            high - low,
            (high - close.shift(1)).abs(),
            (low - close.shift(1)).abs(),
        ],
        axis=1,
    ).max(axis=1)

    atr_5 = tr.rolling(5).mean()
    atr_14 = tr.rolling(14).mean()
    feat["atr_5"] = atr_5
    feat["atr_14"] = atr_14
    feat["atr_norm_5"] = atr_5 / (close + eps)
    feat["atr_norm_14"] = atr_14 / (close + eps)

    hl_range = (high - low) / (close + eps)
    feat["hl_range"] = hl_range
    feat["hl_range_z_20"] = (hl_range - hl_range.rolling(20).mean()) / (hl_range.rolling(20).std() + eps)

    body = (close - open_).abs()
    candle_range = (high - low).abs() + eps
    feat["candle_body_ratio"] = body / candle_range
    feat["upper_wick_ratio"] = (high - np.maximum(open_, close)) / candle_range
    feat["lower_wick_ratio"] = (np.minimum(open_, close) - low) / candle_range

    # --- Trend / mean reversion (short windows) ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()

    feat["price_sma5_dev"] = (close - sma_5) / (sma_5 + eps)
    feat["price_sma10_dev"] = (close - sma_10) / (sma_10 + eps)
    feat["price_sma20_dev"] = (close - sma_20) / (sma_20 + eps)
    feat["sma_cross_5_20"] = (sma_5 - sma_20) / (sma_20 + eps)

    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat["close_pos_10"] = (close - low_10) / (high_10 - low_10 + eps)
    feat["close_pos_20"] = (close - low_20) / (high_20 - low_20 + eps)

    # --- Efficiency / market character CHANGE (critical for regime shifts) ---
    feat["efficiency_10"] = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + eps)
    feat["efficiency_20"] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + eps)
    
    # Rate of change of efficiency (detects regime transitions)
    feat["efficiency_chg_10"] = feat["efficiency_10"].pct_change(8)
    feat["efficiency_rank_30"] = feat["efficiency_20"].rolling(30).rank(pct=True)

    # --- MACD / momentum dynamics (short spans) ---
    ema_5 = close.ewm(span=5, adjust=False).mean()
    ema_13 = close.ewm(span=13, adjust=False).mean()
    ema_21 = close.ewm(span=21, adjust=False).mean()
    macd_5_13 = ema_5 - ema_13
    macd_13_21 = ema_13 - ema_21
    feat["macd_5_13"] = macd_5_13 - macd_5_13.ewm(span=5, adjust=False).mean()
    feat["macd_13_21"] = macd_13_21 - macd_13_21.ewm(span=5, adjust=False).mean()

    # --- Volume / participation (z-scored for regime invariance) ---
    vol_ma_5 = volume.rolling(5).mean()
    vol_ma_10 = volume.rolling(10).mean()
    vol_ma_20 = volume.rolling(20).mean()

    feat["volume_ratio_5"] = volume / (vol_ma_5 + eps)
    feat["volume_ratio_10"] = volume / (vol_ma_10 + eps)
    feat["volume_ratio_20"] = volume / (vol_ma_20 + eps)
    
    # Z-scored volume
    feat["volume_z_20"] = (volume - vol_ma_20) / (volume.rolling(20).std() + eps)

    vwap_10 = (close * volume).rolling(10).sum() / (volume.rolling(10).sum() + eps)
    vwap_15 = (close * volume).rolling(15).sum() / (volume.rolling(15).sum() + eps)
    feat["vwap_dev_10"] = (close - vwap_10) / (close + eps)
    feat["vwap_dev_15"] = (close - vwap_15) / (close + eps)

    # --- Regime state with CHANGE features ---
    vol_regime = vol_10 / (vol_10.rolling(60).mean() + eps)
    vol_regime_fast = vol_5 / (vol_20 + eps)
    feat["vol_regime"] = vol_regime
    feat["vol_regime_fast"] = vol_regime_fast
    
    # Rate of change of regime (detects regime transitions)
    feat["vol_regime_chg_10"] = vol_regime.pct_change(10)
    feat["vol_regime_rank_30"] = vol_regime.rolling(30).rank(pct=True)

    trend_10 = close.pct_change(10)
    trend_20 = close.pct_change(20)
    feat["trend_strength_10"] = trend_10.abs() * 10000
    feat["trend_strength_20"] = trend_20.abs() * 10000
    feat["trend_dir_10"] = np.sign(trend_10)
    feat["trend_chg_10"] = trend_10 - trend_10.shift(10)

    feat["mom_x_volreg"] = ret_3 * vol_regime
    feat["mom_x_volfast"] = ret_3 * vol_regime_fast
    feat["momentum_vol_interaction"] = ret_3 * vol_10
    
    # Z-scored momentum
    feat["momentum_z_20"] = (ret_3 - ret_3.rolling(20).mean()) / (ret_3.rolling(20).std() + eps)
    
    # Volume-price interactions
    feat["volume_price_pressure"] = feat["volume_ratio_10"] * feat["price_sma10_dev"]
    feat["volume_bb_interaction"] = feat["volume_ratio_20"] * feat["bb_pct_15"]

    # --- Acceleration / correlation ---
    feat["ret_accel_3"] = ret_1.diff(3)
    feat["ret_accel_6"] = ret_1.diff(6)
    feat["ret_vol_corr_10"] = log_ret.rolling(10).corr(volume.pct_change(1))
    feat["ret_vol_corr_20"] = log_ret.rolling(20).corr(volume.pct_change(1))

    # --- Time features ---
    hour = df.index.hour
    dow = df.index.dayofweek
    feat["hour_sin"] = np.sin(2 * np.pi * hour / 24.0)
    feat["hour_cos"] = np.cos(2 * np.pi * hour / 24.0)
    feat["dow_sin"] = np.sin(2 * np.pi * dow / 7.0)
    feat["is_weekend"] = (dow >= 5).astype(float)

    # --- Orderbook imbalance (if available) ---
    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.asarray(row["bids"], dtype=float)

                    return 0.0
                bid_w = np.exp(-tau * np.arange(len(bids)))
                ask_w = np.exp(-tau * np.arange(len(asks)))
                bid_vol = np.sum(bids[:, 1] * bid_w)
                ask_vol = np.sum(asks[:, 1] * ask_w)
                return (bid_vol
