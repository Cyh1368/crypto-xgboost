regime_adaptive_short_lookback_crossover

<DESCRIPTION>
This crossover combines the best elements from both programs to address the regime shift problem:

1. **From the inspiration program (-3.00 score)**: Adopts the shorter lookback windows (5-20 bars instead of 60+) and z-score normalization approach that proved more regime-invariant. The inspiration program achieved val_DA=0.528 vs current's 0.521.

2. **From the current program's hyperparameters**: Keeps the more conservative regularization (max_depth=3, min_child_weight=6) which prevents overfitting, but increases learning_rate slightly to 0.10 (between current's 0.08 and inspiration's 0.12) for better expressivity.

3. **Key innovations**:
   - Removes long lookback features (48, 60 periods) that are regime-specific
   - Adds more z-scored features (returns, volume) for regime invariance
   - Focuses on RATE OF CHANGE features (vol_trend, efficiency changes) rather than absolute levels
   - Reduces feature count from ~60 to ~55 by removing redundant long-window features
   - Keeps robust interaction terms but simplifies them
   - Uses shorter windows for all regime indicators (10-20 vs 60-200)

4. **Addressing the val slope collapse**: The primary issue is that features work in train/test but fail in val. This suggests the val period has different market character. By using shorter windows and z-scoring, features become less dependent on the specific volatility/trend regime and more on relative market structure.

The strategy is to sacrifice some train performance (accept lower train_DA) in exchange for better generalization to the val period where the model currently fails.

# evolve/initial.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 240,
    "max_depth": 3,
    "min_child_weight": 6,
    "reg_alpha": 0.3,
    "reg_lambda": 1.5,
    "subsample": 0.82,
    "colsample_bytree": 0.78,
    "learning_rate": 0.10,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 40,
}

CLIP_PERCENTILE = 97
MIN_PRED_STD_RATIO = 0.18
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']

    # --- Returns (shorter lookbacks only) ---
    log_ret = np.log(close / close.shift(1))
    ret_1 = close.pct_change(1)
    
    # Raw returns at key short lags
    for lag in [1, 2, 3, 6, 12]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # Z-scored returns within rolling window (regime-invariant)
    for w in [10, 20]:
        ret_mean = ret_1.rolling(w).mean()
        ret_std = ret_1.rolling(w).std()
        feat[f'ret_z_{w}'] = (ret_1 - ret_mean) / (ret_std + 1e-9)

    # --- Volatility (shorter windows + trend) ---
    vol_5 = log_ret.rolling(5).std()
    vol_10 = log_ret.rolling(10).std()
    vol_20 = log_ret.rolling(20).std()
    
    feat['vol_5'] = vol_5
    feat['vol_10'] = vol_10
    feat['vol_20'] = vol_20

    # Volatility trend (rate of change - key for regime shifts)
    feat['vol_trend_5'] = vol_5.pct_change(5)
    feat['vol_trend_10'] = vol_10.pct_change(10)
    feat['vol_chg_5_20'] = (vol_5 - vol_20) / (vol_20 + 1e-9)

    # --- RSI (shorter period only) ---
    delta = close.diff()
    gain = delta.clip(lower=0).rolling(10).mean()
    loss = (-delta.clip(upper=0)).rolling(10).mean()
    rs = gain / (loss + 1e-9)
    feat['rsi_10'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands (shorter window) ---
    bb_mid_15 = close.rolling(15).mean()
    bb_std_15 = close.rolling(15).std()
    feat['bb_pct_15'] = (close - bb_mid_15) / (2 * bb_std_15 + 1e-9)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    atr_14 = tr.rolling(14).mean()
    feat['atr_14'] = atr_14
    feat['atr_norm'] = atr_14 / close

    # --- MACD (shorter spans) ---
    ema8 = close.ewm(span=8, adjust=False).mean()
    ema17 = close.ewm(span=17, adjust=False).mean()
    macd = ema8 - ema17
    feat['macd_signal'] = macd - macd.ewm(span=6, adjust=False).mean()

    # --- Volume (z-scored for regime invariance) ---
    vol_ma_10 = volume.rolling(10).mean()
    vol_std_10 = volume.rolling(10).std()
    feat['volume_z_10'] = (volume - vol_ma_10) / (vol_std_10 + 1e-9)
    feat['volume_ratio_10'] = volume / (vol_ma_10 + 1e-9)
    
    vol_ma_20 = volume.rolling(20).mean()
    feat['volume_ratio_20'] = volume / (vol_ma_20 + 1e-9)

    # VWAP (shorter window)
    vwap_15 = (close * volume).rolling(15).sum() / (volume.rolling(15).sum() + 1e-9)
    feat['vwap_dev'] = (close - vwap_15) / (close + 1e-9)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Regime features (shorter lookbacks) ---
    feat['trend_strength_10'] = (close / close.shift(10) - 1).abs() * 10000
    feat['trend_strength_20'] = (close / close.shift(20) - 1).abs() * 10000

    # Volatility regime with shorter windows
    feat['vol_regime'] = vol_10 / (vol_10.rolling(60).mean() + 1e-9)
    feat['vol_ratio_10_20'] = vol_10 / (vol_20 + 1e-9)
    
    # Volatility regime change (rate of change)
    feat['vol_regime_chg'] = feat['vol_regime'].pct_change(10)

    # --- Mean reversion features (shorter windows) ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    
    feat['price_sma5_dev'] = (close - sma_5) / (sma_5 + 1e-9)
    feat['price_sma10_dev'] = (close - sma_10) / (sma_10 + 1e-9)
    feat['price_sma20_dev'] = (close - sma_20) / (sma_20 + 1e-9)
    feat['sma_cross_5_20'] = (sma_5 - sma_20) / (sma_20 + 1e-9)

    # --- Efficiency / Trend Consistency (key for regime changes) ---
    feat['efficiency_10'] = (close - close.shift(10)).abs() / (close.diff().abs().rolling(10).sum() + 1e-9)
    feat['efficiency_20'] = (close - close.shift(20)).abs() / (close.diff().abs().rolling(20).sum() + 1e-9)
    
    # Efficiency change (rate of change)
    feat['efficiency_chg_10'] = feat['efficiency_10'].pct_change(10)

    # --- Momentum × Volatility interactions ---
    ret_3 = close.pct_change(3)
    feat['momentum_vol_interaction'] = ret_3 * vol_10

    # Z-scored momentum for regime invariance
    ret_3_mean = ret_3.rolling(20).mean()
    ret_3_std = ret_3.rolling(20).std()
    feat['momentum_z_20'] = (ret_3 - ret_3_mean) / (ret_3_std + 1e-9)

    # Momentum × regime interaction
    feat['momentum_regime'] = ret_3 * feat['vol_regime']

    # --- Volume × Price position interaction ---
    feat['volume_bb_interaction'] = feat['volume_ratio_10'] * feat['bb_pct_15']
    feat['volume_price_pressure'] = feat['volume_ratio_20'] * feat['price_sma10_dev']

    # --- High-low range features ---
    hl_range = (high - low) / close
    feat['hl_range'] = hl_range
    feat['hl_range_ma'] = hl_range.rolling(10).mean()
    feat['candle_body_ratio'] = (close - df['open']).abs() / ((high - low) + 1e-9)

    # --- Close position in range ---
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat['close_position_20'] = (close - low_20) / (high_20 - low_20 + 1e-9)

    # --- Acceleration and correlation ---
    feat['ret_accel_6'] = ret_1.diff(6)
    feat['vol_ret_corr_20'] = log_ret.rolling(20).corr(volume.pct_change(1))

    # --- Orderbook (if real data available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + 1e-9)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau3'] = df.apply(lambda r: obi(r, 3), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()
        feat['funding_momentum'] = fr.diff(4)
        feat['funding_x_volregime'] = fr * feat['vol_regime']

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
