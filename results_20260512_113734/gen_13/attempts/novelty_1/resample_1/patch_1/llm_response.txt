def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build feature matrix from raw OHLCV + orderbook + funding data.
    All features must use only past data (no lookahead).
    Returns a DataFrame aligned to df.index.
    """
    feat = pd.DataFrame(index=df.index)

    close = df['close']
    volume = df['volume']
    high = df['high']
    low = df['low']
    eps = 1e-9

    # --- Returns (shorter horizons for regime adaptation) ---
    ret_1 = close.pct_change(1)
    feat['ret_1'] = ret_1
    for lag in [2, 3, 5, 8, 12, 20]:
        feat[f'ret_{lag}'] = close.pct_change(lag)

    # --- Z-scored returns (regime-invariant momentum) ---
    for w in [5, 10, 20]:
        mu = ret_1.rolling(w).mean()
        sd = ret_1.rolling(w).std()
        feat[f'ret_z_{w}'] = (ret_1 - mu) / (sd + eps)

    # --- Volatility with shorter windows ---
    log_ret = np.log(close / close.shift(1))
    rv_5 = log_ret.rolling(5).std()
    rv_10 = log_ret.rolling(10).std()
    rv_20 = log_ret.rolling(20).std()
    
    feat['vol_5'] = rv_5
    feat['vol_10'] = rv_10
    feat['vol_20'] = rv_20
    
    # Volatility TREND (rate of change in vol regime)
    feat['vol_trend_5_10'] = (rv_5 - rv_10) / (rv_10 + eps)
    feat['vol_trend_10_20'] = (rv_10 - rv_20) / (rv_20 + eps)

    # --- Volatility regime (normalized) ---
    vol_base = rv_20.rolling(100).mean()  # shorter than 200
    feat['vol_regime'] = rv_20 / (vol_base + eps)
    feat['vol_regime_fast'] = rv_10 / (rv_20 + eps)
    feat['vol_regime_change'] = feat['vol_regime'].diff(3)
    
    # Z-score vol regime for regime invariance
    feat['vol_regime_z'] = (feat['vol_regime'] - feat['vol_regime'].rolling(40).mean()) / (feat['vol_regime'].rolling(40).std() + eps)

    # --- RSI (keep standard) ---
    for period in [6, 14]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + eps)
        feat[f'rsi_{period}'] = 100 - 100 / (1 + rs)

    # --- Bollinger Bands (shorter window) ---
    bb_mid_10 = close.rolling(10).mean()
    bb_std_10 = close.rolling(10).std()
    feat['bb_pct_10'] = (close - bb_mid_10) / (2 * bb_std_10 + eps)
    
    bb_mid_20 = close.rolling(20).mean()
    bb_std_20 = close.rolling(20).std()
    feat['bb_pct_20'] = (close - bb_mid_20) / (2 * bb_std_20 + eps)

    # --- ATR ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    feat['atr_14'] = tr.rolling(14).mean()
    feat['atr_norm'] = feat['atr_14'] / (close + eps)

    # --- MACD (shorter periods) ---
    ema8 = close.ewm(span=8, adjust=False).mean()
    ema21 = close.ewm(span=21, adjust=False).mean()
    macd = ema8 - ema21
    feat['macd'] = macd / (close + eps)
    feat['macd_signal'] = (macd - macd.ewm(span=5, adjust=False).mean()) / (close + eps)

    # --- Volume ---
    feat['volume_ratio_5'] = volume / (volume.rolling(5).mean() + eps)
    feat['volume_ratio_20'] = volume / (volume.rolling(20).mean() + eps)
    feat['volume_z_20'] = (volume - volume.rolling(20).mean()) / (volume.rolling(20).std() + eps)
    
    vwap_10 = (close * volume).rolling(10).sum() / (volume.rolling(10).sum() + eps)
    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + eps)
    feat['vwap_dev_10'] = (close - vwap_10) / (close + eps)
    feat['vwap_dev_20'] = (close - vwap_20) / (close + eps)

    # --- Session / Time ---
    feat['hour_sin'] = np.sin(2 * np.pi * df.index.hour / 24)
    feat['hour_cos'] = np.cos(2 * np.pi * df.index.hour / 24)
    feat['dow_sin'] = np.sin(2 * np.pi * df.index.dayofweek / 7)
    feat['dow_cos'] = np.cos(2 * np.pi * df.index.dayofweek / 7)
    feat['is_weekend'] = (df.index.dayofweek >= 5).astype(float)

    # --- Trend features (shorter lookbacks) ---
    for w in [5, 10, 20]:
        feat[f'trend_{w}'] = close / close.shift(w) - 1
        feat[f'trend_abs_{w}'] = feat[f'trend_{w}'].abs()

    # --- Efficiency (trend consistency) ---
    for w in [5, 10, 20]:
        denom = close.diff().abs().rolling(w).sum()
        feat[f'efficiency_{w}'] = (close - close.shift(w)).abs() / (denom + eps)
    
    # Efficiency TREND (changing market microstructure)
    feat['efficiency_trend'] = feat['efficiency_5'] - feat['efficiency_20']
    feat['trend_accel_5'] = feat['trend_5'].diff(2)

    # --- Autocorrelation (regime shift detector) ---
    feat['ret_autocorr_5'] = ret_1.rolling(5).corr(ret_1.shift(1))
    feat['ret_autocorr_10'] = ret_1.rolling(10).corr(ret_1.shift(1))
    feat['autocorr_trend'] = feat['ret_autocorr_5'] - feat['ret_autocorr_10']

    # --- Mean reversion (shorter windows) ---
    sma_5 = close.rolling(5).mean()
    sma_10 = close.rolling(10).mean()
    sma_20 = close.rolling(20).mean()
    feat['price_sma5_dev'] = (close - sma_5) / (sma_5 + eps)
    feat['price_sma10_dev'] = (close - sma_10) / (sma_10 + eps)
    feat['price_sma20_dev'] = (close - sma_20) / (sma_20 + eps)
    feat['sma5_sma20_cross'] = (sma_5 - sma_20) / (sma_20 + eps)

    # --- Momentum × Volatility interactions (normalized) ---
    ret_3 = close.pct_change(3)
    ret_5 = close.pct_change(5)
    feat['momentum_vol_interaction_3'] = ret_3 * feat['vol_regime']
    feat['momentum_vol_interaction_5'] = ret_5 * feat['vol_regime']
    feat['momentum_regime_3'] = ret_3 / (rv_10 + eps)
    feat['momentum_regime_5'] = ret_5 / (rv_20 + eps)
    feat['momentum_regime_fast'] = ret_3 * feat['vol_regime_fast']

    # --- Volume × Price interactions ---
    vol_ratio = volume / (volume.rolling(20).mean() + eps)
    feat['volume_bb_interaction'] = vol_ratio * feat['bb_pct_20']
    feat['volume_price_pressure'] = vol_ratio * feat['price_sma10_dev']
    feat['price_vol_pressure'] = feat['price_sma10_dev'] * feat['vol_regime']

    # --- High-low range features ---
    hl_range = (high - low) / (close + eps)
    feat['hl_range'] = hl_range
    feat['hl_range_ma5'] = hl_range.rolling(5).mean()
    feat['hl_range_ma10'] = hl_range.rolling(10).mean()
    feat['hl_range_change'] = feat['hl_range_ma5'] - feat['hl_range_ma10']
    
    feat['candle_body_ratio'] = (close - df['open']).abs() / ((high - low) + eps)
    feat['upper_wick_ratio'] = (high - df[['open', 'close']].max(axis=1)) / ((high - low) + eps)
    feat['lower_wick_ratio'] = (df[['open', 'close']].min(axis=1) - low) / ((high - low) + eps)

    # --- Close position in range (shorter windows) ---
    low_10 = low.rolling(10).min()
    high_10 = high.rolling(10).max()
    low_20 = low.rolling(20).min()
    high_20 = high.rolling(20).max()
    feat['close_pos_10'] = (close - low_10) / (high_10 - low_10 + eps)
    feat['close_pos_20'] = (close - low_20) / (high_20 - low_20 + eps)

    # --- Cross-feature change indicators ---
    feat['trend_x_vol_change'] = feat['trend_10'] * feat['vol_regime_change']
    feat['efficiency_x_vol'] = feat['efficiency_10'] * feat['vol_regime']
    feat['autocorr_x_vol'] = feat['autocorr_trend'] * feat['vol_regime']

    # --- Orderbook (if available) ---
    if 'bids' in df.columns and 'asks' in df.columns:
        def obi(row, tau=1):
            try:
                bids = np.array(row['bids'])
                asks = np.array(row['asks'])
                if bids.size == 0 or asks.size == 0:
                    return 0.0
                bid_vol = np.sum(bids[:, 1] * np.exp(-tau * np.arange(len(bids))))
                ask_vol = np.sum(asks[:, 1] * np.exp(-tau * np.arange(len(asks))))
                return (bid_vol - ask_vol) / (bid_vol + ask_vol + eps)
            except Exception:
                return 0.0
        feat['obi_tau1'] = df.apply(lambda r: obi(r, 1), axis=1)
        feat['obi_tau2'] = df.apply(lambda r: obi(r, 2), axis=1)

    # --- Funding rate ---
    if 'funding_rate' in df.columns:
        fr = df['funding_rate']
        feat['funding_rate'] = fr
        feat['funding_8h_ma'] = fr.rolling(32).mean()
        feat['funding_momentum'] = fr.diff(4)
        feat['funding_z_32'] = (fr - fr.rolling(32).mean()) / (fr.rolling(32).std() + eps)
        feat['funding_x_volregime'] = fr * feat['vol_regime']

    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    return feat
