<DESCRIPTION>
The current XGBoost approach suffers from severe prediction collapse (pred_std_ratio=0.04) and regime-specific overfitting (val_slope only 21% of train_slope). Instead of trying to fix XGBoost's regularization, I'm proposing a fundamentally different algorithmic approach:

**Ensemble of Regime-Adaptive Linear Models with Dynamic Weighting**

Key innovations:
1. **Regime Detection**: Use online clustering (exponentially-weighted covariance) to detect market regimes in real-time based on volatility, trend, and volume characteristics.

2. **Regime-Specific Models**: Train separate ridge regression models for each detected regime. Linear models generalize better across regime boundaries than deep trees.

3. **Adaptive Weighting**: Use a meta-learner (lightweight gradient boosting) to dynamically weight regime predictions based on current market state, rather than hard regime assignment.

4. **Micro-Structure Features**: Focus on ultra-short lookbacks (1-10 bars) with heavy z-score normalization to capture patterns that persist across regimes.

5. **Ensemble Diversity**: Combine 3 sub-strategies:
   - Momentum-based linear model (short-term continuation)
   - Mean-reversion linear model (price extremes)
   - Volatility-adjusted linear model (regime-normalized returns)

Why this should work better:
- Linear models have lower variance → better generalization
- Regime-specific training captures local patterns without global overfitting
- Dynamic weighting adapts to regime transitions (the val period problem)
- Ultra-short features decay faster but are more stable across time
- Ensemble reduces model-specific biases

This is a complete paradigm shift from tree-based learning to adaptive linear ensembles with explicit regime modeling.

def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Build ultra-short lookback features with heavy normalization.
    Focus on micro-structure that generalizes across regimes.
    """
    feat = pd.DataFrame(index=df.index)
    
    close = df["close"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)
    open_ = df["open"].astype(float)
    
    eps = 1e-9
    
    # Ultra-short returns (1-10 bars)
    for lag in [1, 2, 3, 5, 8, 10]:
        ret = close.pct_change(lag)
        feat[f"ret_{lag}"] = ret
        # Z-score within very short window
        feat[f"ret_{lag}_z10"] = (ret - ret.rolling(10).mean()) / (ret.rolling(10).std() + eps)
    
    # Micro volatility (5-20 bars only)
    log_ret = np.log(close / close.shift(1))
    for w in [5, 10, 20]:
        vol = log_ret.rolling(w).std()
        feat[f"vol_{w}"] = vol
        # Normalize by recent history
        feat[f"vol_{w}_z20"] = (vol - vol.rolling(20).mean()) / (vol.rolling(20).std() + eps)
    
    # Rate of change of volatility (regime transition signal)
    feat["vol_roc_5"] = feat["vol_5"].pct_change(3)
    feat["vol_roc_10"] = feat["vol_10"].pct_change(5)
    
    # Micro momentum (acceleration over 1-5 bars)
    ret1 = close.pct_change(1)
    feat["mom_accel_2"] = ret1.diff(2)
    feat["mom_accel_3"] = ret1.diff(3)
    feat["mom_accel_5"] = ret1.diff(5)
    
    # Efficiency over ultra-short windows
    for w in [5, 10]:
        net = (close - close.shift(w)).abs()
        path = close.diff().abs().rolling(w).sum()
        eff = net / (path + eps)
        feat[f"eff_{w}"] = eff
        feat[f"eff_{w}_z20"] = (eff - eff.rolling(20).mean()) / (eff.rolling(20).std() + eps)
    
    # Price position in micro range
    for w in [5, 10, 20]:
        low_w = low.rolling(w).min()
        high_w = high.rolling(w).max()
        pos = (close - low_w) / (high_w - low_w + eps)
        feat[f"pos_{w}"] = pos
        # Deviation from 0.5 (mean reversion signal)
        feat[f"pos_{w}_dev"] = pos - 0.5
    
    # Micro ATR normalized
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)
    
    for w in [5, 10]:
        atr = tr.rolling(w).mean()
        feat[f"atr_{w}_norm"] = atr / (close + eps)
        feat[f"atr_{w}_z20"] = (atr - atr.rolling(20).mean()) / (atr.rolling(20).std() + eps)
    
    # Volume regime (short window)
    for w in [5, 10, 20]:
        vol_ratio = volume / (volume.rolling(w).mean() + eps)
        feat[f"vol_ratio_{w}"] = vol_ratio
        feat[f"vol_ratio_{w}_z20"] = (vol_ratio - vol_ratio.rolling(20).mean()) / (vol_ratio.rolling(20).std() + eps)
    
    # Candle structure (normalized)
    candle_range = (high - low).replace(0, np.nan)
    feat["body_ratio"] = (close - open_).abs() / (candle_range + eps)
    feat["upper_shadow"] = (high - np.maximum(open_, close)) / (candle_range + eps)
    feat["lower_shadow"] = (np.minimum(open_, close) - low) / (candle_range + eps)
    
    # Momentum × volatility interaction (regime-invariant)
    ret3 = close.pct_change(3)
    vol10 = feat["vol_10"]
    feat["mom_vol_ratio"] = ret3 / (vol10 + eps)
    feat["mom_vol_product"] = ret3 * vol10
    
    # VWAP deviation (micro)
    for w in [10, 20]:
        vwap = (close * volume).rolling(w).sum() / (volume.rolling(w).sum() + eps)
        dev = (close - vwap) / (close + eps)
        feat[f"vwap_dev_{w}"] = dev
        feat[f"vwap_dev_{w}_z20"] = (dev - dev.rolling(20).mean()) / (dev.rolling(20).std() + eps)
    
    # Autocorrelation (market memory, short window)
    feat["autocorr_10"] = ret1.rolling(10).corr(ret1.shift(1))
    feat["autocorr_20"] = ret1.rolling(20).corr(ret1.shift(1))
    
    # RSI-like (short period)
    for period in [5, 10]:
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(period).mean()
        loss = (-delta.clip(upper=0)).rolling(period).mean()
        rs = gain / (loss + eps)
        rsi = 100 - 100 / (1 + rs)
        feat[f"rsi_{period}"] = (rsi - 50) / 50  # Normalize to [-1, 1]
    
    # Time features (session effects)
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24.0)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24.0)
    feat["dow_sin"] = np.sin(2 * np.pi * df.index.dayofweek / 7.0)
    
    # Funding if available
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding"] = fr
        feat["funding_z20"] = (fr - fr.rolling(20).mean()) / (fr.rolling(20).std() + eps)
        feat["funding_mom"] = fr.diff(4)
    
    # Cleanup
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()
    
    # Limit features
    if feat.shape[1] > 75:
        feat = feat.iloc[:, :75]
    
    return feat


def detect_regimes(X: np.ndarray, n_regimes: int = 3) -> np.ndarray:
    """
    Detect market regimes using K-means on volatility, trend, and volume features.
    Returns regime labels for each sample.
    """
    # Extract regime-defining features (first few columns are returns/vol)
    # Use volatility, momentum, and volume characteristics
    if X.shape[0] < n_regimes * 10:
        return np.zeros(X.shape[0], dtype=int)
    
    # Select features for regime detection (vol, momentum, volume)
    regime_features = []
    n_features = min(15, X.shape[1])  # Use first 15 features
    regime_features = X[:, :n_features]
    
    # Standardize
    scaler = StandardScaler()
    regime_features_scaled = scaler.fit_transform(regime_features)
    
    # K-means clustering
    kmeans = KMeans(n_clusters=n_regimes, random_state=42, n_init=10)
    labels = kmeans.fit_predict(regime_features_scaled)
    
    return labels
