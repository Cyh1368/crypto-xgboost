<DESCRIPTION>
This version switches from long-horizon indicator stacking to a regime-robust, short-memory feature design. Instead of relying on absolute levels from 20–60 bar windows, it emphasizes rolling z-scores, short lookbacks, and rate-of-change features that are more stable across market regimes. It also adds a few compact interaction terms such as momentum × volatility regime, and uses a more conservative XGBoost configuration to reduce overfitting while preserving enough expressivity to avoid prediction collapse. The goal is to improve validation slope and predicted dispersion by making features more stationary and less dependent on the training regime.

# evolve/initial.py

import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.preprocessing import StandardScaler
import sys, os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (hyperparameters)
XGB_PARAMS = {
    "n_estimators": 260,
    "max_depth": 3,
    "min_child_weight": 8,
    "reg_alpha": 0.8,
    "reg_lambda": 4.0,
    "subsample": 0.75,
    "colsample_bytree": 0.75,
    "learning_rate": 0.045,
    "gamma": 0.15,
    "objective": "reg:squarederror",
    "eval_metric": "rmse",
    "random_state": 42,
    "tree_method": "hist",
    "early_stopping_rounds": 30,
}

CLIP_PERCENTILE = 95
MIN_PRED_STD_RATIO = 0.18
# EVOLVE-BLOCK-END


# ─────────────────────────────────────────────
# EVOLVE-BLOCK-START  (feature engineering)
def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Regime-robust feature set:
    - shorter lookbacks
    - rolling z-score normalization
    - change-of-regime features
    - compact multiplicative interactions
    All features use only past data.
    """
    feat = pd.DataFrame(index=df.index)

    close = df["close"].astype(float)
    open_ = df["open"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)

    log_close = np.log(close.replace(0, np.nan))
    log_ret1 = log_close.diff()
    ret1 = close.pct_change(1)

    def rolling_z(x, w):
        m = x.rolling(w).mean()
        s = x.rolling(w).std()
        return (x - m) / (s + 1e-9)

    def slope_like(x, w):
        return x / (x.shift(w) + 1e-9) - 1.0

    # --- short-horizon returns and normalized momentum ---
    for lag in [1, 2, 3, 5, 8, 12, 20]:
        feat[f"ret_{lag}"] = close.pct_change(lag)
        feat[f"ret_z_{lag}"] = rolling_z(ret1, max(8, lag * 2))
        feat[f"logret_sum_{lag}"] = log_ret1.rolling(lag).sum()

    # --- short memory trend / acceleration ---
    for w in [5, 10, 20]:
        feat[f"trend_{w}"] = slope_like(close, w)
        feat[f"trend_z_{w}"] = rolling_z(slope_like(close, w), 20)
        feat[f"mom_accel_{w}"] = close.pct_change(1).diff(w)

    # --- realized volatility and volatility regime change ---
    for w in [5, 10, 20]:
        rv = log_ret1.rolling(w).std()
        feat[f"vol_{w}"] = rv
        feat[f"vol_z_{w}"] = rolling_z(rv, 40)
        feat[f"vol_chg_{w}"] = rv / (rv.shift(w) + 1e-9) - 1.0

    rv_10 = log_ret1.rolling(10).std()
    rv_20 = log_ret1.rolling(20).std()
    rv_40 = log_ret1.rolling(40).std()
    feat["vol_regime"] = rv_10 / (rv_40 + 1e-9)
    feat["vol_regime_z"] = rolling_z(feat["vol_regime"], 60)

    # --- range / candle structure ---
    tr = pd.concat([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ], axis=1).max(axis=1)

    feat["hl_range"] = (high - low) / (close + 1e-9)
    feat["atr_5"] = tr.rolling(5).mean() / (close + 1e-9)
    feat["atr_10"] = tr.rolling(10).mean() / (close + 1e-9)
    feat["atr_z_10"] = rolling_z(feat["atr_10"], 40)

    body = (close - open_).abs()
    feat["candle_body"] = body / (tr + 1e-9)
    feat["upper_wick"] = (high - np.maximum(open_, close)) / (tr + 1e-9)
    feat["lower_wick"] = (np.minimum(open_, close) - low) / (tr + 1e-9)

    # --- position within recent range ---
    for w in [10, 20]:
        lo = low.rolling(w).min()
        hi = high.rolling(w).max()
        feat[f"close_pos_{w}"] = (close - lo) / (hi - lo + 1e-9)
        feat[f"close_pos_z_{w}"] = rolling_z(feat[f"close_pos_{w}"], 30)

    # --- mean reversion / deviation from short-term fair value ---
    for w in [5, 10, 20]:
        sma = close.rolling(w).mean()
        ema = close.ewm(span=w, adjust=False).mean()
        feat[f"dev_sma_{w}"] = (close - sma) / (sma + 1e-9)
        feat[f"dev_ema_{w}"] = (close - ema) / (ema + 1e-9)

    # --- efficiency / path dependence ---
    for w in [10, 20]:
        net_move = (close - close.shift(w)).abs()
        path = close.diff().abs().rolling(w).sum()
        feat[f"efficiency_{w}"] = net_move / (path + 1e-9)

    # --- autocorrelation / microstructure persistence ---
    for w in [10, 20]:
        feat[f"ret_autocorr_{w}"] = ret1.rolling(w).corr(ret1.shift(1))
        feat[f"vol_ret_corr_{w}"] = rv_10.rolling(w).corr(volume.pct_change(1))

    # --- volume features (shorter windows, normalized) ---
    vol_chg = volume.pct_change(1)
    for w in [5, 10, 20]:
        vma = volume.rolling(w).mean()
        feat[f"vol_ratio_{w}"] = volume / (vma + 1e-9)
        feat[f"vol_ratio_z_{w}"] = rolling_z(feat[f"vol_ratio_{w}"], 40)
        feat[f"vol_chg_z_{w}"] = rolling_z(vol_chg, max(10, w * 2))

    # --- VWAP-like pressure ---
    vwap_10 = (close * volume).rolling(10).sum() / (volume.rolling(10).sum() + 1e-9)
    vwap_20 = (close * volume).rolling(20).sum() / (volume.rolling(20).sum() + 1e-9)
    feat["vwap_dev_10"] = (close - vwap_10) / (close + 1e-9)
    feat["vwap_dev_20"] = (close - vwap_20) / (close + 1e-9)

    # --- compact regime interactions ---
    feat["mom_x_volreg"] = feat["ret_3"] * feat["vol_regime"]
    feat["mom_x_volz"] = feat["ret_3"] * feat["vol_z_10"]
    feat["trend_x_volreg"] = feat["trend_10"] * feat["vol_regime"]
    feat["dev_x_volreg"] = feat["dev_sma_10"] * feat["vol_regime"]

    # --- time features ---
    feat["hour_sin"] = np.sin(2 * np.pi * df.index.hour / 24.0)
    feat["hour_cos"] = np.cos(2 * np.pi * df.index.hour / 24.0)
    feat["dow_sin"] = np.sin(2 * np.pi * df.index.dayofweek / 7.0)
    feat["dow_cos"] = np.cos(2 * np.pi * df.index.dayofweek / 7.0)
    feat["is_weekend"] = (df.index.dayofweek >= 5).astype(float)

    # --- funding / orderbook, if present ---
    if "funding_rate" in df.columns:
        fr = df["funding_rate"].astype(float)
        feat["funding_rate"] = fr
        feat["funding_z_32"] = rolling_z(fr, 32)
        feat["funding_chg_4"] = fr.diff(4)
        feat["funding_x_volreg"] = fr * feat["vol_regime"]
        feat["funding_x_mom"] = fr * feat["ret_3"]

    if "bids" in df.columns and "asks" in df.columns:
        def obi(row, tau=1.0):
            try:
                bids = np.asarray(row["bids"], dtype=float)
                asks = np.asarray(row["asks"], dtype=float)
                if bids.ndim != 2 or asks.ndim != 2 or bids.shape[1] < 2 or asks.shape[1] < 2:
                    return 0.0
                decay = np.exp(-tau * np.arange(min(len(bids), len(asks)), dtype=float))
                b = np.sum(bids[:len(decay), 1] * decay)
                a = np.sum(asks[:len(decay), 1] * decay)
                return (b - a) / (b + a + 1e-9)
            except Exception:
                return 0.0

        feat["obi_tau1"] = df.apply(lambda r: obi(r, 1.0), axis=1)
        feat["obi_tau2"] = df.apply(lambda r: obi(r, 2.0), axis=1)
        feat["obi_tau1_z"] = rolling_z(feat["obi_tau1"], 40)

    # --- final cleanup ---
    feat = feat.replace([np.inf, -np.inf], np.nan)
    feat = feat.dropna()

    # keep feature count comfortably under limit
    if feat.shape[1] > 80:
        feat = feat.iloc[:, :80]

    return feat

# run_experiment is the fixed interface called by evaluate.py

def run_experiment(
    X_train, y_train,
    X_test,  y_test,
    X_val,   y_val,
):
    """
    Train an XGBoost model and return raw predictions for all three splits.
    Called by evaluate.py via shinka.core.run_shinka_eval.
    """
    # Clip targets
    clip_val = np.percentile(np.abs(y_train), CLIP_PERCENTILE)
    y_train_c = y_train.clip(-clip_val, clip_val)

    # Scale
    scaler = StandardScaler()
    X_tr = scaler.fit_transform(X_train)
    X_te = scaler.transform(X_test)
    X_va = scaler.transform(X_val)

    params = {k: v for k, v in XGB_PARAMS.items()
              if k != 'early_stopping_rounds'}
    early = XGB_PARAMS.get('early_stopping_rounds', 30)

    model = xgb.XGBRegressor(**params, early_stopping_rounds=early)
    model.fit(
        X_tr, y_train_c,
        eval_set=[(X_te, y_test)],
        verbose=False,
    )

    return {
        'train': {'pred': model.predict(X_tr), 'actual': y_train.values},
        'test':  {'pred': model.predict(X_te), 'actual': y_test.values},
        'val':   {'pred': model.predict(X_va), 'actual': y_val.values},
        'model': model,
    }
